64 def validate(self, context: AgentContext) -> ValidationReport:
65 """Validation pure (réutilisable par la boucle du corrector_agent)."""
66 components = context.extra.get("skidl_components") or \
67 [c.__dict__ for c in context.bom]
68 nets = context.extra.get("skidl_nets") or []
69 refs = {c.get("ref") for c in components}
70 issues: list[ValidationIssue] = []
71
72 # 1) cohérence des nets
73 for net in nets:
74 connections = net.get("connections", [])
75 for ref, _pin in connections:
76 if ref not in refs:
77 issues.append(ValidationIssue(
78 code="NET-REF", severity="error",
79 message=f"ref {ref} absente du BOM (net {net['name']})",
80 ref=str(ref), rule="unknown_ref"))
81 if len(connections) < 2:
82 issues.append(ValidationIssue(
83 code="NET-FLOAT", severity="error",
84 message=f"net {net['name']} avec moins de 2 connexions",
85 rule="floating_net"))
86
87 # 2) ERC statique via adaptateur
88 erc = call_adapter("erc_executor",
89 {"components": components, "nets": nets})
90 for raw in erc.value or []:
91 issues.append(ValidationIssue(
92 code=str(raw.get("code", "ERC-000")), severity=str(raw.get("severity", "error")),
93 message=str(raw.get("message", "erreur ERC")),
94 ref=str(raw.get("ref", "")), rule=str(raw.get("rule", ""))))
95
96 # 3) empreintes contre la bibliothèque
97 for comp in components:
98 footprint = str(comp.get("footprint", ""))
99 if not footprint:
100 issues.append(ValidationIssue(
101 code="FP-MISSING", severity="error",
102 message=f"empreinte manquante pour {comp.get('ref')}",
103 ref=str(comp.get("ref", "")), rule="missing_footprint"))
104 elif not footprint.startswith(_KICAD_LIB_PREFIXES):
105 issues.append(ValidationIssue(
106 code="FP-LIB", severity="warning",
107 message=f"empreinte hors bibliothèque KiCad : {footprint}",
108 ref=str(comp.get("ref", ""))))
109
110 report = ValidationReport(
111 passed=not any(i.severity == "error" for i in issues),
112 issues=issues,
113 checked={"components": len(components), "nets": len(nets)})
114 if not report.passed and self.state_manager is not None:
115 for issue in issues:
116 if issue.severity == "error":
117 self.state_manager.publish_event(make_event(
118 EventType.ERC_ERROR, context.project_id, self.namename,
119 code=issue.code, message=issue.message, ref=issue.ref))
120 return report

122 @staticmethod
123 def _justify(report: ValidationReport) -> list[str]:
124 errors = sum(1 for i in report.issues if i.severity == "error")
125 warnings = len(report.issues) - errors
126 justifications = [f"contrôles : {report.checked} — {errors} erreur(s), {warnings} warning(s)"]
127 if report.passed:
128 justifications.append("validation PASS — netlist prête pour le placement RL")
129 return justifications
