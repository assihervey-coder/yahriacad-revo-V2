39def _deaccent(text: str) -> str:
40 """Minuscules sans accents — rend la détection de mots-clés insensible à la casse."""
41 return "".join(char for char in unicodedata.normalize("NFD", (text or "").lower())
42 if unicodedata.category(char) != "Mn")

45def detect_blocks(text: str) -> List[BlockTemplate]:
46 """Sélectionne les blocs fonctionnels évoqués par la requête (mots-clés)."""
47 lowered = _deaccent(text)
48 picked = [block for block in all_blocks()
49 if any(_deaccent(keyword) in lowered for keyword in block.keywords)]
50 if not picked:
51 # repli minimal viable : carte générique alimentée + MCU
52 picked = [get_block("power_regulator"), get_block("mcu_stm32")]
53 logger.info("aucun mot-clé reconnu — repli carte générique")
54 # le régulateur est implicite dès qu'un MCU ou un capteur est présent
55 keys = {block.key for block in picked}
56 if "mcu_stm32" in keys and "power_regulator" not in keys:
57 picked.insert(0, get_block("power_regulator"))
58 return picked

61def llm_refine(text: str, draft: str, llm: LLMRefiner) -> str:
62 """Hook LLM optionnel : `llm(prompt) -> str`. Repli silencieux sur le brouillon.
63
64 Le LLM est une amélioration, jamais une dépendance : toute erreur, réponse
65 vide ou réponse non-script retombe sur le brouillon déterministe.
66 """
67 try:
68 prompt = (
69 f"Affine le script SKiDL suivant pour la demande « {text} ». "
70 "Garde les commentaires en français. Réponds UNIQUEMENT par le code Python.\n\n"
71 f"{draft}"
72 )
73 refined = llm(prompt)
74 if isinstance(refined, str) and "Part(" in refined and "generate_netlist" in refined:
75 logger.info("brouillon SKiDL affiné par le LLM optionnel")
76 return refined
77 logger.warning("réponse LLM inexploitable — brouillon déterministe conservé")
78 except Exception as exc: # réseau absent, quota, timeout... jamais bloquant
79 logger.warning("LLM optionnel indisponible — fallback déterministe", extra={"error": str(exc)})
80 return draft

83def natural_language_to_skidl(text: str, llm: Optional[LLMRefiner] = None) -> SkidlScript:
84 """Convertit une requête en langage naturel en script SKiDL exécutable."""
85 blocks = detect_blocks(text)
86 lines: List[str] = [
87 '"""Script SKiDL généré par pcb_ai_designer_v2 — parser.nl_to_skidl.',
88 "",
89 f"Requête : « {text.strip()} »",
90 "Format : Python (skidl) — lisible, versionnable, exécutable hors ligne.",
91 '"""',
92 "from skidl import Part, Net, generate_netlist",
93 "",
94 "# Alimentations globales du design",
95 "v5v = Net('5V') # rail brut (USB / batterie)",
96 "v3v3 = Net('3V3') # rail régulé",
97 "gnd = Net('GND')",
98 "nreset = Net('NRESET')",
99 "spi_mosi = Net('SPI_MOSI')",
100 "spi_miso = Net('SPI_MISO')",
101 "spi_sck = Net('SPI_SCK')",
102 "spi_nss = Net('SPI_NSS')",
103 "sda = Net('SDA')",
104 "scl = Net('SCL')",
105 "usb_dp = Net('USB_DP')",
106 "usb_dm = Net('USB_DM')",
107 "antenna = Net('ANTENNA')",
108 "motor_in1 = Net('MOTOR_IN1')",
109 "motor_in2 = Net('MOTOR_IN2')",
110 "motor_out1 = Net('MOTOR_OUT1')",
111 "motor_out2 = Net('MOTOR_OUT2')",
112 "",
113 ]
114 for index, block in enumerate(blocks, start=1):
115 lines.append(f"# ═══ Bloc {index}/{len(blocks)} : {block.title} " + "═" * max(0, 34 - len(block.title)))
116 lines.append(block.skidl_template.format(vdd="v3v3", gnd="gnd", p=str(index)))
117 lines.append("")
118 lines.append("# Génération de la netlist (export .net SPICE consommé par le parser)")
119 lines.append("generate_netlist()")
120
121 components = [comp for block in blocks for comp in block.components]
122 bom_usd = sum(comp.price_usd for comp in components)
123 explanation = (
124 "Architecture retenue : " + " → ".join(block.key for block in blocks)
125 + ". " + " ".join(block.explanation for block in blocks)
126 + f" BOM estimé : {len(components)} composants, {bom_usd:.2f} USD."
127 )
128 source = "\n".join(lines)
129 if llm is not None:
130 source = llm_refine(text, source, llm)
131 logger.info("script SKiDL généré", extra={"blocks": [b.key for b in blocks],
132 "components": len(components)})
133 return SkidlScript(python_source=source, functional_blocks=[b.key for b in blocks],
134 explanation=explanation)
