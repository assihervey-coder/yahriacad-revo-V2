18def generate_zephyr_header(assignments: List[PinAssignment]) -> Tuple[str, str]:
19 """pin_map_zephyr.h : #define PIN_MAP_* + bloc devicetree commenté.
20
21 Le contenu suit les conventions Zephyr (DT_NAMED_GPIO, devicetree overlay)
22 tout en restant un en-tête C simple compilable partout.
23 """
24 guard = "PCB_AI_DESIGNER_PIN_MAP_ZEPHYR_H"
25 lines: List[str] = [
26 "/*",
27 " * pin_map_zephyr.h — généré par pcb_ai_designer_v2/firmware_bridge",
28 " * Ne pas éditer à la main : toute modification de brochage dans le design",
29 " * déclenche une régénération automatique (événement FIRMWARE_REGENERATED).",
30 " */",
31 f"#ifndef {guard}",
32 f"#define {guard}",
33 "",
34 "#include <zephyr/kernel.h>",
35 "#include <zephyr/drivers/gpio.h>",
36 "",
37 ]
38
39 seen: set = set()
40 for a in assignments:
41 symbol = _symbol(a)
42 if symbol in seen:
43 continue
44 seen.add(symbol)
45 lines.append(f"#define PIN_MAP_{symbol:<28s} {a.pin:<8s}"
46 f" /* net: {a.net or 'n/c'} */")
47
48 # bloc devicetree overlay commenté — collage direct dans une board overlay
49 lines += [
50 "",
51 "/*",
52 " * Overlay devicetree correspondant (à coller dans app.overlay) :",
53 " *",
54 " * / {",
55 ]
56 seen.clear()
57 for a in assignments:
58 symbol = _symbol(a)
59 if symbol in seen or a.function in {"GND", "POWER_IN"}:
60 continue
61 seen.add(symbol)
62 lines.append(f" * zephyr_user: {symbol.lower()} {{")
63 lines.append(f" * gpios = <&gpio0 {_gpio_index(a.pin)} GPIO_ACTIVE_HIGH>;"
64 f" /* {a.net or 'n/c'} */")
65 lines.append(" * }")
66 lines += [" * };", " */", "", f"#endif /* {guard} */", ""]
67 return "pin_map_zephyr.h", "\n".join(lines)

70def _symbol(a: PinAssignment) -> str:
71 """Symbole C : NET normalisé — UART_TX_1, SPI_MOSI, GPIO_PA5..."""
72 base = (a.net or f"PIN_{a.pin}").upper()
73 cleaned = "".join(c if c.isalnum() else "_" for c in base)
74 cleaned = "_".join(part for part in cleaned.split("_") if part)
75 return cleaned
