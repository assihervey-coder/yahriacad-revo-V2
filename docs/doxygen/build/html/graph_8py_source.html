1"""Graphe d'intention — « mémoire de long terme » partagée par tous les agents.
2
3Stocke les nœuds (bloc fonctionnel, composant, net, zone) et les arêtes typées
4(alimente, mesure, contraint, sensitive_to_noise, beside_of, member_of,
5connecte) dans de simples dictionnaires — aucune base externe requise. Chaque
6mutation incrémente une version (modèle event-sourcing léger) : les agents
7peuvent donc détecter qu'ils travaillent sur une représentation périmée.
8Peuplé dès l'étape 1 (NL → SKiDL), il répond aussi à « pourquoi » via
9:meth:`IntentGraph.why` — intention posée + alternatives rejetées.
10"""

18# Nomenclature canonique des nœuds et arêtes (section 05.1 de la spécification)
19NODE_KINDS = ("functional_block", "component", "net", "zone")

21# Types d'arêtes canoniques + extension documentée :
22# alimente : net d'alim → composant
23# mesure : composant → net analogique qu'il surveille
24# contraint : zone → composant (keepout, thermique)
25# sensitive_to_noise: net sensible → source de bruit voisine
26# beside_of : préférence de voisinage spatial
27# member_of : composant → bloc fonctionnel (extension)
28# connecte : composant → net via ses pastilles (extension)
29EDGE_TYPES = (
30 "alimente", "mesure", "contraint", "sensitive_to_noise",
31 "beside_of", "member_of", "connecte",
32)

64 def __init__(self, project_id: str = "") -> None:
65 self.project_id = project_id
66 self._nodes: Dict[str, IntentNode] = {}
67 self._edges: Dict[tuple, IntentEdge] = {}

73 def _mutate(self) -> int:
74 """Incrémente la version à chaque mutation (contrat event-sourcing)."""

76 return self._version

78 def add_node(self, node_id: str, kind: str, label: str = "", **attrs: Any) -> IntentNode:
79 """Ajoute (ou met à jour) un nœud ; retourne le nœud stocké."""
80 if kind not in NODE_KINDS:
81 raise ValueError(f"kind de nœud inconnu : {kind}")
82 node = self._nodes.get(node_id)
83 if node is None:
84 node = IntentNode(id=node_id, kind=kind, label=label or node_id, attrs=dict(attrs))
85 self._nodes[node_id] = node
86 self._mutate()
87 elif attrs:
88 node.attrs.update(attrs)
89 self._mutate()
90 return node
91
92 def add_edge(self, src: str, dst: str, edge_type: str, **attrs: Any) -> IntentEdge:
93 """Ajoute une arête typée (idempotent sur le triplet src/dst/type)."""
94 if edge_type not in EDGE_TYPES:
95 raise ValueError(f"type d'arête inconnu : {edge_type}")
96 key = (src, dst, edge_type)
97 edge = self._edges.get(key)
98 if edge is None:
99 edge = IntentEdge(src=src, dst=dst, edge_type=edge_type, attrs=dict(attrs))
100 self._edges[key] = edge
101 self._mutate()
102 elif attrs:
103 edge.attrs.update(attrs)
104 self._mutate()
105 return edge

107 def set_intention(self, node_id: str, intention: str) -> None:

112 def reject_alternative(self, node_id: str, description: str) -> None:
113 """Journalise une alternative rejetée — contextuellement décisif pour why()."""
114 self._rejected.setdefault(node_id, []).append(description)

117 # ---- consultation --------------------------------------------------------
118 def neighbors(self, node_id: str, edge_type: Optional[str] = None) -> List[str]:
119 """Voisins d'un nœud (sortants + entrants), filtrables par type d'arête."""
120 found: List[str] = []
121 for (src, dst, etype) in self._edges:
122 if node_id not in (src, dst):
123 continue
124 if edge_type is not None and etype != edge_type:
125 continue
126 found.append(dst if src == node_id else src)
127 return sorted(set(found))
128
129 def why(self, node_id: str) -> Dict[str, Any]:
130 """Contexte décisionnel d'un nœud : intention, alternatives rejetées, voisinage."""
131 node = self._nodes[node_id]
132 return {
133 "node": {"id": node.id, "kind": node.kind, "label": node.label, "attrs": node.attrs},
134 "intention": self._intentions.get(node_id, ""),
135 "alternatives_rejetees": list(self._rejected.get(node_id, [])),
136 "voisinage": {
137 etype: self.neighbors(node_id, etype)
138 for etype in EDGE_TYPES
139 if self.neighbors(node_id, etype)
140 },
141 "version_graphe": self._version,
142 }
143
144 def node(self, node_id: str) -> Optional[IntentNode]:
145 return self._nodes.get(node_id)

148 def version(self) -> int:

152 def stats(self) -> Dict[str, int]:

155 by_kind[node.kind] = by_kind.get(node.kind, 0) + 1

158 # ---- synchronisation avec le modèle de design ----------------------------
159 def update_from_board(self, board: Any) -> int:
160 """Synchronise le graphe depuis un Board du modèle commun.
161
162 Règles déterministes : nœud composant/net/zone par entité, nœud bloc
163 fonctionnel regroupant ses composants (member_of), arêtes connecte
164 depuis net.connections, alimente pour les nets d'alim, mesure pour les
165 nets analogiques, contraint pour les zones recouvrant un composant.
166 Retourne le nombre de mutations appliquées.
167 """
168 before = self._version
169 for comp in board.components.values():
170 self.add_node(f"component/{comp.ref}", "component", comp.ref,
171 mpn=comp.mpn, power_w=comp.power_w, pins=comp.pins)
172 if comp.functional_block:
173 block_id = f"block/{comp.functional_block}"
174 self.add_node(block_id, "functional_block", comp.functional_block)
175 self.add_edge(f"component/{comp.ref}", block_id, "member_of")
176 for net in board.nets.values():
177 self.add_node(f"net/{net.name}", "net", net.name,
178 net_class=net.net_class, impedance=net.impedance_target_ohm)
179 for (ref, _pad) in net.connections:
180 self.add_edge(f"component/{ref}", f"net/{net.name}", "connecte")
181 if net.net_class == "power":
182 for (ref, _pad) in net.connections:
183 self.add_edge(f"net/{net.name}", f"component/{ref}", "alimente")
184 if net.net_class == "analog":
185 for (ref, _pad) in net.connections:
186 self.add_edge(f"component/{ref}", f"net/{net.name}", "mesure")
187 for zone in board.zones:
188 zid = f"zone/{zone.name}"
189 self.add_node(zid, "zone", zone.name, zone_kind=zone.kind, max_temp_c=zone.max_temp_c)
190 for ref, placement in board.placements.items():
191 comp = board.components.get(ref)
192 if comp is None:
193 continue
194 x1, y1, x2, y2 = comp.bounding_box(placement)
195 if x1 >= zone.x_min_mm and x2 <= zone.x_max_mm and y1 >= zone.y_min_mm and y2 <= zone.y_max_mm:
196 self.add_edge(zid, f"component/{ref}", "contraint")
197 return self._version - before

210 "rejected": {k: list(v) for k, v in self._rejected.items()},
211 }

213 def import_json(self, data: Dict[str, Any]) -> None:
214 """Restaure un état exporté — remplace intégralement le graphe courant."""
215 self.__init__(project_id=data.get("project_id", ""))
216 for raw in data.get("nodes", []):
217 self.add_node(raw["id"], raw["kind"], raw.get("label", ""), **raw.get("attrs", {}))
218 for raw in data.get("edges", []):
219 self.add_edge(raw["src"], raw["dst"], raw["edge_type"], **raw.get("attrs", {}))
220 self._intentions.update(data.get("intentions", {}))
221 for node_id, items in data.get("rejected", {}).items():
222 for item in items:
223 self.reject_alternative(node_id, item)
224 self._version = int(data.get("version", self._version))
