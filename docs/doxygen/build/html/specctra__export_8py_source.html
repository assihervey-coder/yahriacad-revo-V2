27def export_dsn(board: Any, design_name: str = "pcb_ai_design") -> str:
28 """Sérialise le design en texte Specctra DSN simplifié."""
29 lines: list[str] = []
30 lines.append(f"(dsn \"{design_name}\"")
31 lines.append(f" (resolution um 1000)")
32 lines.append(f" (boundary (rect pcb 0 0 {_fmt(board.width_mm)} {_fmt(board.height_mm)}))")
33 # placement — un (place ...) par composant avec rotation et couche
34 lines.append(" (placement")
35 for ref, placement in sorted(board.placements.items()):
36 layer = _LAYER_NAMES.get(placement.layer, "top")
37 lines.append(f" (place {ref} {_fmt(placement.x_mm)} {_fmt(placement.y_mm)} "
38 f"{_fmt(placement.rotation_deg)} {layer})")
39 lines.append(" )")
40 # netlist — pins sous la forme REF-PAD
41 lines.append(" (net")
42 for name, net in sorted(board.nets.items()):
43 pins = " ".join(f"{ref}-{pad}" for ref, pad in net.connections)
44 lines.append(f" (net {name} (pins {pins}))")
45 lines.append(" )")
46 lines.append(")")
47 return "\n".join(lines)

50def write_dsn(board: Any, path: Path, design_name: str = "pcb_ai_design") -> Path:

54 return path
