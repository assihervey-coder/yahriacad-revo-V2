20# Regex d'extraction SKiDL (tolérantes : ref='X', var = Part(...), var = Net(...))
21_RE_PART_REF = re.compile(r"ref\s*=\s*['\"]([A-Za-z]\w*)['\"]")
22_RE_PART_VAR = re.compile(r"(\w+)\s*=\s*Part\s*\‍(")
23_RE_NET_NAME = re.compile(r"Net\s*\‍(\s*['\"]([\w\-/+\.]+)['\"]")
24_RE_NET_VAR = re.compile(r"(\w+)\s*=\s*Net\s*\‍(")
25_RE_CONNECT = re.compile(r"(\w+)(?:\[[^\]]+\])?\s*\+=\s*(\w+)(?:\[[^\]]+\])?")

27_POWER_NET_RE = re.compile(r"(?i)(vcc|vdd|3v3|5v|1v8|2v5|vbat|vin|vout|vrail)")

46 def is_triggered(self, parts_lower: set, nets_lower: set) -> bool:
47 """Règle toujours active si aucun déclencheur n'est défini."""
48 if self.trigger_part is None and self.trigger_net is None:
49 return True
50 hit_part = self.trigger_part is not None and any(self.trigger_part.search(p) for p in parts_lower)
51 hit_net = self.trigger_net is not None and any(self.trigger_net.search(n) for n in nets_lower)
52 return hit_part or hit_net

95 """Vue plate du script : refs de parts, noms de nets, liens part↔net."""

97 parts: List[str] = field(default_factory=list)
98 nets: List[str] = field(default_factory=list)
99 links: List[Tuple[str, str]] = field(default_factory=list) # (net, part)

102def parse_skidl(script: str) -> SkidlSummary:
103 """Extraction robuste des Part/Net/connexions d'un script SKiDL."""
104 summary = SkidlSummary()
105 summary.parts = sorted(set(_RE_PART_REF.findall(script))
106 | {v for v in _RE_PART_VAR.findall(script)
107 if v.lower() not in {"net", "part"}})
108 summary.nets = sorted(set(_RE_NET_NAME.findall(script))
109 | {v for v in _RE_NET_VAR.findall(script)})
110 known_nets = {n.lower() for n in summary.nets}
111 for left, right in _RE_CONNECT.findall(script):
112 pair = (left, right)
113 lo, ro = left.lower(), right.lower()
114 if ro in known_nets and lo not in known_nets:
115 pair = (right, left) # normalise (net, part)
116 elif lo not in known_nets and ro not in known_nets:
117 continue # ligne non résolue — ignorée
118 if pair[0].lower() in known_nets and pair[1].lower() not in known_nets:
119 summary.links.append(pair)
120 return summary

126 def __init__(self, kg_client: Optional[Neo4jClient] = None) -> None:

129 def validate(self, script: str) -> List[Violation]:
130 """Liste des écarts (vide = script conforme aux règles canoniques)."""
131 summary = parse_skidl(script)
132 parts_lower = {p.lower() for p in summary.parts}
133 nets_lower = {n.lower() for n in summary.nets}
134 violations: List[Violation] = []
135 for rule in KNOWN_RULES:
136 if not rule.is_triggered(parts_lower, nets_lower):
137 continue
138 missing_part = rule.require_part is not None and not any(
139 rule.require_part.search(p) for p in parts_lower)
140 missing_net = rule.require_net is not None and not any(
141 rule.require_net.search(n) for n in nets_lower)
142 if missing_part or missing_net:
143 what = "partie requise absente" if missing_part else "net requis absent"
144 violations.append(Violation(
145 code=rule.code, detail=f"{rule.description} — {what}",
146 pattern=rule.name, hint=rule.hint))
147 return violations

149 def learn_validated(self, script: str) -> bool:
150 """Si le script est conforme, stocke son motif dans le graphe ( apprentissage)."""
151 if self.validate(script):
152 return False
153 summary = parse_skidl(script)
154 pattern = {
155 "tags": ["skidl", "validated", "design_pattern"],
156 "parts": summary.parts, "nets": summary.nets, "links": summary.links,
157 "learned_at": time.time(),
158 }
159 self.kg.store_pattern(pattern)
160 return True

162 def validate_against_learned(self, script: str) -> List[Violation]:
163 """Compare aux motifs appris : signale un design qui s'écarte du corpus."""
164 summary = parse_skidl(script)
165 nets_lower = {n.lower() for n in summary.nets}
166 patterns = self.kg.find_patterns(["validated", "skidl"])
167 if not patterns:
168 return []
169 best, best_ratio = None, -1.0
170 for pattern in patterns:
171 known = {str(n).lower() for n in pattern.get("nets", [])}
172 if not known:
173 continue
174 ratio = len(known & nets_lower) / len(known)
175 if ratio > best_ratio:
176 best, best_ratio = pattern, ratio
177 if best is not None and best_ratio < 0.4:
178 return [Violation(
179 code="deviates_from_learned",
180 detail=(f"Écart avec le motif validé « {best.get('id', '?')} » : "
181 f"{best_ratio:.0%} de nets communs seulement"),
182 pattern=str(best.get("id", "learned")),
183 hint="Revoir le plan multi-agents ou justifier l'écart d'architecture")]
184 return []
