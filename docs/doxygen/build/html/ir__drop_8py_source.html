1"""Chute de tension IR du réseau d'alimentation — analyse nodale numpy.
2
3Modèle (« chute = I·R cumulée depuis la source ») :
4 - nœuds = extrémités des segments routés du rail (arrondis à 10 µm) ;
5 - résistance d'un segment : R = R_sheet · (L / W) avec R_sheet = ρ_cu / ép_cuv
6 (≈ 4.9e-4 Ω/□ pour 35 µm) ; chaque via ajoute R_VIA fixe ;
7 - charges = composants connectés au rail, I = power_w / rail_v, injectées au
8 nœud « ancre » (relié par une piste) le plus proche ;
9 - sources = composants fonctionnels `power`/`regulator`/`pmic` (Dirichlet à
10 rail_v) ; à défaut, la première connexion du net sert de source ;
11 - rail non routé → chaîne linéaire synthétique source→charges (largeur par
12 défaut), afin de ne jamais retourner un résultat vide.
13
14Système G·V = b résolu en dense (numpy.linalg.solve) — le nombre de nœuds d'un
15rail reste modeste. Nets au-delà de 3 % de chute : flag + composants fautifs.
16"""

31RHO_CU = 1.72e-8 # Ω·m — résistivité du cuivre
32CU_THICKNESS_MM = 0.035 # 35 µm standard
33R_SHEET = RHO_CU / (CU_THICKNESS_MM * 1e-3) # ≈ 4.914e-4 Ω/□
34R_VIA_OHM = 1.0e-3 # résistance d'un via d'alimentation
35DEFAULT_TRACE_WIDTH_MM = 1.0
36LIMIT_PCT = 3.0 # consigne : chute ≤ 3 % du rail
37SOURCE_BLOCKS = {"power", "regulator", "pmic", "vrm"}
38NODE_QUANTUM_MM = 0.01

78def _pad_point(board, ref: str, pad_name: str) -> Tuple[float, float]:
79 """Position absolue d'une pastille — pads absolus si dans la carte, sinon centrés.
80
81 Le modèle commun ne fixe pas la convention pad local/absolu : on détecte
82 (pad dans l'emprise carte → absolu ; sinon → offset du placement).
83 """
84 comp = board.components[ref]
85 placement = board.placements.get(ref)
86 for pad in comp.pads:
87 if pad.name == pad_name:
88 inside = 0.0 <= pad.x_mm <= board.width_mm and 0.0 <= pad.y_mm <= board.height_mm
89 if inside or placement is None:
90 return (pad.x_mm, pad.y_mm)
91 return (placement.x_mm + pad.x_mm, placement.y_mm + pad.y_mm)
92 if placement is not None:
93 return (placement.x_mm, placement.y_mm)
94 return (board.width_mm / 2.0, board.height_mm / 2.0)

117 """Point d'ancrage le plus proche (nœud réellement relié par une piste)."""

121def analyze_ir_drop(
122 board,
123 net_name: Optional[str] = None,
124 rail_v: float = 3.3,
125 r_sheet_ohm_per_sq: float = R_SHEET,
126 r_via_ohm: float = R_VIA_OHM,
127 limit_pct: float = LIMIT_PCT,
128) -> IRDropResult:
129 """Résolution nodale du rail d'alimentation → IRDropResult.
130
131 Retourne la chute max, le pourcentage et les composants au-delà de
132 `limit_pct` (3 % par défaut). Jamais d'exception : rail absent/non routé →
133 résultat annoté (note) pour rester consommable par la boucle multiphysique.
134 """
135 t0 = time.perf_counter()
136 selected = _pick_power_net(board, net_name)
137 if selected is None:
138 return IRDropResult(note="aucun rail d'alimentation détecté")
139 net = board.nets[selected]
140
141 nodes: Dict[Tuple[float, float], int] = {}
142
143 def node_of(x: float, y: float) -> int:
144 key = (round(x / NODE_QUANTUM_MM) * NODE_QUANTUM_MM,
145 round(y / NODE_QUANTUM_MM) * NODE_QUANTUM_MM)
146 if key not in nodes:
147 nodes[key] = len(nodes)
148 return nodes[key]
149
150 edges: List[Tuple[int, int, float]] = [] # (i, j, R ohm)
151
152 if net.routed_segments:
153 for seg in net.routed_segments:
154 if seg.is_via:
155 if abs(seg.x1_mm - seg.x2_mm) > 1e-9 or abs(seg.y1_mm - seg.y2_mm) > 1e-9:
156 edges.append((node_of(seg.x1_mm, seg.y1_mm),
157 node_of(seg.x2_mm, seg.y2_mm), r_via_ohm))
158 else:
159 node_of(seg.x1_mm, seg.y1_mm) # via ponctuel → nœud de raccord
160 continue
161 length = seg.length_mm
162 if length < NODE_QUANTUM_MM:
163 node_of(seg.x1_mm, seg.y1_mm)
164 continue
165 width = seg.width_mm if seg.width_mm > 0 else DEFAULT_TRACE_WIDTH_MM
166 resistance = r_sheet_ohm_per_sq * (length / width)
167 edges.append((node_of(seg.x1_mm, seg.y1_mm),
168 node_of(seg.x2_mm, seg.y2_mm), resistance))
169
170 # ---- classification des connexions --------------------------------------
171 source_refs = [
172 ref for (ref, _pad) in net.connections
173 if (board.components[ref].functional_block or "").lower() in SOURCE_BLOCKS
174 ] or ([net.connections[0][0]] if net.connections else [])
175
176 loads: List[Tuple[str, float, Tuple[float, float]]] = []
177 source_points: List[Tuple[float, float]] = []
178 for ref, pad_name in net.connections:
179 point = _pad_point(board, ref, pad_name)
180 if ref in source_refs:
181 source_points.append(point)
182 elif board.components[ref].power_w > 0:
183 loads.append((ref, board.components[ref].power_w / rail_v, point))
184
185 if not source_refs:
186 return IRDropResult(net=selected, rail_v=rail_v,
187 note="aucune source d'alimentation identifiée")
188
189 # rail non routé : chaîne synthétique source → charges triées par distance
190 if not net.routed_segments and loads:
191 logger.info("rail non routé — estimation en chaîne linéaire", extra={"net": selected})
192 ordered = sorted(loads, key=lambda l: math.hypot(
193 l[2][0] - source_points[0][0], l[2][1] - source_points[0][1]))
194 chain: List[Tuple[float, float]] = [source_points[0]] + [l[2] for l in ordered]
195 width = _default_trace_width(net)
196 for a, b in zip(chain, chain[1:]):
197 length = math.hypot(b[0] - a[0], b[1] - a[1])
198 if length > NODE_QUANTUM_MM:
199 edges.append((node_of(*a), node_of(*b),
200 r_sheet_ohm_per_sq * (length / width)))
201
202 if not nodes:
203 return IRDropResult(net=selected, rail_v=rail_v,
204 note="rail sans géométrie exploitable")
205
206 # nœuds ancres = nœuds participant à au moins une piste (snapping des pads)
207 anchors = [k for k, i in nodes.items() if any(i in (a, b) for a, b, _r in edges)]
208 if not anchors:
209 return IRDropResult(net=selected, rail_v=rail_v,
210 note="rail sans segment de cuivre exploitable")
211
212 # ---- assemblage du système G·V = b ---------------------------------------
213 n = len(nodes)
214 G = np.zeros((n, n), dtype=np.float64)
215 b = np.zeros(n, dtype=np.float64)
216 for i, j, r in edges:
217 g = 1.0 / max(r, 1e-9)
218 G[i, i] += g; G[j, j] += g
219 G[i, j] -= g; G[j, i] -= g
220 for _ref, current, point in loads:
221 idx = nodes[_snap(point, anchors)]
222 b[idx] -= current
223
224 # Dirichlet : nœuds de source fixés à rail_v
225 source_nodes = {nodes[_snap(p, anchors)] for p in source_points}
226 for s in source_nodes:
227 G[s, :] = 0.0
228 G[s, s] = 1.0
229 b[s] = rail_v
230 # nœuds isolés (aucune piste, aucune charge) — régularisation numérique
231 diag_idx = np.arange(n)
232 isolated = np.diag(G) < 1e-12
233 if isolated.any():
234 G[diag_idx[isolated], diag_idx[isolated]] += 1.0
235
236 try:
237 V = np.linalg.solve(G, b)
238 converged = bool(np.all(np.isfinite(V)))
239 except np.linalg.LinAlgError:
240 V, *_ = np.linalg.lstsq(G, b, rcond=None)
241 converged = False
242
243 drops_mv: Dict[str, float] = {}
244 failing: List[str] = []
245 min_v = rail_v
246 for ref, _current, point in loads:
247 idx = nodes[_snap(point, anchors)]
248 v = float(V[idx])
249 min_v = min(min_v, v)
250 drop_mv = (rail_v - v) * 1000.0
251 drops_mv[ref] = round(drop_mv, 2)
252 if _flag_over_budget((rail_v - v) / rail_v * 100.0, limit_pct):
253 failing.append(ref)
254
255 ir_drop_mv = (rail_v - min_v) * 1000.0
256 ir_pct = (rail_v - min_v) / rail_v * 100.0 if rail_v > 0 else 0.0
257 if failing:
258 logger.warning("chute IR au-delà de la consigne",
259 extra={"net": selected, "pct": round(ir_pct, 2), "components": failing})
260
261 return IRDropResult(
262 net=selected, rail_v=rail_v,
263 source_ref=", ".join(sorted(set(source_refs))),
264 source_v=rail_v, min_v=round(min_v, 4),
265 ir_drop_mv=round(ir_drop_mv, 2), ir_drop_pct=round(ir_pct, 3),
266 failing_components=failing, component_drops_mv=drops_mv,
267 node_count=n, iterations=n, converged=converged,
268 note="",
269 )
