42def generate_headers(board, target: str = "zephyr") -> List[GeneratedHeader]:
43 """Génère les en-têtes C synchronisés avec le design.
44
45 target : "zephyr" → pin_map_zephyr.h (style devicetree commenté),

49 assignments = export_pins(board)
50 headers: List[GeneratedHeader] = []

59 headers.append(GeneratedHeader(filename=filename, c_source=source,
