1"""World model « DreamerV3-like » en numpy pur — le simulateur interne du cerveau.
2
3POURQUOI un world model : exécuter puis évaluer une action réelle (routage,
4DRC, thermique) coûte des secondes ; le prédire coûte des microsecondes. En
5apprivoisant une représentation prédictive de l'environnement, l'agent explore
6~2 ordres de grandeur d'hypothèses en plus à budget constant — c'est le cœur
7de l'approche Dreamer : « apprendre le monde, rêver dedans, agir au réveil ».
8
9Implémentation : réseau minimal 2 couches fully-connected (activation tanh,
10forward manuel SANS torch). La première couche est une projection aléatoire
11fixe (random features — approxime un noyau non linéaire), la seconde est
12ajustée par régression aux moindres carrés (ridge, forme fermée numpy) sur les
13expériences observées. AVANT tout entraînement, ``predict_consequences``
14retombe sur des heuristiques déterministes (densité, distances, puissance) —
15le code fonctionne donc intégralement hors ligne.
16"""

33class TinyNet:
34 """2 couches FC : tanh(X·W1 + b1)·W2 + b2 — W1 fixe, W2 appris par LS."""
35
36 def __init__(self, feature_dim: int, hidden: int = 32, output_dim: int = 3,
37 seed: int = 42) -> None:
38 rng = np.random.default_rng(seed)
39 self.W1 = rng.normal(0.0, 0.5, size=(feature_dim, hidden))
40 self.b1 = np.zeros(hidden)
41 self.W2 = np.zeros((hidden, output_dim))
42 self.b2 = np.zeros(output_dim)
43
44 def hidden(self, X: np.ndarray) -> np.ndarray:
45 """Activation de la couche cachée — tanh, forward manuel."""
46 return np.tanh(X @ self.W1 + self.b1)
47
48 def forward(self, X: np.ndarray) -> np.ndarray:
49 """Sortie (n, 3) : [congestion, thermal_rise, si_risk] estimés."""
50 return self.hidden(X) @ self.W2 + self.b2
51
52 def fit_ridge(self, X: np.ndarray, Y: np.ndarray, lam: float = _RIDGE_LAMBDA) -> None:
53 """Résout min ||H·W2 + b2 − Y||² + λ||W2||² en forme fermée."""
54 H = self.hidden(X)
55 Hb = np.hstack([H, np.ones((H.shape[0], 1))])
56 reg = lam * np.eye(Hb.shape[1])
57 solution, *_ = np.linalg.lstsq(Hb.T @ Hb + reg, Hb.T @ Y, rcond=None)
58 self.W2 = solution[:-1, :]
59 self.b2 = solution[-1, :]

62class DreamerWorldModel:
63 """Prédit {congestion_score, thermal_rise_estimate, si_risk} d'une action."""
64
65 def __init__(self, feature_dim: int = FEATURE_DIM, hidden: int = 32) -> None:
66 self.net = TinyNet(feature_dim, hidden)
67 self.trained = False
68 self._features: List[np.ndarray] = []

71 # ---- features -------------------------------------------------------------
72 def extract_features(self, board: Any, ref: str, x_mm: float, y_mm: float,
73 rotation_deg: float = 0.0, layer: int = 0) -> np.ndarray:
74 """Vecteur de contexte : position, distances, densité, puissance locale."""
75 comp = board.components[ref]
76 x1, y1, x2, y2 = comp.bounding_box(
77 Placement(ref=ref, x_mm=x_mm, y_mm=y_mm,
78 rotation_deg=rotation_deg, layer=layer))
79 cx, cy = (x1 + x2) / 2.0, (y1 + y2) / 2.0
80 min_dist, density, power_nearby = 1e9, 0.0, 0.0
81 for other_ref, other_p in board.placements.items():
82 if other_ref == ref or other_ref not in board.components:
83 continue
84 other = board.components[other_ref]
85 ox1, oy1, ox2, oy2 = other.bounding_box(other_p)
86 ocx, ocy = (ox1 + ox2) / 2.0, (oy1 + oy2) / 2.0
87 gap = max(0.0, max(ox1 - x2, x1 - ox2, oy1 - y2, y1 - oy2))
88 min_dist = min(min_dist, gap)
89 if math.hypot(cx - ocx, cy - ocy) <= 10.0:
90 density += 1.0
91 power_nearby += other.power_w
92 edge = min(cx, cy, board.width_mm - cx, board.height_mm - cy)
93 return np.array([
94 cx / board.width_mm, cy / board.height_mm, rotation_deg / 360.0,
95 float(layer), min(min_dist / 20.0, 1.0), density / 8.0,
96 min(power_nearby / 5.0, 1.0), min(comp.power_w / 2.0, 1.0),
97 (x2 - x1) * (y2 - y1) / 100.0, edge / max(1.0, min(board.width_mm, board.height_mm) / 2.0),
98 float(len(board.nets)) / 50.0,
99 1.0 if any(not (x2 <= z.x_min_mm or z.x_max_mm <= x1 or y2 <= z.y_min_mm
100 or z.y_max_mm <= y1) for z in board.zones if z.kind == "keepout")
101 else 0.0,
102 ], dtype=np.float64)

105 def predict_consequences(self, board: Any, action: Any) -> Dict[str, float]:

117 # ---- apprentissage ------------------------------------------------------------
118 def observe(self, board: Any, action: Any, metrics: Dict[str, float],
119 weight: float = 1.0) -> None:
120 """Stocke une expérience (features, conséquences réelles) — buffer LRU."""
121 features = self.extract_features(board, action.ref, action.x_mm, action.y_mm,
122 getattr(action, "rotation_deg", 0.0),
123 getattr(action, "layer", 0))
124 target = np.array([metrics.get("congestion_score", 0.0),
125 metrics.get("thermal_rise_estimate", 0.0),
126 metrics.get("si_risk", 0.0)]) * weight
127 self._features.append(features)
128 self._targets.append(np.clip(target, 0.0, 2.0))
129 if len(self._features) > _BUFFER_MAX: # LRU : oublie les plus anciennes
130 self._features.pop(0)
131 self._targets.pop(0)

133 def train(self, min_samples: int = 4) -> int:
134 """Régression ridge sur le buffer ; retourne le nb d'échantillons utilisés."""
135 if len(self._features) < max(4, min_samples):
136 return 0
137 X = np.vstack(self._features)
138 Y = np.vstack(self._targets)
139 self.net.fit_ridge(X, Y)
140 self.trained = True
141 return X.shape[0]

145 if X.shape[0] < 4:
146 return False
147 self.net.fit_ridge(X, Y)

173def _heuristic_consequences(features: np.ndarray) -> Dict[str, float]:
174 """Fallback déterministe — les mêmes signaux, calculés à la main.
175
176 indices : 4=distance min, 5=densité locale, 6=puissance voisine,
177 7=puissance propre, 8=aire, 9=proximité du bord, 11=keepout touché.
178 """
179 min_dist, density = features[4], features[5]
180 power_nearby, own_power = features[6], features[7]
181 edge, keepout_hit = features[9], features[11]
182 congestion = float(np.clip(0.35 * density + 0.25 * (1.0 - min_dist)
183 + 0.2 * keepout_hit + 0.2 * (1.0 - edge), 0.0, 1.0))
184 thermal = float(np.clip(0.45 * power_nearby + 0.35 * own_power + 0.2 * density, 0.0, 1.0))
185 si_risk = float(np.clip(0.5 * (1.0 - min_dist) + 0.3 * density + 0.2 * keepout_hit, 0.0, 1.0))
186 return {"congestion_score": congestion, "thermal_rise_estimate": thermal, "si_risk": si_risk}
