61 def arbitrate(self, constraint_a: dict[str, Any],
62 constraint_b: dict[str, Any]) -> ConflictResolution:
63 """Compare deux contraintes {"family", "name", ...} et tranche.
64
65 Égalité de famille (imprévu) : la première contrainte (a) gagne —
66 convention documentée, la décision reste auditable.
67 """
68 family_a = str(constraint_a.get("family", "cost")).lower()
69 family_b = str(constraint_b.get("family", "cost")).lower()
70 rank_a = self._rank(family_a)
71 rank_b = self._rank(family_b)
72 if rank_a <= rank_b:
73 winner, loser = family_a, family_b
74 win, lose = constraint_a, constraint_b
75 else:
76 winner, loser = family_b, family_a
77 win, lose = constraint_b, constraint_a
78 resolution = ConflictResolution(
79 constraint_a=constraint_a, constraint_b=constraint_b,
80 winner=winner, loser=loser,
81 resolution=(f"priorité à « {win.get('name', winner)} » sur "
82 f"« {lose.get('name', loser)} »"),
83 rationale=_RATIONALES[winner],
84 project_id=self.project_id,
85 )
86 self.history.append(resolution)
87 self._record_in_intent_graph(resolution)
88 return resolution

90 def _record_in_intent_graph(self, resolution: ConflictResolution) -> None:
91 """Chaque décision est auditable dans le graphe d'intention."""

93 "type": "conflict_arbitration",

97 "rationale": resolution.rationale,
98 })
99 if result.source == "fallback":
100 logger.info("arbitrage enregistré localement (graphe d'intention absent)",
101 extra=resolution.to_json())
