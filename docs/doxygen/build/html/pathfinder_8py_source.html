48 def __init__(self, board: Board, grid_step_mm: float = GRID_STEP_MM,
49 layer_allowlist: Optional[List[int]] = None) -> None:
50 self.board = board
51 self.step = grid_step_mm
52 self.layers = layer_allowlist or [l.index for l in board.layers if l.is_copper]
53 self.nx = int(board.width_mm / grid_step_mm) + 1
54 self.ny = int(board.height_mm / grid_step_mm) + 1
55 # Usage de congestion par cellule (x, y, layer) — négociation PathFinder
56 self._usage: Dict[Tuple[int, int, int], int] = {}
57 # Cellules interdites : keepouts, périphérie, emprises de pads
58 self._blocked: set = set()
59 self._build_blocked()

62 def _to_cell(self, x_mm: float, y_mm: float) -> Tuple[int, int]:
63 return (max(0, min(self.nx - 1, int(round(x_mm / self.step)))),
64 max(0, min(self.ny - 1, int(round(y_mm / self.step)))))

66 def _to_mm(self, cell: Tuple[int, int]) -> Tuple[float, float]:
67 return (cell[0] * self.step, cell[1] * self.step)

69 def _build_blocked(self) -> None:
70 """Marque les keepouts et la périphérie comme inroutables."""
71 for zone in self.board.zones:
72 if zone.kind == "keepout":
73 cx_min, cy_min = self._to_cell(zone.x_min_mm, zone.y_min_mm)
74 cx_max, cy_max = self._to_cell(zone.x_max_mm, zone.y_max_mm)
75 for cx in range(cx_min, cx_max + 1):
76 for cy in range(cy_min, cy_max + 1):
77 for layer in self.layers:
78 self._blocked.add((cx, cy, layer))
79 edge = max(1, int(EDGE_CLEARANCE_MM / self.step))
80 for cx in range(self.nx):
81 for cy in range(self.ny):
82 if cx < edge or cy < edge or cx >= self.nx - edge or cy >= self.ny - edge:
83 for layer in self.layers:
84 self._blocked.add((cx, cy, layer))

86 def block_pads(self, radius_cells: int = 1) -> None:
87 """Bloque les cellules autour des pastilles (clearance de base).
88
89 Les cellules de départ/arrivée restent accessibles : A* exempte
90 toujours la cellule but (cf. `_astar`).
91 """
92 for ref, comp in self.board.components.items():
93 placement = self.board.placements.get(ref)
94 layer = placement.layer if placement else 0
95 for pad in comp.pads:
96 cx, cy = self._to_cell(pad.x_mm, pad.y_mm)
97 for ddx in range(-radius_cells, radius_cells + 1):
98 for ddy in range(-radius_cells, radius_cells + 1):
99 self._blocked.add((cx + ddx, cy + ddy, layer))

101 def block_cells(self, cells: List[Tuple[int, int, int]]) -> None:
102 """Bloque explicitement des cellules (contraintes du bus, re-route local)."""
103 self._blocked.update(cells)

106 def route_net(self, graph: ConnectivityGraph, net: Net) -> Optional[RouteResult]:
107 """Route un net : paires MST (Prim) reliées successivement par A*."""
108 pairs = graph.mst_pairs(net.name)
109 if not pairs:
110 logger.warning("net sans paire de pads — ignoré", extra={"net": net.name})
111 return None
112
113 segments: List[Segment] = []
114 cells_used = 0
115 vias = 0
116 for a, b in pairs:
117 path = self._astar(a, b, net)
118 if path is None:
119 logger.warning("net inroutable — escalade super_agent", extra={"net": net.name})
120 if segments:
121 return RouteResult(net=net.name, segments=segments,
122 cells_used=cells_used, vias=vias)
123 return None
124 segments.extend(self._path_to_segments(net, path))
125 cells_used += len(path)
126 vias += sum(1 for c1, c2 in zip(path, path[1:]) if c1[2] != c2[2])
127
128 # enregistre l'usage (pour la négociation des nets suivants)
129 for seg in segments:
130 for cell in self._segment_cells(seg):
131 self._usage[cell] = self._usage.get(cell, 0) + 1
132 return RouteResult(net=net.name, segments=segments, cells_used=cells_used, vias=vias)

134 def _astar(self, a: PadNode, b: PadNode, net: Net) -> Optional[List[Tuple[int, int, int]]]:
135 allow = set(self.layers) if not net.layer_allowlist \
136 else set(self.layers) & set(net.layer_allowlist) or set(self.layers)
137 start_layer = min(a.layer, max(self.layers))
138 goal_layer = min(b.layer, max(self.layers))
139 start = (*self._to_cell(a.x_mm, a.y_mm), start_layer)
140 goal = (*self._to_cell(b.x_mm, b.y_mm), goal_layer)
141
142 open_heap: List[Tuple[float, float, tuple]] = [(0.0, 0.0, start)]
143 g_score: Dict[tuple, float] = {start: 0.0}
144 came: Dict[tuple, tuple] = {}
145 closed: set = set()
146
147 def h(c: tuple) -> float:
148 dx = (c[0] - goal[0]) * self.step
149 dy = (c[1] - goal[1]) * self.step
150 dl = abs(c[2] - goal[2]) * VIA_COST
151 return math.hypot(dx, dy) + dl
152
153 while open_heap:
154 _, g, current = heapq.heappop(open_heap)
155 if current == goal:
156 return self._reconstruct(came, current)
157 if current in closed:
158 continue
159 closed.add(current)
160 for nxt in self._neighbors(current, allow):
161 if nxt in self._blocked and nxt != goal:
162 continue
163 move_cost = self.step
164 if nxt[2] != current[2]:
165 move_cost += VIA_COST
166 usage = self._usage.get(nxt, 0)
167 move_cost *= (1.0 + usage * (CONGESTION_GROWTH - 1.0))
168 tentative = g + move_cost
169 if tentative < g_score.get(nxt, math.inf):
170 g_score[nxt] = tentative
171 came[nxt] = current
172 heapq.heappush(open_heap, (tentative + h(nxt), tentative, nxt))
173 return None

175 def _neighbors(self, cell: tuple, allow: set) -> List[tuple]:
176 x, y, layer = cell
177 out: List[tuple] = []
178 for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
179 nx_, ny_ = x + dx, y + dy
180 if 0 <= nx_ < self.nx and 0 <= ny_ < self.ny:
181 out.append((nx_, ny_, layer))
182 if layer + 1 in allow:
183 out.append((x, y, layer + 1))
184 if layer - 1 in allow:
185 out.append((x, y, layer - 1))
186 return out

188 def _reconstruct(self, came: Dict[tuple, tuple], node: tuple) -> List[tuple]:
189 path = [node]
190 while node in came:
191 node = came[node]
192 path.append(node)
193 path.reverse()
194 return path

197 def _path_to_segments(self, net: Net, path: List[tuple]) -> List[Segment]:
198 """Fusionne les cellules alignées en segments, insère les vias.
199
200 Un run = suite de cellules colinéaires sur la même couche. Un changement
201 de couche produit un via au point de transition.
202 """
203 segments: List[Segment] = []
204 if len(path) < 2:
205 return segments
206 run_start = path[0]
207 prev_dir: Optional[Tuple[int, int]] = None
208 for i in range(1, len(path)):
209 cell = path[i]
210 prev = path[i - 1]
211 dx, dy = cell[0] - prev[0], cell[1] - prev[1]
212 layer_change = cell[2] != prev[2]
213 cur_dir = None if layer_change else (dx, dy)
214 if layer_change or (prev_dir is not None and cur_dir != prev_dir):
215 segments.append(self._make_segment(net, run_start, prev))
216 if layer_change:
217 segments.append(self._make_via(net, prev))
218 run_start = cell # le nouveau run part de l'autre couche
219 else:
220 run_start = prev # coude : nouveau run au même point
221 prev_dir = cur_dir
222 elif prev_dir is None:
223 prev_dir = cur_dir
224 segments.append(self._make_segment(net, run_start, path[-1]))
225 return [s for s in segments if s.is_via or s.length_mm > 0]

227 def _make_segment(self, net: Net, c1: tuple, c2: tuple) -> Segment:
228 x1, y1 = self._to_mm((c1[0], c1[1]))
229 x2, y2 = self._to_mm((c2[0], c2[1]))
230 width = 0.4 if (net.net_class or "").lower().startswith(("pwr", "power")) else 0.2
231 return Segment(net=net.name, x1_mm=x1, y1_mm=y1, x2_mm=x2, y2_mm=y2,
232 layer=c1[2], width_mm=width, is_via=False)
233
234 def _make_via(self, net: Net, cell: tuple) -> Segment:
235 x, y = self._to_mm((cell[0], cell[1]))
236 return Segment(net=net.name, x1_mm=x, y1_mm=y, x2_mm=x, y2_mm=y,
237 layer=cell[2], width_mm=0.3, is_via=True)

239 def _segment_cells(self, seg: Segment) -> List[tuple]:
240 c1 = self._to_cell(seg.x1_mm, seg.y1_mm)
241 c2 = self._to_cell(seg.x2_mm, seg.y2_mm)
242 cells = []
243 steps = max(abs(c2[0] - c1[0]), abs(c2[1] - c1[1]), 1)
244 for i in range(steps + 1):
245 cx = c1[0] + round((c2[0] - c1[0]) * i / steps)
246 cy = c1[1] + round((c2[1] - c1[1]) * i / steps)
247 cells.append((cx, cy, seg.layer))
248 return cells
