67 # ---- score composite ----------------------------------------------------------------
68 def composite_score(self, drc: float, si: float, thermal: float, vias: int) -> float:
69 """Score 0..1 : 0.5·DRC + 0.2·SI + 0.2·thermique + 0.1·(1 − vias/max)."""
70 drc_norm = min(1.0, max(0.0, drc / 100.0)) if drc > 1.0 else min(1.0, max(0.0, drc))
71 si_n = min(1.0, max(0.0, si))
72 thermal_n = min(1.0, max(0.0, thermal))
73 via_term = 1.0 - min(1.0, max(0, vias) / max(1, self.max_vias))
74 return 0.5 * drc_norm + 0.2 * si_n + 0.2 * thermal_n + 0.1 * via_term
