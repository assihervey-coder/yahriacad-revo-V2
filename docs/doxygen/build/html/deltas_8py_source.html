22def compute_board_delta(before: Board, after: Board) -> dict[str, Any]:
23 """Diff structurelle minimale entre deux états de carte."""
24 before_refs = set(before.placements)
25 after_refs = set(after.placements)
26 moved = []
27 for ref in sorted(before_refs & after_refs):
28 if _place_snapshot(before, ref) != _place_snapshot(after, ref):
29 moved.append({"ref": ref, "from": _place_snapshot(before, ref),
30 "to": _place_snapshot(after, ref)})
31 nets_before = set(before.nets)
32 nets_after = set(after.nets)
33 segments_changed = []
34 for name in sorted(nets_before & nets_after):
35 sb = before.nets[name].routed_segments
36 sa = after.nets[name].routed_segments
37 if [(s.x1_mm, s.y1_mm, s.x2_mm, s.y2_mm, s.layer) for s in sb] != \
38 [(s.x1_mm, s.y1_mm, s.x2_mm, s.y2_mm, s.layer) for s in sa]:
39 segments_changed.append(name)
40 return {
41 "components_added": sorted(after_refs - before_refs),
42 "components_removed": sorted(before_refs - after_refs),
43 "components_moved": moved,
44 "nets_added": sorted(nets_after - nets_before),
45 "nets_removed": sorted(nets_before - nets_after),
46 "nets_rerouted": segments_changed,
47 "summary": (f"{len(moved)} déplacement(s), {len(segments_changed)} net(s) re-routé(s), "
48 f"{len(after_refs - before_refs)} ajout(s), "
49 f"{len(before_refs - after_refs)} suppression(s)"),
50 }

53def events_from_delta(delta: dict[str, Any], project_id: str,
54 design_version: int, emitter: str = "state_manager") -> list[Event]:
55 """Traduit un delta en événements WebSocket (component_moved, net_routed)."""
56 events: list[Event] = []
57 for item in delta.get("components_moved", []):
58 events.append(make_event(EventType.COMPONENT_MOVED, project_id, emitter,
59 design_version=design_version, ref=item["ref"],
60 **item["to"]))
61 for name in delta.get("nets_rerouted", []) + delta.get("nets_added", []):
62 events.append(make_event(EventType.NET_ROUTED, project_id, emitter,
63 design_version=design_version, net=name))
64 return events

67def events_since(events: Iterable[Event], since_seq: int,
68 project_id: str | None = None) -> list[Event]:
69 """Filtre un flux d'événements pour la reprise par delta (seq > since_seq)."""
70 selected = [e for e in events if e.seq > since_seq]
