48def build_demo_board() -> Board:
49 """Mini-carte de référence (miroir du self-test ai_engine) : 3 composants,
50 4 nets, 1 keepout — déterministe, exécutable hors ligne."""
51 board = Board(width_mm=50.0, height_mm=40.0)
52 board.add_component(Component(ref="U1", mpn="STM32F407", value="MCU",
53 footprint="LQFP-64", pins=64,
54 width_mm=10.0, height_mm=10.0, power_w=0.9))
55 board.add_component(Component(ref="U2", mpn="L6981", value="buck",
56 footprint="QFN-16", pins=16,
57 width_mm=4.0, height_mm=4.0, power_w=1.2))
58 board.add_component(Component(ref="J1", mpn="USB4105", value="USB-C",
59 footprint="USB-C", pins=12,
60 width_mm=9.0, height_mm=8.0, power_w=0.1))
61 # placements initiaux : ligne de base (point de départ volontairement médiocre)
62 board.placements["U1"] = replace(board.placements["U1"], x_mm=10.0, y_mm=8.0)
63 board.placements["U2"] = replace(board.placements["U2"], x_mm=30.0, y_mm=30.0)
64 board.placements["J1"] = replace(board.placements["J1"], x_mm=40.0, y_mm=8.0)
65 board.nets["VBUS"] = Net(name="VBUS", connections=[("J1", "VBUS"), ("U2", "VIN")],
66 net_class="power")
67 board.nets["VDD_3V3"] = Net(name="VDD_3V3", connections=[("U2", "SW"), ("U1", "VDD")],
68 net_class="power")
69 board.nets["USB_DP"] = Net(name="USB_DP", connections=[("J1", "DP"), ("U1", "PA12")],
70 net_class="USB", impedance_target_ohm=90.0)
71 board.nets["GND"] = Net(name="GND", connections=[("J1", "GND"), ("U1", "VSS"), ("U2", "PGND")],
72 net_class="power")
73 board.zones.append(Zone(name="keepout_antenna", x_min_mm=36.0, y_min_mm=30.0,
74 x_max_mm=48.0, y_max_mm=38.0, kind="keepout"))
75 return board

78def _install_runtime_npz(npz_path: Path, runtime_npz: Path) -> bool:
79 """Copie l'export torch vers le slot runtime du world model (warm start nocturne).
80
81 La boucle nocturne (``AiEngine.run_night_optimization``) charge ce slot à
82 l'ouverture de la nuit et y réécrit les poids mis à jour en fin de nuit.
83 """
84 try:
85 runtime_npz.parent.mkdir(parents=True, exist_ok=True)
86 shutil.copyfile(npz_path, runtime_npz)
87 return True
88 except OSError as exc:
89 print(f"slot runtime {runtime_npz} non écrit ({exc})", file=sys.stderr)
90 return False

93def main(argv: List[str] | None = None) -> int:
94 """Point d'entrée CLI — 0 si la passe complète a réussi."""
95 parser = argparse.ArgumentParser(
96 prog="rl_training.train",
97 description="Passe RL du cerveau IA : world model (torch) + policy-gradient REINFORCE.")
98 parser.add_argument("--episodes-dataset", type=int, default=40,
99 help="épisodes de rollouts aléatoires pour le dataset (défaut 40)")
100 parser.add_argument("--episodes-rl", type=int, default=150,
101 help="épisodes REINFORCE (défaut 150)")
102 parser.add_argument("--epochs-world", type=int, default=80,
103 help="epochs max de régression supervisée (défaut 80)")
104 parser.add_argument("--out", type=Path, default=Path("data/trained_models/rl_checkpoints"),
105 help="répertoire des artefacts")
106 parser.add_argument("--runtime-npz", type=Path, default=None,
107 help="slot runtime du world model pour la boucle nocturne "
108 "(défaut : PCB_WORLD_MODEL_NPZ / settings.world_model_npz)")
109 parser.add_argument("--no-install", action="store_true",
110 help="n'installe pas l'export au slot runtime")
111 parser.add_argument("--archive", type=Path, default=None,
112 help="archive keeper JSONL (contexte du rapport)")
113 parser.add_argument("--seed", type=int, default=42)
114 parser.add_argument("--device", type=str, default="cpu")
115 parser.add_argument("--skip-world", action="store_true", help="saute la régression supervisée")
116 parser.add_argument("--skip-rl", action="store_true", help="saute REINFORCE")
117 args = parser.parse_args(argv)

129 started = time.perf_counter()

145 # ---- 2) world model supervisé ------------------------------------------------------
146 runtime_npz = args.runtime_npz or Path(get_settings("ai_engine").world_model_npz)
147 if not args.skip_world:
148 world = TorchWorldModel(seed=args.seed)
149 result = train_supervised_world_model(world, X, Y, epochs=args.epochs_world,

154 report["world_model"] = {"epochs_run": result["epochs_run"],
155 "best_val_mse": result["best_val_mse"],
156 "final_train_mse": result["final_train_mse"],
157 "exported": exported, "npz": str(npz_path),
158 "runtime_npz": {"path": str(runtime_npz),
159 "installed": installed}}
160 print(f"[2/4] world model : {result['epochs_run']} epochs, "

163 else:
164 report["world_model"] = {"skipped": True, "runtime_npz": str(runtime_npz)}
165
166 # ---- 3) REINFORCE ------------------------------------------------------------------
167 if not args.skip_rl:
168 policy = PlacementPolicyNet()
169 trainer = ReinforceTrainer(policy, env,
170 config=ReinforceConfig(seed=args.seed, device=args.device))
171 ckpt_path = args.out / "policy_reinforce.pt"
172 rl_result = trainer.train(episodes=args.episodes_rl, checkpoint_path=ckpt_path)
173 report["reinforce"] = {**rl_result, "checkpoint": str(ckpt_path)}
174 print(f"[3/4] REINFORCE : {rl_result['episodes']} épisodes, "
175 f"retour moyen {rl_result['mean_reward']:.3f}, "
176 f"meilleur fenêtre 10 {rl_result['best_mean_reward_10']:.3f}")
177 else:
178 report["reinforce"] = {"skipped": True}
179
180 # ---- 4) rapport --------------------------------------------------------------------
181 report["keeper_archive"] = summarize_archive(args.archive)
182 report["elapsed_s"] = round(time.perf_counter() - started, 2)
183 report_path = args.out / "training_report.json"
184 try:
185 report_path.write_text(json.dumps(report, indent=2, ensure_ascii=False),
186 encoding="utf-8")
187 print(f"[4/4] rapport : {report_path}")
188 except OSError as exc:
189 print(f"[4/4] rapport non écrit ({exc}) — artefacts npz/pt disponibles", file=sys.stderr)
190 print(f"Passe RL terminée en {report['elapsed_s']} s")
191 return 0
