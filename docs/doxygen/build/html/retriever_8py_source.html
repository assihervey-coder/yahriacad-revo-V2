46 def answer(self, query: str, k: int = 4) -> Dict[str, object]:
47 """Réponse ancrée : texte + liste de citations + mode réellement utilisé."""
48 passages = self.retrieve(query, k=k)
49 citations = [doc.citation() for doc, _score in passages]
50 provider = self.settings.llm_provider
51 if provider == "ollama":
52 llm_text = self._ollama_answer(query, passages)
53 if llm_text is not None:
54 return {"answer": llm_text, "citations": citations, "mode": "ollama"}
55 elif provider != "local": # openai-compatible / distant
56 llm_text = self._llm_answer(query, passages)
57 if llm_text is not None:
58 return {"answer": llm_text, "citations": citations, "mode": "llm"}
59 return {"answer": self._extractive_answer(query, passages),

62 def _ollama_answer(self, query: str,

64 """Réponse par le LLM local Ollama — None = fallback extractif immédiat.

66 Le prompt système impose l'ancrage : chaque affirmation doit citer sa

68 n'est jamais autorisé à répondre « hors contexte ».

71 client = OllamaClient(settings=self.settings)
72 if not passages:
73 return None
74 context = "\n\n".join(
75 f"{doc.citation()} {doc.content[:600]}" for doc, _s in passages[:4])
76 messages = [

78 "content": "Tu es l'assistant RAG d'une plateforme de conception PCB. "

88 def _extractive_answer(self, query: str, passages: List[Tuple[Document, float]]) -> str:
89 """Concatène les meilleures phrases, chacune citée — déterministe."""
90 if not passages:
91 return ("Aucun passage pertinent dans l'index RAG : alimentez l'index "
92 "avec les datasheets du projet (rag_engine.indexer.TfidfIndex.upsert).")
93 keywords = {w for w in re.findall(r"[a-zà-ÿ0-9]{2,}", query.lower())
94 if w not in _STOPWORDS}
95 selected: List[str] = []
96 for doc, _score in passages[:3]:
97 sentences = _SENTENCE_RE.split(doc.content)
98 scored = [(sum(1 for w in keywords if w in s.lower()), s)
99 for s in sentences]
100 scored.sort(key=lambda pair: -pair[0])
101 best = scored[0][1].strip() if scored and scored[0][0] > 0 else sentences[0].strip()
102 if best:
103 selected.append(f"{best} {doc.citation()}")
104 return " ".join(selected)

106 def _llm_answer(self, query: str, passages: List[Tuple[Document, float]]) -> str | None:
107 """Adaptateur LLM optionnel — None = fallback immédiat sur le mode extractif."""
108 try:
109 import httpx # import gardé — dépendance optionnelle à l'exécution
110 except ImportError:
111 return None
112 context = "\n\n".join(
113 f"{doc.citation()} {doc.content[:600]}" for doc, _s in passages[:4])
114 prompt = (f"Réponds en français, chaque affirmation doit citer sa source "
115 f"[source p.X]. Contexte :\n{context}\n\nQuestion : {query}")
116 try:
117 response = httpx.post(
118 f"{self.settings.llm_api_base.rstrip('/')}/chat/completions",
119 json={"model": self.settings.llm_model,
120 "messages": [{"role": "user", "content": prompt}]},
121 timeout=4.0,
122 )
123 response.raise_for_status()
124 data = response.json()
125 return str(data["choices"][0]["message"]["content"]).strip()
126 except Exception as exc: # réseau, JSON, timeout — fallback déterministe
127 logger.warning("adaptateur LLM indisponible — réponse extractive",
128 extra={"error": str(exc)[:160]})
129 return None
