201_BLOCK_PRIORITY = {key: index for index, key in enumerate(

205def get_block(key: str) -> BlockTemplate:
206 """Accès typé à un bloc — KeyError explicite si la clé est inconnue."""
207 if key not in BLOCKS:
208 raise KeyError(f"bloc fonctionnel inconnu : {key} (connus : {sorted(BLOCKS)})")
209 return BLOCKS[key]
210
211
212def all_blocks() -> List[BlockTemplate]:
213 """Blocs triés par priorité d'assemblage."""
214 return sorted(BLOCKS.values(), key=lambda b: _BLOCK_PRIORITY[b.key])
