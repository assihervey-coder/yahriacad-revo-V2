82def _jwt_encode_homegrown(payload: dict[str, Any], secret: str) -> str:
83 header = {"alg": _ALGORITHM, "typ": "JWT"}
84 segments = [_b64url(json.dumps(header, separators=(",", ":")).encode()),
85 _b64url(json.dumps(payload, separators=(",", ":")).encode())]
86 signing_input = ".".join(segments).encode()
87 signature = hmac.new(secret.encode(), signing_input, hashlib.sha256).digest()
88 segments.append(_b64url(signature))
89 return ".".join(segments)

92def _jwt_decode_homegrown(token: str, secret: str) -> dict[str, Any]:
93 try:
94 header_b64, payload_b64, signature_b64 = token.split(".")
95 except ValueError as exc:
96 raise AuthError(401, "jeton malformé") from exc
97 signing_input = f"{header_b64}.{payload_b64}".encode()
98 expected = hmac.new(secret.encode(), signing_input, hashlib.sha256).digest()
99 if not hmac.compare_digest(_b64url(expected), signature_b64):
100 raise AuthError(401, "signature JWT invalide")
101 try:
102 return json.loads(_b64url_decode(payload_b64))
103 except (ValueError, json.JSONDecodeError) as exc:
104 raise AuthError(401, "payload JWT illisible") from exc

117 # ---- JWT ---------------------------------------------------------------
118 def issue_token(self, user: str, scopes: list[str] | None = None,
119 tier: str = "free", expires_min: int | None = None) -> str:
120 """Émet un JWT HS256 (python-jose si dispo sinon HMAC maison)."""
121 now = int(time.time())
122 payload = {"sub": user, "scopes": scopes or ["read", "write"], "tier": tier,
123 "iat": now, "exp": now + (expires_min or self.settings.jwt_expires_min) * 60,
124 "iss": self.settings.jwt_issuer}
125 if _jose_jwt is not None:
126 return str(_jose_jwt.encode(payload, self.settings.jwt_secret, algorithm=_ALGORITHM))
127 return _jwt_encode_homegrown(payload, self.settings.jwt_secret)
128
129 def verify(self, token: str) -> Claims:
130 """Vérifie signature + expiration → Claims (AuthError 401 sinon)."""
131 try:
132 if _jose_jwt is not None:
133 payload = dict(_jose_jwt.decode(token, self.settings.jwt_secret,
134 algorithms=[_ALGORITHM]))
135 else:
136 payload = _jwt_decode_homegrown(token, self.settings.jwt_secret)
137 except AuthError:
138 raise
139 except Exception as exc: # jose lève JWTError (importé conditionnellement)
140 raise AuthError(401, f"jeton rejeté : {exc}") from exc
141 claims = Claims.from_payload(payload)
142 if claims.exp and claims.exp < time.time():
143 raise AuthError(401, "jeton expiré")
144 return claims

159 # ---- point d'entrée middleware ------------------------------------------------
160 def authenticate(self, headers: dict[str, str]) -> Identity:
161 """Authentifie via `Authorization: Bearer <jwt>` ou `X-API-Key: pcbak_…`."""
162 authorization = _header(headers, "authorization")
163 if authorization.lower().startswith("bearer "):
164 claims = self.verify(authorization[7:].strip())
165 return Identity(sub=claims.sub, scopes=claims.scopes, tier=claims.tier, via="jwt")
166 api_key = _header(headers, "x-api-key")
167 if api_key.startswith(API_KEY_PREFIX):
168 record = self._api_keys.get(self._hash_key(api_key))
169 if record is None:
170 raise AuthError(401, "clé API inconnue")
171 return Identity(sub=record["name"], scopes=record["scopes"],
172 tier=record["tier"], via="api_key")
173 raise AuthError(401, "authentification requise (JWT ou X-API-Key)")
