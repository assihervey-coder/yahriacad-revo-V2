58def route(board: Board, project_id: str = "demo", nets_filter: Optional[List[str]] = None,
59 minimize_vias: bool = True, design_version: int = 1) -> "tuple[Board, RouteStats]":
60 """Route les nets non routés (ou la liste fournie) et applique le via_minimizer.
61
62 nets_filter non vide = re-route chirurgical limité à ces nets (étape 6 du
63 workflow). Chaque net routé émet un événement NET_ROUTED consommé par le
64 websocket_live (latence cible < 100 ms) et le plugin KiCad.
65 """
66 start = time.perf_counter()
67 get_settings("router")
68 stats = RouteStats()
69
70 targets = [n for name in (nets_filter or list(board.nets)) if (n := board.nets.get(name))]
71 stats.nets_total = len(targets)
72
73 graph = ConnectivityGraph(board)
74 finder = Pathfinder(board)
75 # bloque les emprises de pads (clearance de base — buts exemptés dans A*)
76 finder.block_pads()
77
78 for net in targets:
79 if net.is_routed and not nets_filter:
80 continue # déjà routé — sauf demande explicite de re-route
81 result = finder.route_net(graph, net)
82 if result and result.segments:
83 net.routed_segments = result.segments
84 stats.nets_routed += 1
85 stats.via_count += result.vias
86 stats.routed_length_mm += sum(s.length_mm for s in result.segments)
87 make_event(EventType.NET_ROUTED, project_id, "router", design_version,
88 net=net.name, layer=result.segments[0].layer,
89 segments=len(result.segments), vias=result.vias)
90 else:
91 stats.nets_failed.append(net.name)
92 logger.warning("net inroutable — escalade super_agent", extra={"net": net.name})
93
94 if minimize_vias:
95 board, stats.minimizer = ViaMinimizer().minimize(board)
96 stats.via_count = board.via_count()
97
98 stats.duration_s = time.perf_counter() - start
99 logger.info("passe de routage terminée", extra=stats.as_payload())
100 return board, stats
