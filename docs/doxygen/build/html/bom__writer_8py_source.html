15def export_bom(board: Board) -> Tuple[str, Dict[str, float]]:
16 """Génère le BOM CSV groupé par (mpn, value) + sous-totaux par bloc.
17
18 Retourne (csv_text, {functional_block: subtotal_usd}) — les sous-totaux par
19 bloc fonctionnel alimentent le credit_dashboard et les revues de coût.
20 """
21 groups: Dict[Tuple[str, str], List] = {}
22 for comp in board.components.values():
23 groups.setdefault((comp.mpn, comp.value), []).append(comp)
24
25 block_subtotals: Dict[str, float] = {}
26 buffer = io.StringIO()
27 writer = csv.writer(buffer)
28 writer.writerow(_HEADER)
29 for (mpn, value), comps in sorted(groups.items()):
30 first = comps[0]
31 qty = len(comps)
32 subtotal = round(qty * first.price_usd, 4)
33 block = first.functional_block or "uncategorized"
34 block_subtotals[block] = round(block_subtotals.get(block, 0.0) + subtotal, 4)
35 writer.writerow([", ".join(c.ref for c in comps), mpn, value, first.footprint,
36 qty, first.price_usd, subtotal, block])
37 return buffer.getvalue(), block_subtotals
