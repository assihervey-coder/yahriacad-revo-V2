1"""Environnement de placement type gym — le terrain de jeu de la passe RL.
2
3Chaque épisode place séquentiellement les composants non verrouillés d'une
4carte. À l'étape *t*, l'agent reçoit l'état s_t : le vecteur de 12 features du
5world model (``DreamerWorldModel.extract_features``) calculé à l'ancre du
6composant courant (centroïde de ses partenaires de nets). Il répond par un
7vecteur continu-discret ``(dx_norm, dy_norm, rot_idx, layer)`` :
8
9 * ``(dx, dy)`` — déplacement relatif à l'ancre, en unités normalisées
10 (× ``spread_mm`` = 12 mm par défaut) ;
11 * ``rot_idx`` — index de rotation ∈ {0, 90, 180, 270}° ;
12 * ``layer`` — couche ∈ {0 = top, 1 = bottom}.
13
14Récompense (déterministe, donc reproductible — exigence Circuitron) :
15
16 r = 0.6 · (1 − risque) + 0.4 · gain_HPWL_normalisé − 0.5 · [action illégale]
17
18où ``risque = 0.4·congestion + 0.3·thermique + 0.3·si`` (heuristique du world
19model, cf. ``dreamer._heuristic_consequences``). Une action illégale (hors
20carte, keepout, collision) n'est PAS appliquée : le composant garde son
21emplacement courant et l'agent encaisse la pénalité — c'est le mur bas.
22
23L'environnement travaille sur les placements d'une copie restaurée à chaque
24``reset()`` : le board appelant n'est jamais muté.
25"""

53 self.seed = seed
54 self.spread_mm = spread_mm
55 self.action_space = ActionSpace()

64 def reset(self) -> np.ndarray:
65 """Restaure les placements initiaux et retourne l'état du 1er composant."""
66 for ref, placement in self._initial.items():
67 self.board.placements[ref] = replace(placement)
68 self._queue = [ref for ref, p in self.board.placements.items() if not p.locked]
69 self._anchors = {ref: self._anchor_for(ref) for ref in self._queue}
70 self.steps_taken = 0
71 return self._state(self._queue[0]) if self._queue else np.zeros(12)

73 def step(self, action_vec: Sequence[float]) -> Tuple[np.ndarray, float, bool, Dict[str, Any]]:
74 """Applique (dx, dy, rot, layer) sur le composant courant.
75
76 Retour : ``(état_suivant, récompense, terminé, info)`` — convention gym.
77 """
78 if not self._queue:
79 return np.zeros(12), 0.0, True, {"reason": "épisode déjà terminé"}
80 dx_norm, dy_norm = float(action_vec[0]), float(action_vec[1])
81 rot_idx = int(np.clip(round(float(action_vec[2])), 0, len(ROTATIONS) - 1))
82 layer = int(np.clip(round(float(action_vec[3])), 0, 1))
83 ref = self._queue[0]
84 anchor_x, anchor_y = self._anchors[ref]
85 action = Action(
86 ref=ref,
87 x_mm=self.action_space.snap(anchor_x + dx_norm * self.spread_mm),
88 y_mm=self.action_space.snap(anchor_y + dy_norm * self.spread_mm),
89 rotation_deg=float(ROTATIONS[rot_idx]), layer=layer)
90
91 legal = bool(self.action_space.legal_mask(self.board, [action])[0])
92 hpwl_before = hpwl_mm(self.board)
93 if legal:
94 self.board.move(ref, action.x_mm, action.y_mm,
95 rotation_deg=action.rotation_deg)
96 self.board.placements[ref] = replace(self.board.placements[ref],
97 layer=action.layer)
98 features = self.world.extract_features(self.board, ref, action.x_mm, action.y_mm,
99 action.rotation_deg, action.layer)
100 consequences = _heuristic_consequences(features)
101 risk = (_RISK_WEIGHTS[0] * consequences["congestion_score"]
102 + _RISK_WEIGHTS[1] * consequences["thermal_rise_estimate"]
103 + _RISK_WEIGHTS[2] * consequences["si_risk"])
104 hpwl_after = hpwl_mm(self.board)
105 hpwl_gain = 0.0
106 if hpwl_before > 1e-9:
107 hpwl_gain = float(np.clip((hpwl_before - hpwl_after) / hpwl_before, -1.0, 1.0))
108 reward = 0.6 * (1.0 - risk) + 0.4 * hpwl_gain
109 if not legal:
110 reward += _ILLEGAL_PENALTY
111
112 self._queue.pop(0)
113 self.steps_taken += 1
114 done = not self._queue
115 next_state = self._state(self._queue[0]) if self._queue else np.zeros(12)
116 info: Dict[str, Any] = {"ref": ref, "legal": legal,
117 "consequences": consequences,
118 "hpwl_mm": round(hpwl_after, 3),
119 "step": self.steps_taken}
120 return next_state, float(reward), done, info

122 # ---- helpers ------------------------------------------------------------------------------
123 def _anchor_for(self, ref: str) -> Tuple[float, float]:
124 """Ancre = centroïde des partenaires de nets placés, sinon position courante."""
125 xs: List[float] = []
126 ys: List[float] = []
127 for net in self.board.nets.values():
128 partners = [r for r, _pad in net.connections if r != ref and r in self.board.placements]
129 if not partners:
130 continue
131 xs.extend(self.board.placements[r].x_mm for r in partners)
132 ys.extend(self.board.placements[r].y_mm for r in partners)
133 if xs and ys:
134 return (float(np.clip(sum(xs) / len(xs), 0.0, self.board.width_mm)),
135 float(np.clip(sum(ys) / len(ys), 0.0, self.board.height_mm)))
136 current = self.board.placements.get(ref)
137 if current is not None and (current.x_mm or current.y_mm):
138 return (current.x_mm, current.y_mm)
139 return (self.board.width_mm / 2.0, self.board.height_mm / 2.0)
