1"""Modules torch de la passe RL — world model + policy de placement.
2
3``TorchWorldModel`` reflète EXACTEMENT l'architecture du ``TinyNet`` numpy du
4runtime (``dreamer.TinyNet``) : première couche = projection aléatoire FIXE
5(random features), seconde couche = tête linéaire entraînée. Cette contrainte
6de symétrie est délibérée : les poids torch entraînés par Adam + early
7stopping sont exportables en ``.npz`` au format du runtime
8(``DreamerWorldModel.load``) — la passe d'entraînement améliore donc
9directement le cerveau embarqué, sans distillation ni traduction de graphe.
10
11``PlacementPolicyNet`` est la tête de décision entraînée par policy-gradient
12(REINFORCE, cf. ``reinforce.py``) : tronc MLP sur les 12 features du world
13model, tête gaussienne pour (dx, dy) continus, têtes catégorielles pour la
14rotation (4 classes) et la couche (2 classes).
15
16torch est importé de façon gardée : sans torch, ce module reste importable
17(les classes ne sont simplement pas définies) — cohérent avec la convention
18du dépôt « le runtime ne dépend jamais d'une lib lourde ».

44 class TorchWorldModel(nn.Module):
45 """World model torch — miroir exact du TinyNet runtime (W1 fixe, W2 appris)."""
46
47 def __init__(self, feature_dim: int = 12, hidden: int = 32,
48 output_dim: int = 3, seed: int = 42) -> None:
49 super().__init__()
50 # Même initialisation que dreamer.TinyNet : projection aléatoire partagée
51 rng = np.random.default_rng(seed)
52 W1 = rng.normal(0.0, 0.5, size=(feature_dim, hidden)).astype(np.float32)
53 b1 = np.zeros(hidden, dtype=np.float32)
54 self.register_buffer("W1", torch.from_numpy(W1))
55 self.register_buffer("b1", torch.from_numpy(b1))
56 self.head = nn.Linear(hidden, output_dim) # W2, b2 entraînables
57 nn.init.zeros_(self.head.weight)
58 nn.init.zeros_(self.head.bias)
59
60 def forward(self, X: "torch.Tensor") -> "torch.Tensor":
61 """tanh(X·W1 + b1)·W2 + b2 — mêmes maths que le runtime numpy."""
62 hidden = torch.tanh(X @ self.W1 + self.b1)
63 return self.head(hidden)

65 @torch.no_grad()
66 def export_npz(self, path: Path) -> bool:
67 """Export au format des checkpoints du runtime — DreamerWorldModel.load."""
68 try:
69 path = Path(path)
70 path.parent.mkdir(parents=True, exist_ok=True)
71 W2 = self.head.weight.detach().cpu().numpy().T # (hidden, out)
72 b2 = self.head.bias.detach().cpu().numpy()
73 np.savez(path,
74 W1=self.W1.cpu().numpy().astype(np.float64),
75 b1=self.b1.cpu().numpy().astype(np.float64),
76 W2=W2.astype(np.float64), b2=b2.astype(np.float64),
77 trained=np.array(True))
78 return True
79 except OSError:
80 return False

82 def train_supervised_world_model(
83 model: "TorchWorldModel", X: np.ndarray, Y: np.ndarray,
84 epochs: int = 60, lr: float = 1e-2, batch_size: int = 32,
85 weight_decay: float = 1e-4, val_ratio: float = 0.2,
86 patience: int = 8, seed: int = 42,
87 device: str = "cpu") -> Dict[str, Any]:
88 """Régression supervisée (MSE + Adam) avec split train/val et early stopping.
89
90 Retour : ``{"epochs_run", "best_val_mse", "final_train_mse", "history"}``.
91 Le meilleur état (val minimale) est restauré avant la sortie — jamais
92 d'overfit silencieux vers les derniers epochs.
93 """
94 assert model is not None and X.shape[0] >= 8, "dataset trop petit (< 8 lignes)"
95 rng = np.random.default_rng(seed)
96 indices = rng.permutation(X.shape[0])
97 n_val = max(2, int(X.shape[0] * val_ratio))
98 val_idx, train_idx = indices[:n_val], indices[n_val:]
99 X_train = torch.tensor(X[train_idx], dtype=torch.float32, device=device)
100 Y_train = torch.tensor(Y[train_idx], dtype=torch.float32, device=device)
101 X_val = torch.tensor(X[val_idx], dtype=torch.float32, device=device)
102 Y_val = torch.tensor(Y[val_idx], dtype=torch.float32, device=device)
103
104 optimizer = torch.optim.Adam(model.parameters(), lr=lr, weight_decay=weight_decay)
105 scheduler = torch.optim.lr_scheduler.CosineAnnealingLR(optimizer, T_max=max(1, epochs))
106 history: List[Dict[str, float]] = []
107 best_val, best_state, bad_epochs = float("inf"), None, 0
108
109 for epoch in range(1, max(1, epochs) + 1):
110 model.train()
111 perm = torch.randperm(X_train.shape[0])
112 epoch_loss, n_batches = 0.0, 0
113 for start in range(0, X_train.shape[0], batch_size):
114 batch = perm[start:start + batch_size]
115 optimizer.zero_grad()
116 prediction = model(X_train[batch])
117 loss = torch.nn.functional.mse_loss(prediction, Y_train[batch])
118 loss.backward()
119 optimizer.step()
120 epoch_loss += float(loss.item())
121 n_batches += 1
122 scheduler.step()
123 model.eval()
124 with torch.no_grad():
125 val_loss = float(torch.nn.functional.mse_loss(model(X_val), Y_val).item())
126 history.append({"epoch": epoch,
127 "train_mse": round(epoch_loss / max(1, n_batches), 6),
128 "val_mse": round(val_loss, 6)})
129 if val_loss < best_val - 1e-6: # early stopping
130 best_val, bad_epochs = val_loss, 0
131 best_state = {k: v.clone() for k, v in model.state_dict().items()}
132 else:
133 bad_epochs += 1
134 if bad_epochs >= patience:
135 break
136 if best_state is not None:
137 model.load_state_dict(best_state)
138 return {"epochs_run": len(history), "best_val_mse": round(best_val, 6),
139 "final_train_mse": history[-1]["train_mse"] if history else None,
140 "history": history}

142 class PlacementPolicyNet(nn.Module):
143 """Policy π_θ(a|s) : tronc MLP → gaussienne (dx, dy) + catégorielles (rot, couche).
144
145 Les déplacements sont produits en unités normalisées (−1..1 ≈ ±spread_mm
146 de l'environnement) : la policy reste invariante à la taille de carte.
147 """
148
149 def __init__(self, feature_dim: int = 12, hidden: int = 64,
150 n_rotations: int = 4, n_layers: int = 2) -> None:
151 super().__init__()
152 self.trunk = nn.Sequential(
153 nn.Linear(feature_dim, hidden), nn.Tanh(),
154 nn.Linear(hidden, hidden), nn.Tanh())
155 self.mu_head = nn.Linear(hidden, 2) # (dx, dy) normalisés
156 self.log_std = nn.Parameter(torch.zeros(2)) # std appris, état-indépendant
157 self.rot_head = nn.Linear(hidden, n_rotations)
158 self.layer_head = nn.Linear(hidden, n_layers)
159
160 def forward(self, state: "torch.Tensor") -> Dict[str, Any]:
161 """Distributions d'action — appelé par ``act`` et par le trainer."""
162 features = self.trunk(state)
163 mu = self.mu_head(features)
164 std = self.log_std.exp().expand_as(mu)
165 continuous = torch.distributions.Normal(mu, std)
166 rotations = torch.distributions.Categorical(logits=self.rot_head(features))
167 layers = torch.distributions.Categorical(logits=self.layer_head(features))
168 return {"continuous": continuous, "rotations": rotations, "layers": layers}
169
170 def act(self, state: "torch.Tensor",
171 deterministic: bool = False) -> Tuple["torch.Tensor", Dict[str, "torch.Tensor"]]:
172 """Échantillonne une action — retourne (vecteur_action, {log_prob, entropy})."""
173 dists = self(state)
174 if deterministic:
175 dx_dy = dists["continuous"].mean
176 rot_idx = dists["rotations"].logits.argmax(dim=-1)
177 layer_idx = dists["layers"].logits.argmax(dim=-1)
178 else:
179 dx_dy = dists["continuous"].rsample()
180 rot_idx = dists["rotations"].sample()
181 layer_idx = dists["layers"].sample()
182 dx_dy = torch.clamp(dx_dy, -1.0, 1.0)
183 log_prob = (dists["continuous"].log_prob(dx_dy).sum(dim=-1)
184 + dists["rotations"].log_prob(rot_idx)
185 + dists["layers"].log_prob(layer_idx))
186 entropy = (dists["continuous"].entropy().sum(dim=-1)
187 + dists["rotations"].entropy()
188 + dists["layers"].entropy())
189 action_vec = torch.stack([dx_dy[:, 0], dx_dy[:, 1],
190 rot_idx.to(dx_dy.dtype),
191 layer_idx.to(dx_dy.dtype)], dim=-1)
192 return action_vec, {"log_prob": log_prob, "entropy": entropy}
