57 # ---- conversion design -> contraintes --------------------------------------
58 def broadcast_board_constraints(self, board: Any, source: str = "ai_engine") -> int:
59 """Diffuse les contraintes dérivées du Board ; retourne le nb de messages.
60
61 Mapping déterministe : zones keepout → KEEPOUT_ZONE, zones thermiques →
62 THERMAL_ZONE_UPDATE, nets à impédance cible → IMPEDANCE_TARGET, groupes
63 d'appariement de longueur → LENGTH_MATCH_RULE, nets d'alim →
64 CURRENT_BUDGET, plus une clearance globale par défaut (0.2 mm).
65 """
66 count = 0
67 for zone in board.zones:
68 rect = {
69 "x_min_mm": zone.x_min_mm, "y_min_mm": zone.y_min_mm,
70 "x_max_mm": zone.x_max_mm, "y_max_mm": zone.y_max_mm,
71 }
72 if zone.kind == "keepout":
73 self.publish(self.make_message(ConstraintKind.KEEPOUT_ZONE, f"keepout/{zone.name}", rect, source))
74 count += 1
75 elif zone.kind == "thermal":
76 payload = dict(rect, max_temp_c=zone.max_temp_c)
77 self.publish(self.make_message(ConstraintKind.THERMAL_ZONE_UPDATE, f"thermal/{zone.name}", payload, source))
78 count += 1
79 for net in board.nets.values():
80 if net.impedance_target_ohm is not None:
81 self.publish(self.make_message(
82 ConstraintKind.IMPEDANCE_TARGET, f"impedance/{net.name}",
83 {"target_ohm": net.impedance_target_ohm, "net_class": net.net_class}, source))
84 count += 1
85 if net.length_match_group:
86 self.publish(self.make_message(
87 ConstraintKind.LENGTH_MATCH_RULE, f"length/{net.name}",
88 {"group": net.length_match_group}, source))
89 count += 1
90 if net.net_class == "power":
91 self.publish(self.make_message(
92 ConstraintKind.CURRENT_BUDGET, f"current/{net.name}",
93 {"budget_a": 1.0, "net_class": "power"}, source))
94 count += 1
95 self.publish(self.make_message(
96 ConstraintKind.CLEARANCE, "clearance/global",
97 {"min_clearance_mm": 0.2}, source))
98 count += 1
99 self._broadcasts += 1
100 return count
