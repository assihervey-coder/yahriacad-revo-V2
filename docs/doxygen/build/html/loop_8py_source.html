1"""Boucle multiphysique continue [Cadence AuraStack] — section 6.2.
2
3À chaque commit du state_manager (hook `notify_commit()` déclenché par callback
4d'abonnement) ou périodiquement (thread léger daemon), la boucle rejoue
5thermal + SI (+ EM) sur l'état courant de la carte, puis :
6
7 1. publie les contraintes révisées sur le constraint_bus :
8 - THERMAL_ZONE_UPDATE clé "thermal/{ref}" (hotspots composants/zones)
9 - IMPEDANCE_TARGET clé "impedance/{net}" (nets SI en échec d'œil)
10 2. expose les « zones à re-optimiser » via `latest_findings()` — le rl_agent
11 relance alors une exploration locale sans refaire tout le pipeline.
12
13Publication dédupliquée : une contrainte n'est republiée que si sa valeur a
14changé au-delà d'un epsilon (évite de saturer le bus à 50 ms de latence cible).
15"""

72
73 Deux déclencheurs cumulables :
74 - `notify_commit()` — hook appelé par le state_manager (abonnement

76 - le thread `start()` — rejoue l'analyse au plus toutes les `interval_s`.

109 # ---- cycle de vie -------------------------------------------------------
110 def start(self) -> None:
111 """Démarre le thread daemon d'analyse continue (no-op si déjà actif)."""
112 if self._thread_thread is not None and self._thread_thread.is_alive():
113 return
114 self._stop.clear()
115 self._thread_thread = threading.Thread(
116 target=self._run, name=f"multi_physics_{self.project_id}", daemon=True,
117 )
118 self._thread_thread.start()
119 logger.info("boucle multiphysique démarrée",
120 extra={"project_id": self.project_id, "interval_s": self.interval_s})
121
122 def stop(self, timeout_s: float = 3.0) -> None:
123 """Arrêt gracieux — termine l'itération en cours puis joint le thread."""
124 self._stop.set()
125 self._wake.set()
126 if self._thread_thread is not None:
127 self._thread_thread.join(timeout=timeout_s)
128 self._thread_thread = None
129 logger.info("boucle multiphysique arrêtée", extra={"project_id": self.project_id})
130
131 def notify_commit(self, _board=None) -> None:
132 """Hook à brancher sur le state_manager : `sm.subscribe(cb=lambda s: loop.notify_commit())`.
133
134 Réveille immédiatement le thread (analyse sur l'état frais) ; sans
135 thread démarré, l'analyse sera effectuée au prochain `start()`.
136 """
137 self._wake.set()
138
139 # ---- thread -------------------------------------------------------------
140 def _run(self) -> None:
141 while not self._stop.is_set():
142 self._wake.wait(timeout=self.interval_s)
143 self._wake.clear()
144 if self._stop.is_set():
145 break
146 try:
147 self.analyze_once()
148 except Exception: # une itération défaillante n'arrête jamais la boucle
149 logger.exception("itération multiphysique en erreur",
150 extra={"project_id": self.project_id})
151
152 # ---- analyse ------------------------------------------------------------
153 def analyze_once(self, board=None) -> List[Finding]:
154 """Une passe complète thermal + SI (+ IR drop) — thread-safe, sans thread requis."""
155 if not self._lock.acquire(blocking=False):
156 return list(self.findings) # analyse déjà en cours — état précédent conservé
157 try:
158 t0 = time.perf_counter()
159 current = board if board is not None else self._board_provider()
160 findings: List[Finding] = []
161
162 # ---- thermique : hotspots ----------------------------------------
163 thermal = diffusion.steady_state(current, cell_size_mm=self.cell_size_mm)
164 for hotspot in thermal.hotspots:
165 key = f"thermal/{hotspot.component_ref}" if hotspot.component_ref else "thermal/board"
166 severity = "critical" if hotspot.kind == "zone" else (
167 "warning" if hotspot.temp_c > thermal.ambient_c + 25.0 else "info"
168 )
169 findings.append(Finding(
170 key=key, kind="thermal_hotspot", severity=severity,
171 zone=(hotspot.x_min_mm, hotspot.y_min_mm, hotspot.x_max_mm, hotspot.y_max_mm),
172 detail={**hotspot.to_dict(), "max_temp_c": thermal.max_temp_c},
173 ))
174 self._publish(
175 ConstraintKind.THERMAL_ZONE_UPDATE, key,
176 {**hotspot.to_dict(), "max_temp_c": thermal.max_temp_c,
177 "ambient_c": thermal.ambient_c, "backend": thermal.backend},
178 )
179
180 # ---- SI : ouverture d'œil ----------------------------------------
181 si = eye_diagram.analyze(current, crosstalk_db=-35.0)
182 for margin in si.margins:
183 if margin.risk == "ok":
184 continue
185 net = current.nets.get(margin.net)
186 target = None
187 if net is not None and net.impedance_target_ohm is not None:
188 target = float(net.impedance_target_ohm)
189 else:
190 target = SUGGESTED_IMPEDANCE_OHM.get(
191 (margin.net_class or "default").lower(),
192 SUGGESTED_IMPEDANCE_OHM["default"],
193 )
194 key = f"impedance/{margin.net}"
195 detail = {
196 "net": margin.net, "eye_opening_pct": margin.eye_opening_pct,
197 "insertion_loss_db": margin.insertion_loss_db,
198 "overshoot_mv": margin.overshoot_mv, "risk": margin.risk,
199 "target_ohm": target,
200 }
201 findings.append(Finding(
202 key=key, kind="si_eye_closure",
203 severity="critical" if margin.risk == "fail" else "warning",
204 zone=None, detail=detail,
205 ))
206 self._publish(ConstraintKind.IMPEDANCE_TARGET, key, detail)
207
208 # ---- IR drop (rail principal) -------------------------------------
209 ird = ir_drop.analyze_ir_drop(current)
210 if ird.note == "" and ird.ir_drop_pct > 0.5:
211 findings.append(Finding(
212 key=f"ir_drop/{ird.net}", kind="ir_drop",
213 severity="warning" if ird.failing_components else "info",
214 zone=None,
215 detail={**ird.to_dict()},
216 ))
217
218 self.findings = findings
219 self.analyses += 1
220 self.last_analysis_ts = time.time()
221 self.last_analysis_mslast_analysis_ms = (time.perf_counter() - t0) * 1000.0
222 logger.info(
223 "analyse multiphysique terminée",
224 extra={
225 "project_id": self.project_id, "findings": len(findings),
226 "duration_ms": round(self.last_analysis_mslast_analysis_ms, 1),
227 },
228 )
229 return findings
230 finally:
231 self._lock.release()

242 def _publish(self, kind: ConstraintKind, key: str, value: dict) -> None:
243 """Publie sur le bus si la valeur a changé (déduplication par hash JSON)."""
244 blob = json.dumps(value, sort_keys=True, default=str)
245 digest = hashlib.sha1(blob.encode()).hexdigest()
246 if self._published_hashes.get(key) == digest:
247 return
248 self._published_hashes[key] = digest
249 self._bus.publish(ConstraintMessage(
250 kind=kind, key=key, value=value,
251 project_id=self.project_id, source=self.emitter,
252 ))
253 self.published_count += 1
