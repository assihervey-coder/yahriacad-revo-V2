72 balance_usd: float = 50.0
73 ledger_path: Path = Path("data/projects/credits.json")

77 def debit(self, project_id: str, item: Pricing, user: str = "system",
78 quantity: int = 1) -> CreditTransaction:
79 cost = round(item.value * quantity, 4)
80 with self._lock:
81 if cost > self.balance_usd:
82 raise InsufficientCredits(
83 f"solde insuffisant : {self.balance_usd:.2f} USD requis {cost:.2f} USD ({item.label})"
84 )
85 self.balance_usd = round(self.balance_usd - cost, 4)
86 tx = CreditTransaction(
87 tx_id=uuid.uuid4().hex[:12],
88 project_id=project_id,
89 item=item.value,
90 amount_usd=cost,
91 balance_after_usd=self.balance_usd,
92 ts=time.time(),
93 user=user,
94 )
95 self.transactions.append(tx)
96 self._persist()
97 return tx

131 def _persist(self) -> None:
132 try:
133 self.ledger_path.parent.mkdir(parents=True, exist_ok=True)
134 payload = {
135 "balance_usd": self.balance_usd,
136 "transactions": [t.__dict__ for t in self.transactions[-500:]],

138 self.ledger_path.write_text(json.dumps(payload, indent=2))
