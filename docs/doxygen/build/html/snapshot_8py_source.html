92 """Reconstruit une Board depuis le dict produit par `board_to_dict`."""
93 from common.design_model import Layer
94
95 board = Board(width_mm=data.get("width_mm", 100.0), height_mm=data.get("height_mm", 80.0))
96 if data.get("layers"):
97 board.layers = [Layer(index=l["index"], name=l["name"],
98 thickness_um=l.get("thickness_um", 35.0),
99 is_copper=l.get("is_copper", True)) for l in data["layers"]]
100 for comp in data.get("components", []):
101 pads = [Pad(name=p["name"], x_mm=p["x_mm"], y_mm=p["y_mm"],
102 diameter_mm=p.get("diameter_mm", 0.6), net=p.get("net"))
103 for p in comp.get("pads", [])]
104 board.components[comp["ref"]] = Component(
105 ref=comp["ref"], mpn=comp.get("mpn", ""), value=comp.get("value", ""),
106 footprint=comp.get("footprint", ""), pins=comp.get("pins", 0),
107 width_mm=comp.get("width_mm", 5.0), height_mm=comp.get("height_mm", 5.0),
108 power_w=comp.get("power_w", 0.0), price_usd=comp.get("price_usd", 0.0),
109 stock=comp.get("stock", 0), functional_block=comp.get("functional_block", ""),
110 pads=pads,
111 )
112 for pl in data.get("placements", []):
113 board.placements[pl["ref"]] = Placement(
114 ref=pl["ref"], x_mm=pl.get("x_mm", 0.0), y_mm=pl.get("y_mm", 0.0),
115 rotation_deg=pl.get("rotation_deg", 0.0), layer=pl.get("layer", 0),
116 locked=pl.get("locked", False),
117 )
118 for net in data.get("nets", []):
119 segments = [Segment(net=s["net"], x1_mm=s["x1_mm"], y1_mm=s["y1_mm"],
120 x2_mm=s["x2_mm"], y2_mm=s["y2_mm"], layer=s["layer"],
121 width_mm=s.get("width_mm", 0.2), is_via=s.get("is_via", False))
122 for s in net.get("routed_segments", [])]
123 board.nets[net["name"]] = Net(
124 name=net["name"],
125 connections=[tuple(c) for c in net.get("connections", [])],
126 net_class=net.get("net_class", "default"),
127 impedance_target_ohm=net.get("impedance_target_ohm"),
128 length_match_group=net.get("length_match_group"),
129 routed_segments=segments,
130 layer_allowlist=net.get("layer_allowlist", list(range(8))),
131 )
132 board.zones = [Zone(name=z["name"], x_min_mm=z["x_min_mm"], y_min_mm=z["y_min_mm"],
133 x_max_mm=z["x_max_mm"], y_max_mm=z["y_max_mm"], kind=z.get("kind", "keepout"),
134 max_temp_c=z.get("max_temp_c"))
135 for z in data.get("zones", [])]
136 return board

158 def canonical_payload(self) -> Dict[str, Any]:
159 """Charge utile checksumée (tout sauf le checksum lui-même)."""
160 return {k: v for k, v in asdict(self).items() if k != "checksum"}

162 def to_json(self) -> Dict[str, Any]:
163 payload = self.canonical_payload()
164 payload["checksum"] = self.checksum
165 return payload

184def compute_checksum(payload: Dict[str, Any]) -> str:
185 """SHA-256 du JSON canonique (clés triées) — détecte toute altération."""
186 blob = json.dumps(payload, sort_keys=True, ensure_ascii=False, separators=(",", ":"))
187 return hashlib.sha256(blob.encode("utf-8")).hexdigest()

190def capture(
191 project_id: str,
192 session_id: str,
193 board: Board,
194 design_version: int = 1,
195 engine_position: Optional[Dict[str, Any]] = None,
196 inflight_events: Optional[List[Dict[str, Any]]] = None,
197 compute_residue: Optional[Dict[str, Any]] = None,
198 channel: str = "browser",
199) -> SessionSnapshot:
200 """Capture un snapshot cohérent — checksum calculé sur le contenu canonique."""
201 snapshot = SessionSnapshot(
202 schema_version=SNAPSHOT_SCHEMA_VERSION,
203 project_id=project_id,
204 session_id=session_id,
205 created_at=time.time(),
206 channel=channel,
207 design=board_to_dict(board),
208 design_version=design_version,
209 engine_position=engine_position or {},
210 inflight_events=sorted(inflight_events or [], key=lambda e: e.get("seq", 0)),
211 compute_residue=compute_residue or {},
212 )
213 snapshot.checksum = compute_checksum(snapshot.canonical_payload())
214 logger.info("snapshot de session capturé",
215 extra={"project_id": project_id, "session_id": session_id,
216 "events": len(snapshot.inflight_events)})
217 return snapshot

220def snapshot_path(directory: Path, project_id: str, session_id: str) -> Path:
221 """Chemin du fichier snapshot (identifiants assainis)."""
222 safe = lambda text: "".join(c if c.isalnum() or c in "-_" else "_" for c in text) # noqa: E731
223 return Path(directory) / f"{SNAPSHOT_PREFIX}{safe(project_id)}__{safe(session_id)}.json"

226def save(snapshot: SessionSnapshot, directory: Path) -> Path:
227 """Écrit le snapshot (atomique : fichier temporaire + rename)."""
228 path = snapshot_path(directory, snapshot.project_id, snapshot.session_id)
229 path.parent.mkdir(parents=True, exist_ok=True)
230 tmp = path.with_suffix(".tmp")
231 tmp.write_text(json.dumps(snapshot.to_json(), ensure_ascii=False, indent=1),
232 encoding="utf-8")
233 tmp.replace(path)
234 logger.info("snapshot persisté", extra={"path": str(path)})
235 return path

238def load(path: Path) -> SessionSnapshot:
239 """Charge un snapshot en vérifiant l'intégrité — SnapshotIntegrityError si corrompu."""
240 raw = json.loads(Path(path).read_text(encoding="utf-8"))
241 snapshot = SessionSnapshot.from_json(raw)
242 expected = compute_checksum(snapshot.canonical_payload())
243 if snapshot.checksum != expected:
244 raise SnapshotIntegrityError(
245 f"checksum invalide : {snapshot.checksum[:12]}… ≠ {expected[:12]}…")
246 if snapshot.schema_version != SNAPSHOT_SCHEMA_VERSION:
247 logger.warning("version de snapshot différente — reprise best-effort",
248 extra={"found": snapshot.schema_version,
249 "expected": SNAPSHOT_SCHEMA_VERSION})
250 return snapshot
