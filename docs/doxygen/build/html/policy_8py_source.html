1"""Policy ε-greedy — score = récompense prédite − λ·risque, sur les prédictions
2du world model.
3
4La policy n'exécute jamais une action réelle pour l'évaluer : elle interroge
5``DreamerWorldModel.predict_consequences`` (le « rêve »), convertit les trois
6risques en récompense, pénalise le pire scénario (λ·risque) et explore avec
7une probabilité ε. L'apprentissage est supervisé par les verdicts du
8self_verifier : expériences positives = actions « keep » acceptées, négatives
9= rollbacks — la policy apprend à rêver juste, pas à avoir de la chance.
10"""

21# pondération de la récompense sur les trois conséquences prédites
22_W_CONGESTION, _W_THERMAL, _W_SI = 0.4, 0.3, 0.3
23_EPSILON_MIN = 0.01
24_NEGATIVE_FACTOR = 1.15 # cible majorée pour les rollbacks
25_POSITIVE_FACTOR = 0.85 # cible minorée pour les keepers

41 def __init__(self, world_model: Any, epsilon: float = 0.15,
42 lambda_risk: float = 0.5, seed: int = 42) -> None:
43 self.world_model = world_model
44 self.epsilon = epsilon
45 self.lambda_risk = lambda_risk
46 self._rng = random.Random(seed)
47 self._last_features: List[np.ndarray] = []
48 self.n_positive = 0
49 self.n_negative = 0

51 # ---- évaluation --------------------------------------------------------------
52 def score_action(self, board: Any, action: Any) -> Tuple[float, Dict[str, float]]:
53 """(score, détails) : score = reward − λ·risk sur les prédictions."""
54 prediction = self.world_model.predict_consequences(board, action)
55 reward = 1.0 - (_W_CONGESTION * prediction["congestion_score"]
56 + _W_THERMAL * prediction["thermal_rise_estimate"]
57 + _W_SI * prediction["si_risk"])
58 risk = max(prediction["congestion_score"], prediction["thermal_rise_estimate"],
59 prediction["si_risk"])
60 return reward - self.lambda_risk * risk, prediction
61
62 # ---- sélection -----------------------------------------------------------------
63 def select_action(self, board: Any, candidates: Sequence[Any],
64 legal_mask: Optional[np.ndarray] = None) -> Tuple[Optional[Any], Dict[str, Any]]:
65 """Meilleure candidate (ou exploration ε) — None si tout est illégal."""
66 legal = [(i, a) for i, a in enumerate(candidates)
67 if legal_mask is None or i < len(legal_mask) and bool(legal_mask[i])]
68 if not legal:
69 return None, {"reason": "aucune action légale", "explored": False}
70 if self._rng.random() < self.epsilon: # exploration
71 idx, action = self._rng.choice(legal)
72 score, prediction = self.score_action(board, action)
73 action.predicted_reward = score
74 return action, {"explored": True, "score": score, "prediction": prediction}
75 best, best_score, best_pred = None, -float("inf"), {}
76 for _i, action in legal:
77 score, prediction = self.score_action(board, action)
78 if score > best_score:
79 best, best_score, best_pred = action, score, prediction
80 best.predicted_reward = best_score
81 return best, {"explored": False, "score": best_score, "prediction": best_pred}

83 def rank_candidates(self, board: Any, candidates: Sequence[Any]) -> List[Any]:
84 """Candidates triées par score décroissant (pour propose_placement)."""
85 scored = [(self.score_action(board, a)[0], a) for a in candidates]
86 scored.sort(key=lambda pair: -pair[0])
87 for score, action in scored:
88 action.predicted_reward = score
89 return [a for _s, a in scored]

91 # ---- apprentissage -----------------------------------------------------------------
92 def learn_from(self, experiences: Sequence[Experience]) -> int:
93 """Alimente le world model : keepers → cibles améliorées, rollbacks → dégradées."""
94 if not experiences:
95 return 0
96 rows_X: List[np.ndarray] = []
97 rows_Y: List[np.ndarray] = []
98 for exp in experiences:
99 if not self._last_features:
100 break # pas de features mémorisées → rien à apprendre
101 features = self._last_features[-1]
102 base = 1.0 - exp.reward # niveau de risque observé
103 factor = _POSITIVE_FACTOR if exp.kept else _NEGATIVE_FACTOR
104 target = np.full(3, min(1.0, base * factor), dtype=np.float64)
105 rows_X.append(features)
106 rows_Y.append(target)
107 self.n_positive += 1 if exp.kept else 0
108 self.n_negative += 0 if exp.kept else 1
109 if not rows_X:
110 return 0
111 self.world_model.fit_arrays(np.vstack(rows_X), np.vstack(rows_Y))
112 return len(rows_X)
