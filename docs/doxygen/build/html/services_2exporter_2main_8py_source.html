51def export(board: Board, project_id: str = "demo", factory_profile: str = "pcbway",
52 outputs: List[str] | None = None, out_dir: Path | None = None,
53 design_version: int = 1) -> ExportResult:
54 """Exporte le design : dernière vérification usine → fichiers → archive.
55
56 outputs : sous-ensemble de {"gerber", "odb", "bom", "pick_place"}.
57 La vérification finale interdit l'export si une erreur DRC/DFM subsiste —
58 sauf warnings, journalisés comme avertissements de l'archive.
59 """
60 get_settings("exporter")
61 outputs = outputs or ["gerber", "bom", "pick_place"]
62 out_dir = out_dir or Path("data/projects") / project_id / "exports"
63 out_dir.mkdir(parents=True, exist_ok=True)
64
65 # 1. dernière vérification contre le profil de l'usine cible
66 warnings: List[str] = []
67 try:
68 from backend.services.drc_dfm_engine.main import check as drc_check
69
70 reply = drc_check(board, project_id=project_id, factory_profile=factory_profile)
71 if not reply.pass_:
72 errors = [v for v in reply.violations if v.severity == "error"]
73 if errors:
74 return ExportResult(ok=False, warnings=[
75 f"{v.rule_id}: {v.message}" for v in errors[:5]
76 ])
77 warnings.extend(v.message for v in reply.violations if v.severity == "warning")
78 except ImportError:
79 warnings.append("drc_dfm_engine indisponible — vérification finale sautée (dev only)")
80 logger.warning("drc_dfm_engine non importable — vérification finale sautée")
81
82 # 2. génération des fichiers demandés
83 files: Dict[str, str] = {}
84 if "gerber" in outputs:
85 files.update(export_gerbers(board))
86 files["drill.drl"] = export_drill(board)
87 if "odb" in outputs:
88 odb = export_odb(board, project_id)
89 for rel, content in odb.files.items():
90 files[f"odb/{rel}"] = content
91 if "bom" in outputs:
92 bom_csv, _subtotals = export_bom(board)
93 files["bom.csv"] = bom_csv
94 if "pick_place" in outputs:
95 files["pick_place.csv"] = export_pick_place(board)
96
97 # 3. emballage + checksums
98 ts = time.strftime("%Y%m%d_%H%M%S", time.gmtime())
99 archive_path = out_dir / f"{project_id}_export_{ts}.zip"
100 sha256 = hashlib.sha256()
101 with zipfile.ZipFile(archive_path, "w", zipfile.ZIP_DEFLATED) as zf:
102 for name, content in sorted(files.items()):
103 data = content.encode("utf-8")
104 sha256.update(data)
105 zf.writestr(name, data)
106 manifest = (
107 f"project: {project_id}\nfactory: {factory_profile}\n"
108 f"outputs: {', '.join(outputs)}\nfiles: {len(files)}\n"
109 f"warnings: {len(warnings)}\n"
110 )
111 zf.writestr("MANIFEST.txt", manifest)
112
113 result = ExportResult(ok=True, archive_path=str(archive_path), files=sorted(files),
114 sha256=sha256.hexdigest(), warnings=warnings)
115 make_event(EventType.EXPORT_READY, project_id, "exporter", design_version,
116 archive=str(archive_path), sha256=result.sha256, files=len(files),
117 factory=factory_profile)
118 logger.info("export terminé", extra={"archive": str(archive_path), "files": len(files)})
119 return result
