122def get_settings(service_name: str | None = None, **overrides) -> Settings:
123 """Retourne la configuration courante (singleton par service).
124
125 `service_name` est réinjecté si fourni — chaque binaire appelle

128 global _SETTINGS
129 if _SETTINGS is None or service_name is not None or overrides:
130 _SETTINGS = Settings()
131 if service_name:
132 _SETTINGS.service_name = service_name
133 for key, value in overrides.items():
134 setattr(_SETTINGS, key, value)
135 return _SETTINGS
