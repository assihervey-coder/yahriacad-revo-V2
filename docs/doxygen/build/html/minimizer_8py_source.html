48 def minimize(self, board: Board) -> Tuple[Board, MinimizerStats]:
49 stats = MinimizerStats()
50 stats.vias_before = board.via_count()
51
52 for net in board.nets.values():
53 if not net.routed_segments:
54 continue
55 # Passe 1 — fusion des collinearités et absorption des détours en L
56 net.routed_segments = self._merge_collinear(net.routed_segments)
57 # Passe 2 — suppression des vias redondants (même position, même paire de couches)
58 net.routed_segments = self._drop_redundant_vias(net.routed_segments)
59
60 stats.segments_merged = self._merged_count(board)
61 stats.vias_after = board.via_count()
62 if stats.vias_before > 0:
63 stats.reduction_pct = round(
64 100.0 * (stats.vias_before - stats.vias_after) / stats.vias_before, 1
65 )
66 # Levier 3 : la propagation préférentielle sur couche courante est déjà
67 # en amont (VIA_COST du pathfinder) — ici on journalise son apport.
68 if stats.reduction_pct > 0:
69 stats.notes.append(
70 f"réduction {stats.reduction_pct} % (cible DeepPCB : −{TARGET_REDUCTION_PCT:.0f} %)"
71 )
72 logger.info("via_minimizer terminé",
73 extra={"vias_before": stats.vias_before, "vias_after": stats.vias_after,
74 "reduction_pct": stats.reduction_pct})
75 return board, stats

78 def _merge_collinear(self, segments: List[Segment]) -> List[Segment]:
79 """Fusionne les segments colinéaires consécutifs du même net/couche."""
80 out: List[Segment] = []
81 for seg in segments:
82 if seg.is_via:
83 out.append(seg)
84 continue
85 merged = False
86 for i, prev in enumerate(out):
87 if prev.is_via or prev.layer != seg.layer or prev.is_via:
88 continue
89 if self._touches(prev, seg) and self._collinear(prev, seg):
90 out[i] = self._union(prev, seg)
91 merged = True
92 break
93 if not merged:
94 out.append(seg)
95 return out

97 def _drop_redundant_vias(self, segments: List[Segment]) -> List[Segment]:
98 """Supprime un via si les deux segments adjacents sont déjà sur la même couche."""
99 cleaned: List[Segment] = []
100 for i, seg in enumerate(segments):
101 if seg.is_via:
102 prev_tr = next((s for s in reversed(cleaned) if not s.is_via), None)
103 next_tr = next((s for s in segments[i + 1:] if not s.is_via), None)
104 if prev_tr is not None and next_tr is not None and prev_tr.layer == next_tr.layer:
105 continue # via redondant : la piste reste sur la même couche
106 cleaned.append(seg)
107 return cleaned
