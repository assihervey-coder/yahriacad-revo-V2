41def load_design(path: Path) -> ReferenceDesign:
42 """Charge un design JSON du corpus et valide les champs obligatoires."""
43 raw = json.loads(path.read_text(encoding="utf-8"))
44 for key in ("id", "name", "description", "components", "nets"):
45 if key not in raw:
46 raise ValueError(f"{path}: champ obligatoire manquant '{key}'")
47 if not raw["components"] or not raw["nets"]:
48 raise ValueError(f"{path}: un design de référence doit avoir composants ET nets")
49 return ReferenceDesign(
50 id=str(raw["id"]),
51 name=str(raw["name"]),
52 description=str(raw["description"]),
53 components=list(raw["components"]),
54 nets=list(raw["nets"]),
55 constraints=dict(raw.get("constraints", {})),
56 expected_metrics=dict(raw.get("expected_metrics", {})),
57 )

60def load_corpus(corpus_dir: Path) -> list[ReferenceDesign]:
61 """Charge tous les designs du corpus (tri par id — ordre déterministe)."""
62 if not corpus_dir.is_dir():
63 raise FileNotFoundError(f"corpus introuvable : {corpus_dir}")
64 designs = [load_design(p) for p in sorted(corpus_dir.glob("*.json"))]
65 if not designs:
66 raise ValueError(f"corpus vide : {corpus_dir}")
67 ids = [d.id for d in designs]
68 if len(set(ids)) != len(ids):
69 raise ValueError(f"ids dupliqués dans le corpus : {ids}")
70 return designs
