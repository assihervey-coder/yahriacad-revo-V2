51 # ---- construction ------------------------------------------------------------
52 def fit(self, docs: List[Document]) -> None:
53 """(Re)construit vocabulaire, IDF et matrice depuis une liste de documents."""
54 self._docs_docs_docs = list(docs)
55 self._vocab = {}
56 for doc in self._docs_docs_docs:
57 for token in _tokenize(doc.content):
58 self._vocab.setdefault(token, len(self._vocab))
59 n_docs = max(1, len(self._docs_docs_docs))
60 df = np.zeros(len(self._vocab), dtype=np.float64)
61 for doc in self._docs_docs_docs:
62 for token in set(_tokenize(doc.content)):
63 df[self._vocab[token]] += 1.0
64 self._idf = np.log((1.0 + n_docs) / (1.0 + df)) + 1.0 # lissage type sklearn
65 self._matrix = np.zeros((len(self._docs_docs_docs), len(self._vocab)), dtype=np.float64)
66 for i, doc in enumerate(self._docs_docs_docs):
67 self._matrix[i] = self._vectorize(_tokenize(doc.content))
68
69 def upsert(self, docs: List[Document]) -> None:
70 """Ajoute des documents puis reconstruit l'index (idempotent et simple)."""
71 self.fit(list(self._docs_docs_docs) + list(docs))

73 # ---- vecteurs ------------------------------------------------------------------
74 def _vectorize(self, tokens: List[str]) -> np.ndarray:
75 """Sac de mots TF-IDF L2-normalisé pour une liste de tokens."""
76 vec = np.zeros(len(self._vocab), dtype=np.float64)
77 for token in tokens:
78 idx = self._vocab.get(token)
79 if idx is not None:
80 vec[idx] += 1.0
81 vec *= self._idf
82 norm = np.linalg.norm(vec)
83 return vec / norm if norm > 0 else vec

85 def search(self, query: str, k: int = 4) -> List[Tuple[Document, float]]:
86 """Retourne les (document, score) les plus proches de la requête."""
87 if self._matrix is None or not self._docs_docs_docs:
88 return []
89 q = self._vectorize(_tokenize(query))
90 scores = self._matrix @ q
91 order = np.argsort(-scores)[: max(1, k)]
92 return [(self._docs_docs_docs[int(i)], float(scores[int(i)]))
93 for i in order if scores[i] > 0.0]

115 def load(self, directory: Optional[Path] = None, name: str = "tfidf_store.npz") -> bool:
116 """Recharge un store persisté ; False si absent ou corrompu."""
117 target = Path(directory) if directory else Path(get_settings().vector_store_dir)
118 path = target / name
119 if not path.exists():
120 return False
121 try:
122 with np.load(path, allow_pickle=False) as data:
123 self._vocab = {t: i for i, t in enumerate(json.loads(str(data["vocab_json"])))}
124 self._idf = data["idf"]
125 self._matrix = data["matrix"]
126 raw_docs = json.loads(str(data["docs_json"]))
127 self._docs_docs_docs = [Document(content=d["content"], metadata=d.get("metadata", {}))
128 for d in raw_docs]
129 return True
130 except (OSError, ValueError, KeyError):
131 return False
