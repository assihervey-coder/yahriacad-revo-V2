2
3Fonctionnement : AVANT chaque action, ``push_state`` empile une copie complète
4du Board + une entrée de journal ; si la vérification échoue après application,
5``rollback_last`` restaure l'état précédent, émet un événement

53 def rollback_last(self, board: Any = None, reason: str = "",
54 violated: Optional[List[str]] = None,
55 constraint: Optional[ConstraintMessage] = None,

57 """Restaure le dernier état empilé, journalise et réinjecte la contrainte.
58
59 ``board`` (optionnel) est l'objet design courant du chiamant : il est
60 restauré EN PLACE depuis le snapshot (identité préservée pour tous les
61 détenteurs de la référence). Retourne le résumé d'annulation, ou None

63 """
64 if not self._stack:
65 logger.warning("rollback demandé sur pile vide — état initial conservé")
66 return None
67 snapshot = self._stack.pop()
68 if board is not None:
69 restored = snapshot["board"]
70 board.components = restored.components
71 board.placements = restored.placements
72 board.nets = restored.nets
73 board.zones = restored.zones
74 board.width_mm = restored.width_mm
75 board.height_mm = restored.height_mm
76 self.n_rollbacks += 1
77 violated = list(violated or [])
78 summary = {
79 "reason": reason,
80 "violated_constraints": violated,
81 "restored_entry": snapshot["entry"],
82 "remaining_snapshots": len(self._stack),
83 }
84 # 1) événement d'audit
85 event = make_event(
86 EventType.ROLLBACK_PERFORMED, self.project_id, emitter=self.emitter,
87 reason=reason, violated_constraints=violated,
88 action=action or {}, restored_drc=snapshot["entry"].get("drc_score"),
89 )
90 self._events.append(event)
91 # 2) réinjection de la contrainte violée sur le bus (les agents réagissent)
92 if constraint is not None and self.bus is not None:
93 self.bus.publish(constraint)
94 # 3) expérience négative pour la policy
95 self._negative.append({
96 "ts": time.time(), "reason": reason, "violated": violated,
97 "action": action or {}, "reward": 0.0, "kept": False,
98 })
99 logger.info("rollback effectué", extra=summary)
100 return summary
