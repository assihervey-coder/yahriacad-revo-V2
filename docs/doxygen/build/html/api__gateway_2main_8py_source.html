112 def guard(context: dict[str, Any]) -> None:
113 """Applique auth + rate limit avant chaque dispatch (ordre : 401 puis 429)."""
114 if context.get("path", "") in PUBLIC_PATHS:
115 return
116 headers = context.get("headers") or {}
117 try:
118 identity = auth_service.authenticate(headers)
119 rate_limiter.check(identity.sub, tier=identity.tier,
120 action=context.get("path", "/"))
121 except AuthError as exc:
122 raise HTTPException(exc.status_code, exc.detail)
123 except RateLimitExceeded as exc:
124 raise HTTPException(429, str(exc))
125 context["identity"] = identity
126
