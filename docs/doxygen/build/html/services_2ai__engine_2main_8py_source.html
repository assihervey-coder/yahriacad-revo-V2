129 # ---- RPC 1 : rl_agent ---------------------------------------------------------------------
130 def propose_placement(self, component_ref: str, candidates: int = 8) -> List[Action]:
131 """Actions candidates pour un composant, triées par récompense prédite."""
132 if self.board is None:
133 raise RuntimeError("aucun design chargé — appeler load_board() d'abord")
134 board = self.board
135 anchor = self._net_anchor(board, component_ref)
136 pool = self.action_space.anchored_candidates(board, component_ref,
137 max(candidates * 3, 24), anchor)
138 if not pool: # carte saturée : échantillon brut
139 pool = self.action_space.sample((board.width_mm, board.height_mm),
140 candidates * 4, component_ref, anchor=None)
141 for action in pool:
142 self.policy.remember_features(board, action)
143 return self.policy.rank_candidates(board, pool)[:max(0, candidates)]

145 def _net_anchor(self, board: Board, ref: str) -> tuple:
146 """Centroïde des partenaires de nets — l'ancre sémantique du sampling."""
147 points: List[tuple] = []
148 for net in board.nets.values():
149 refs = {r for r, _p in net.connections}
150 if ref in refs:
151 points += [(board.placements[r].x_mm, board.placements[r].y_mm)
152 for r in refs - {ref} if r in board.placements]
153 if not points:
154 return (board.width_mm / 2.0, board.height_mm / 2.0)
155 return (sum(p[0] for p in points) / len(points),
156 sum(p[1] for p in points) / len(points))

158 # ---- RPC 2 : self_verifier -----------------------------------------------------------------
159 def verify_action(self, action: Action) -> Dict[str, Any]:
160 """Vérification SANS engagement (équivalent VerifyActionReply du proto)."""
161 if self.board is None:
162 raise RuntimeError("aucun design chargé — appeler load_board() d'abord")
163 verdict = self.checker.check(self.board, action)
164 return {"valid": verdict.valid, "violated_constraints": verdict.violated_constraints,
165 "reason": verdict.reason, "rolled_back": False}

167 def try_action(self, action: Action) -> Dict[str, Any]:
168 """Applique puis vérifie ; rollback propre + expérience négative si invalide."""
169 board = self.board
170 assert board is not None
171 self.rollback.push_state(board, {"note": f"avant action sur {action.ref}"})
172 try:
173 board.move(action.ref, action.x_mm, action.y_mm,
174 rotation_deg=getattr(action, "rotation_deg", None))
175 except PermissionError as exc: # verrou surgical_edit humain
176 self.rollback.rollback_last(board, reason=str(exc), violated=["locked"],
177 action=action.to_json())
178 return {"valid": False, "violated_constraints": ["locked"],
179 "reason": f"composant verrouillé : {exc}", "rolled_back": True}
180 verdict = self.checker.check(board, action)
181 if verdict.valid:
182 self._emit(EventType.COMPONENT_MOVED, ref=action.ref,
183 x_mm=action.x_mm, y_mm=action.y_mm,
184 rotation_deg=getattr(action, "rotation_deg", 0.0))
185 return {"valid": True, "violated_constraints": [], "reason": verdict.reason,
186 "rolled_back": False}
187 # ---- verdict invalide : CONSTRAINT_VIOLATED + rollback + apprentissage négatif ----
188 self._emit(EventType.CONSTRAINT_VIOLATED, ref=action.ref,
189 violated=verdict.violated_constraints, reason=verdict.reason)
190 constraint = self.bus.latest().get(
191 verdict.violated_constraints[0]) if verdict.violated_constraints else None
192 self.rollback.rollback_last(board, reason=verdict.reason,
193 violated=verdict.violated_constraints,
194 constraint=constraint, action=action.to_json())
195 self.world_model.observe(board, action, {"congestion_score": 1.0,
196 "thermal_rise_estimate": 1.0,
197 "si_risk": 1.0})
198 self.policy.learn_from([Experience(ref=action.ref, reward=0.0, kept=False)])
199 return {"valid": False, "violated_constraints": verdict.violated_constraints,
200 "reason": verdict.reason, "rolled_back": True}

202 # ---- RPC 3 : autonomous_optimizer ------------------------------------------------------------
203 def run_night_optimization(self, max_iter: int = 300,
204 warm_start_path: Optional[Path] = None,
205 save_path: Optional[Path] = None) -> Iterator[Dict[str, Any]]:
206 """Générateur d'IterationReport — boucle proposer → vérifier → évaluer → ratchet.
207
208 Branchement ``world_model_torch.npz`` : à l'ouverture de la nuit, les
209 poids exportés par la passe RL torch (format miroir du TinyNet numpy)
210 sont chargés en warm start depuis ``warm_start_path`` — défaut :
211 ``settings.world_model_npz`` (``PCB_WORLD_MODEL_NPZ``) — si le fichier
212 existe. En fin de nuit, les poids mis à jour par les
213 ``observe()``/``train()`` des itérations gardées sont réécrits au même
214 slot : chaque nuit repart du modèle de la veille. La sauvegarde passe
215 dans un ``finally`` — elle s'exécute même si le consommateur referme le
216 générateur avant épuisement. Bilan disponible via ``last_night_summary``.
217 """
218 board = self.board
219 if board is None:
220 raise RuntimeError("aucun design chargé — appeler load_board() d'abord")
221 warm_info = self._warm_start_world_model(warm_start_path)
222 try:
223 detail = self.evaluator.evaluate_detailed(board)
224 baseline = self._composite(detail, board)
225 self.keeper.set_baseline(baseline)
226 stagnant = 0 # itérations consécutives sans gain
227 for _iteration in range(max_iter):
228 proposals = self.proposer.propose(board, self.intent_graph, n=3)
229 if not proposals:
230 break
231 progressed = False
232 for proposal in proposals:
233 trial = copy.deepcopy(board)
234 if not apply_proposal(trial, proposal):
235 continue
236 # 1) self_verifier d'abord : chaque position finale issue de la
237 # proposition (move/rotate/swap) est vérifiée sur la copie.
238 actions = self._actions_for_trial(trial, proposal)
239 if not actions or any(not self.checker.check(trial, a).valid for a in actions):
240 continue
241 # 2) fast_evaluator (< 5 s garanti) puis règle du ratchet.
242 trial_detail = self.evaluator.evaluate_detailed(trial)
243 score = self._composite(trial_detail, trial)
244 if not self.keeper.keep_if_better(proposal.to_json(), score):
245 continue # ratchet : rejet
246 self.rollback.push_state(board, {"note": f"keep iter {self.keeper.iterations}"})
247 apply_proposal(board, proposal) # engagement réel
248 self.world_model.observe(board, actions[0], {
249 "congestion_score": trial_detail["si_risk"],
250 "thermal_rise_estimate": 1.0 - trial_detail["thermal_ok"],
251 "si_risk": trial_detail["si_risk"]})
252 self.world_model.train()
253 self._emit(EventType.PLAN_UPDATED, iteration=self.keeper.iterations,
254 best_score=self.keeper.best_score)
255 progressed = True
256 yield self.keeper.reports[-1].to_json()
257 self.policy.anneal()
258 stagnant = 0 if progressed else stagnant + 1
259 if stagnant >= 10:
260 break # dix passes sans aucun gain : convergé
261 finally:
262 self._finalize_night(warm_info, save_path)

296 def _composite(self, detail: Dict[str, Any], board: Board) -> float:
297 """Score composite du keeper depuis une évaluation détaillée."""
298 return self.keeper.composite_score(drc=float(detail["drc_proxy"]),
299 si=1.0 - float(detail["si_risk"]),
300 thermal=float(detail["thermal_ok"]),
301 vias=board.via_count())

303 @staticmethod
304 def _actions_for_trial(trial_board: Board, proposal: Proposal) -> List[Action]:
305 """Actions à vérifier pour une proposition, lues sur la copie APRÈS
306 application (positions finales) — move/rotate → 1 action, swap → 2."""
307 payload = proposal.json
308 kind = payload.get("action")
309 refs = []
310 if kind in ("move", "rotate"):
311 refs = [payload.get("ref")]
312 elif kind == "swap":
313 refs = [payload.get("ref_a"), payload.get("ref_b")]
314 actions: List[Action] = []
315 for ref in refs:
316 placement = trial_board.placements.get(ref) if ref else None
317 if placement is None:
318 return []
319 actions.append(Action(ref=ref, x_mm=placement.x_mm, y_mm=placement.y_mm,
320 rotation_deg=placement.rotation_deg, layer=placement.layer))
321 return actions

398def self_test() -> Dict[str, Any]:
399 """Self-test complet du cerveau IA — imprime et retourne le rapport."""
400 engine = AiEngine("proj-selftest")
401 report: Dict[str, Any] = {}
402
403 # ---- 0) chargement : graphe d'intention + bus de contraintes ---------------------
404 report["load"] = engine.load_board(build_demo_board())
405 print("[5.1] graphe d'intention + bus :", json.dumps(report["load"], ensure_ascii=False))
406
407 # ---- 1) rl_agent : proposition de placements --------------------------------------
408 proposals = engine.propose_placement("U1", candidates=5)
409 report["proposals_u1"] = [a.to_json() for a in proposals]
410 print("[5.3] propose_placement(U1, 5) :", json.dumps(report["proposals_u1"], ensure_ascii=False))
411
412 # ---- 2) self_verifier : vérification des actions proposées --------------------------
413 verdicts = [engine.verify_action(action) for action in proposals]
414 report["verdicts_valid"] = sum(1 for v in verdicts if v["valid"])
415 assert report["verdicts_valid"] == len(verdicts), "des actions légales ont été rejetées"
416
417 # ---- 3) action invalide volontaire (keepout) → rollback propre ------------------------
418 bad = Action(ref="U2", x_mm=42.0, y_mm=30.0) # dans antenna_keepout
419 reply = engine.verify_action(bad)
420 assert not reply["valid"], "le keepout aurait dû être détecté"
421 applied = engine.try_action(bad)
422 assert applied["rolled_back"] and not applied["valid"]
423 assert abs(engine.board.placements["U2"].x_mm - 12.0) < 1e-6, "rollback non restauré"
424 report["invalid_action"] = {"violated": reply["violated_constraints"],
425 "rolled_back": applied["rolled_back"]}
426 print("[5.4] action invalide + rollback :", json.dumps(report["invalid_action"], ensure_ascii=False))
427
428 # ---- 4) llm_orchestrator : RAG ancré + motifs + prompts --------------------------------
429 engine.rag_index.upsert(chunk_datasheet(_DEMO_DATASHEET, "ADS1115_datasheet.pdf"))
430 rag_answer = engine.rag.answer("What is the maximum junction temperature?", k=2)
431 report["rag"] = {"chunks": engine.rag_index.size, "citations": rag_answer["citations"],
432 "mode": rag_answer["mode"]}
433 violations = engine.pattern_validator.validate(
434 "u1 = Part('MCU', 'STM32', ref='U1')\n"
435 "u3 = Part('Regulator', 'AP2112K', ref='U3')\n"
436 "vcc = Net('VCC')\nsda = Net('SDA')\ngnd = Net('GND')\n"
437 "vcc += u1[1]; sda += u1[2]; gnd += u1[3]\n")
438 report["pattern_violations"] = [v.to_json() for v in violations]
439 learned = engine.pattern_validator.learn_validated(_demo_skidl_valid())
440 for _ in range(3):
441 engine.prompts.feedback(True, "default_placement")
442 engine.prompts.feedback(False, "default_placement")
