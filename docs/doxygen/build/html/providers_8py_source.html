72def load_catalogue() -> list[dict[str, Any]]:
73 """Charge data/component_library/seed_components.json sinon le mini-catalogue.
74
75 Formats tolérés : liste directe de fiches, ou objet {"meta": ...,
76 "components": [...]} (format du seed versionné du dépôt).
77 """
78 candidate = Path("data/component_library/seed_components.json")
79 try:
80 if candidate.exists():
81 parsed = json.loads(candidate.read_text(encoding="utf-8"))
82 if isinstance(parsed, dict):
83 parsed = parsed.get("components", [])
84 if isinstance(parsed, list) and parsed:
85 return [c for c in parsed if isinstance(c, dict)]
86 except (OSError, json.JSONDecodeError):
87 pass

96def _haystack(record: dict[str, Any]) -> str:
97 """Texte de recherche : mpn + value + description + mots-clés."""
98 parts = [str(record.get("mpn", "")), str(record.get("value", "")),
99 str(record.get("description", ""))]
100 parts.extend(str(k) for k in record.get("keywords", []) or [])
101 return " ".join(parts)

104def _filter_local(catalogue: list[dict[str, Any]], query: str,
105 category: str) -> list[dict[str, Any]]:
106 """Filtre local : catégorie (deux nomenclatures tolérées) puis texte libre."""
107 cat = (category or "").lower()
108 hits = [c for c in catalogue if not cat or _category_of(c).lower() == cat]
109 text = (query or "").lower()
110 if text:
111 keyword_hits = [c for c in hits if text in _haystack(c).lower()]
112 hits = keyword_hits or hits # repli : toute la catégorie
113 return hits
