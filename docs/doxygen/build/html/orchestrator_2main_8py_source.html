123 def run_pipeline(self, project_id: str, request_text: str,
124 steps: list[int] | None = None,
125 night_mode: bool = False) -> Iterator[PipelineEvent]:
126 """Exécute le workflow — générateur de PipelineEvent (stream gRPC côté proto)."""
127 started = time.perf_counter()
128 state = self.state_manager.get_or_create(project_id, request_text)
129 plan = self.super_agent.build_plan(project_id, request_text)
130 plan.status = "running"
131 self.state_manager.publish_event(self.super_agent.plan_updated_event(
132 project_id, state.version))
133 selected = steps or list(range(1, 9))
134 context = AgentContext(project_id=project_id, request_text=request_text,
135 board=state.board)
136 self._context = context # partagé avec les handlers via _agent_context()
137 priority = Priority.NIGHT_BATCH if night_mode else Priority.INTERACTIVE
138
139 for index in selected:
140 plan_step = plan.step(index)
141 status = StepStatus(step=index, label=WorkflowStep(index).label, state="running")
142 yield PipelineEvent(step_status=status)
143 self._emit_progress(project_id, state.version, status)
144 step_started = time.perf_counter()
145 attempts = 0
146 max_attempts = 3
147 while True:
148 attempts += 1
149 try:
150 handler = getattr(self, f"_step{index}")
151 run_ctx = {"project_id": project_id, "night_mode": night_mode,
152 "surgical_edit": context.extra.get("surgical_edit")}
153 if plan_step is not None:
154 outcome = self.super_agent.execute_step(
155 plan, plan_step, handler, run_ctx, priority=priority)
156 detail = str(outcome["result"])
157 else:
158 detail = str(handler(run_ctx))
159 status.state, status.detail = "done", detail
160 break
161 except SkipStep as skip:
162 status.state, status.detail = "skipped", str(skip)
163 break
164 except Exception as exc: # noqa: BLE001 — escalade contractualisée
165 decision = self.super_agent.handle_escalation(
166 exc, {"project_id": project_id, "step": index}, attempts=attempts)
167 yield PipelineEvent(escalation=str(decision))
168 if decision["decision"] == "abort" or attempts >= max_attempts:
169 status.state = "failed"
170 status.detail = f"{exc} → {decision['decision']}"
171 break
172 if plan_step is not None:
173 plan_step.retries += 1
174 status.duration_s = round(time.perf_counter() - step_started, 3)
175 if plan_step is not None:
176 plan.replace_step(index, status=status.state, detail=status.detail,
177 retries=plan_step.retries)
178 plan.bump_version()
179 yield PipelineEvent(step_status=status)
180 self._emit_progress(project_id, state.version, status)
181 if status.state == "failed":
182 plan.status = "failed"
183 break
184 else:
185 plan.status = "done"
186
187 metrics = {**self.state_manager.metrics(project_id),
188 "duration_total_s": round(time.perf_counter() - started, 3),
189 "plan_version": plan.version, "night_mode": night_mode}
190 yield PipelineEvent(metrics=metrics)

200 def _step1(self, ctx: dict[str, Any]) -> str:
201 """nl_to_skidl (+constraint_extractor, init graphe d'intention)."""
202 context = self._agent_context(ctx)
203 generated = self.agents["code_generator"].run(context)
204 validated = self.agents["validator"].run(context)
205 corrected = None
206 if not validated.outputs.get("report", {}).get("passed", False):
207 corrected = self.agents["corrector"].run(context)
208 call_adapter("constraint_extractor", context.request_text)
209 call_adapter("intent_graph_init", context.project_id)
210 components = context.extra.get("skidl_components") or []
211 nets = context.extra.get("skidl_nets") or []
212 self._load_netlist(context.project_id, components, nets)
213 self._mark_step(context.project_id, 1)
214 parts = [f"SKiDL ({generated.outputs.get('source')})",
215 f"{len(components)} composant(s)", f"{len(nets)} net(s)"]
216 if corrected is not None:
217 parts.append(f"corrector: {corrected.outputs.get('iterations')} itération(s)")
218 return " · ".join(parts)
