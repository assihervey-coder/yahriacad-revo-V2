1"""Client Ollama natif — le LLM local réellement branché dans le RAG.
2
3Contrairement à l'adaptateur générique « OpenAI-compatible » (retriever._llm_answer,
4qui cible ``/v1/chat/completions``), ce client parle l'API **native** d'Ollama :
5
6 * ``GET /api/tags`` — liste des modèles installés (health-check gratuit) ;
7 * ``POST /api/chat`` — génération conversationnelle (stream=False) ;
8 * ``POST /api/generate`` — génération brute par prompt ;
9 * ``POST /api/embeddings`` — vecteurs d'embedding du modèle embarqué.
10
11Garanties héritées des conventions du dépôt :
12 * httpx reste un import gardé — le module s'importe sans réseau ni dépendance ;
13 * toute erreur (serveur arrêté, timeout, JSON invalide) est convertie en
14 ``None``/``[]`` + log warning : le RAG retombe alors sur le mode extractif
15 déterministe, jamais sur une exception ;
16 * aucun appel réseau à l'import — le client ne contacte Ollama qu'à la
17 première méthode appelée explicitement.
18"""

31class OllamaClient:
32 """Passerelle vers un serveur Ollama local (http://localhost:11434 par défaut).
33
34 Exemple :
35 >>> client = OllamaClient(model="llama3.1:8b")
36 >>> client.is_available()
37 True
38 >>> client.chat([{"role": "user", "content": "Liste les couches d'une carte 4 cuivres."}])
39 'Une carte 4 cuivres empile F.Cu, GND, PWR et B.Cu.'
40 """

42 def __init__(self, base_url: Optional[str] = None, model: Optional[str] = None,
43 timeout_s: Optional[float] = None, num_ctx: Optional[int] = None,
44 settings: Optional[Settings] = None) -> None:
45 self.settings = settings or get_settings("ai_engine")
46 self.base_url = (base_url or self.settings.ollama_base_url).rstrip("/")
47 self.model = model or self.settings.llm_model
48 self.timeout_s = float(timeout_s or self.settings.ollama_timeout_s)
49 self.num_ctx = int(num_ctx or self.settings.ollama_num_ctx)
50 self.last_latency_ms: Optional[float] = None # diagnostic / Prometheus

52 # ---- infrastructure --------------------------------------------------------------
53 def _post(self, path: str, payload: Dict[str, Any],
54 timeout_s: Optional[float] = None) -> Optional[Dict[str, Any]]:
55 """POST JSON vers l'API native — None si httpx absent ou serveur muet."""
56 try:
57 import httpx # import gardé — dépendance optionnelle à l'exécution
58 except ImportError:
59 logger.warning("httpx absent — client Ollama inopérant (pip install httpx)")
60 return None
61 started = time.perf_counter()
62 try:
63 response = httpx.post(f"{self.base_url}{path}", json=payload,
64 timeout=timeout_s or self.timeout_s)
65 response.raise_for_status()
66 self.last_latency_ms = round((time.perf_counter() - started) * 1000.0, 2)
67 return response.json()
68 except Exception as exc: # connexion refusée, timeout, HTTP 5xx, JSON
69 logger.warning("Ollama injoignable (%s%s) — fallback déterministe",
70 self.base_url, path, extra={"error": str(exc)[:160]})
71 return None
72
73 def _get(self, path: str, timeout_s: Optional[float] = None) -> Optional[Dict[str, Any]]:
74 """GET JSON vers l'API native — None si indisponible."""
75 try:
76 import httpx
77 except ImportError:
78 return None
79 try:
80 response = httpx.get(f"{self.base_url}{path}",
81 timeout=timeout_s or self.timeout_s)
82 response.raise_for_status()
83 return response.json()
84 except Exception as exc:
85 logger.warning("Ollama injoignable (GET %s) — %s", self.base_url, str(exc)[:120])
86 return None

89 def is_available(self) -> bool:
90 """True si le serveur répond au health-check ``/api/tags``."""
91 return self._get("/api/tags", timeout_s=min(2.0, self.timeout_s)) is not None

93 def list_models(self) -> List[str]:
94 """Noms des modèles installés (``ollama list`` côté serveur)."""
95 data = self._get("/api/tags", timeout_s=min(2.0, self.timeout_s))
96 if not data:
97 return []
98 return [str(entry.get("name", "")) for entry in data.get("models", []) if entry.get("name")]
99
100 def status(self) -> Dict[str, Any]:
101 """Diagnostic complet — injecté dans les logs de démarrage du service."""
102 models = self.list_models()
103 return {"base_url": self.base_url, "model": self.model,
104 "available": bool(models), "models": models,
105 "timeout_s": self.timeout_s, "num_ctx": self.num_ctx}

107 # ---- génération --------------------------------------------------------------
108 def chat(self, messages: Sequence[Dict[str, str]],
109 temperature: float = 0.2, stream: bool = False) -> Optional[str]:
110 """Génération conversationnelle — contenu texte, ou None si échec.
111
112 ``stream=False`` impose une réponse JSON unique (le RAG n'a pas besoin
113 de streaming server-side ; la latence cible est portée par le WebSocket
114 du frontend, pas par le token-streaming du LLM).
115 """
116 payload = {"model": self.model, "messages": list(messages),
117 "stream": stream, "options": {"num_ctx": self.num_ctx,
118 "temperature": temperature}}
119 data = self._post("/api/chat", payload)
120 if not data:
121 return None
122 message = data.get("message") or {}
123 content = str(message.get("content", "")).strip()
124 return content or None
125
126 def generate(self, prompt: str, system: Optional[str] = None,
127 temperature: float = 0.2) -> Optional[str]:
128 """Génération brute (``/api/generate``) — utile pour les one-shots."""
129 payload: Dict[str, Any] = {"model": self.model, "prompt": prompt,
130 "stream": False,
131 "options": {"num_ctx": self.num_ctx,
132 "temperature": temperature}}
133 if system:
134 payload["system"] = system
135 data = self._post("/api/generate", payload)
136 if not data:
137 return None
138 return str(data.get("response", "")).strip() or None
139
140 def embeddings(self, text: str) -> Optional[List[float]]:
141 """Vecteur d'embedding du modèle serveur — None si échec.
142
143 Utilisé en appoint de l'index TF-IDF : quand Ollama sert un modèle
144 d'embedding (ex. ``bge-m3``, cf. ``EMBEDDING_MODEL``), le retriever
145 peut affiner le classement des passages par similarité cosinus dense.
146 """
147 data = self._post("/api/embeddings",
148 {"model": self.settings.embedding_model, "prompt": text})
149 if not data:
150 return None
151 vector = data.get("embedding")
152 return [float(x) for x in vector] if isinstance(vector, list) else None
