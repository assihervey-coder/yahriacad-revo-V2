2
3Deux transports interchangeables derrière l'interface `Transport` :
4 - `WebSocketClient` : ws://gateway/ws — bibliothèque `websockets` importée

88class WebSocketClient(Transport):
89 """Client WebSocket vers la passerelle — reconnexion exponentielle.
90
91 La bibliothèque `websockets` est importée dans `start()` (import gardé) :
92 si elle est absente, `start()` retourne False et la fabrique
93 `create_transport` bascule sur FileIPC. La latence est mesurée par
94 ping/pong applicatif toutes les `ping_interval_s` et comparée à la cible
95 `websocket_target_latency_ms` (100 ms) — un dépassement est journalisé.
96 """
97
98 def __init__(self, url: str, target_latency_ms: Optional[int] = None,
99 ping_interval_s: float = 5.0) -> None:
100 self.url = url
101 self.target_latency_mstarget_latency_ms = target_latency_ms or get_settings().websocket_target_latency_ms
102 self.ping_interval_s = ping_interval_s
103 self._stats = TransportStats()
104 self._connected = False
105 self._stop = threading.Event()
106 self._thread: Optional[threading.Thread] = None
107 self._inbox: "asyncio.Queue[dict]" = asyncio.Queue()
108 self._loop: Optional[asyncio.AbstractEventLoop] = None
109
110 # -- API synchrone --------------------------------------------------------
111 def start(self) -> bool:
112 try:
113 import websockets # noqa: F401 — import gardé (voir docstring)
114 except ImportError:
115 logger.warning("websockets indisponible — transport WebSocket inutilisable")
116 return False
117 self._stop.clear()
118 self._thread = threading.Thread(
119 target=self._run_loop, name="kicad_ws", daemon=True,
120 )
121 self._thread.start()
122 return True
123
124 def stop(self) -> None:
125 self._stop.set()
126 if self._thread is not None:
127 self._thread.join(timeout=3.0)
128 self._thread = None
129
130 def send(self, message: dict) -> bool:
131 """Dépose un message sur la file d'émission (traité par la boucle asyncio)."""
132 if self._loop is None or not self._connected:
133 return False
134 asyncio.run_coroutine_threadsafe(self._outbox.put(message), self._loop)
135 self._stats.sent += 1
136 return True
137
138 def poll(self, timeout_s: float = 0.0) -> Optional[dict]:
139 try:
140 if self._loop is None:
141 return None
142 fut = asyncio.run_coroutine_threadsafe(self._inbox.get(), self._loop)
143 return fut.result(timeout=timeout_s) if timeout_s > 0 else fut.result(timeout=0.01)
144 except Exception:
145 return None
146
147 @property
148 def connected(self) -> bool:
149 return self._connected
150
151 @property
152 def stats(self) -> TransportStats:
153 """Statistiques du transport (latence, erreurs, reconnexions)."""
154 return self._stats
155
156 # -- boucle asyncio interne -----------------------------------------------
157 async def _session(self) -> None:
158 import websockets # import gardé — garanti présent après start()
159
160 backoff = BACKOFF_BASE_S
161 while not self._stop.is_set():
162 try:
163 async with websockets.connect(self.url, open_timeout=3.0) as ws:
164 self._connected = True
165 self._stats.reconnects += 1 if backoff < BACKOFF_MAX_S else 0
166 backoff = BACKOFF_BASE_S
167 logger.info("websocket connecté", extra={"url": self.url})
168 await self._pump(ws)
169 except Exception as exc:
170 self._connected = False
171 self._stats.errors += 1
172 if self._stop.is_set():
173 break
174 jitter = backoff * (0.8 + 0.4 * random.random())
175 logger.warning("websocket déconnecté — reconnexion programmée",
176 extra={"url": self.url, "backoff_s": round(jitter, 2),
177 "error": str(exc)})
178 await asyncio.sleep(jitter)
179 backoff = min(BACKOFF_MAX_S, backoff * 2)
180
181 async def _pump(self, ws) -> None:
182 """Pompe bidirectionnelle : messages entrants + pings de latence."""
183 last_ping = time.perf_counter()
184 while not self._stop.is_set():
185 recv_task = asyncio.ensure_future(ws.recv())
186 sleep_task = asyncio.ensure_future(asyncio.sleep(0.05))
187 done, pending = await asyncio.wait(
188 {recv_task, sleep_task}, return_when=asyncio.FIRST_COMPLETED,
189 )
190 for task in pending:
191 task.cancel()
192 if recv_task in done:
193 try:
194 raw = recv_task.result()
195 self._inbox.put_nowait(json.loads(raw))
196 self._stats.received += 1
197 except asyncio.CancelledError:
198 raise
199 except Exception as exc: # payload corrompu — ignoré mais compté
200 self._stats.errors += 1
201 logger.warning("message websocket illisible", extra={"error": str(exc)})
202 now = time.perf_counter()
203 if now - last_ping >= self.ping_interval_s:
204 last_ping = now
205 latency_ms = (await self._measure_latency(ws)) * 1000.0
206 self._stats.record_latency(latency_ms)
207 if latency_ms > self.target_latency_mstarget_latency_ms:
208 logger.warning(
209 "latence websocket au-dessus de la cible",
210 extra={"latency_ms": round(latency_ms, 1),
211 "target_ms": self.target_latency_mstarget_latency_ms},
212 )
213
214 @staticmethod
215 async def _measure_latency(ws) -> float:
216 """Aller-retour d'une trame ping/pong applicative (secondes)."""
217 t0 = time.perf_counter()
218 pong = await ws.ping()
219 await asyncio.wait_for(pong, timeout=2.0)
220 return time.perf_counter() - t0
221
222 def _run_loop(self) -> None:
223 self._loop = asyncio.new_event_loop()
224 asyncio.set_event_loop(self._loop)
225 try:
226 self._loop.run_until_complete(self._session())
227 finally:
228 self._loop.close()
229 self._loop = None

232class FileIPC(Transport):
233 """IPC fichiers JSONL — mode local mono-machine (zéro dépendance réseau).
234
235 Convention de dossier d'échange :
236 out.jsonl : plateforme → plugin (segments à appliquer, acks moteur)
237 in.jsonl : plugin → plateforme (acks, événements d'édition)
238 Chaque transport écrit dans `out` et lit `in` avec un curseur d'octets —
239 deux processus peuvent partager le même dossier sans se marcher dessus.
240 """
241
242 def __init__(self, exchange_dir: Path, suffix: str = "") -> None:
243 self.dir = Path(exchange_dir)
244 self.dir.mkdir(parents=True, exist_ok=True)
245 tag = f".{suffix}" if suffix else ""
246 self.out_path = self.dir / f"out{tag}.jsonl"
247 self.in_path = self.dir / f"in{tag}.jsonl"
248 self.out_path.touch(exist_ok=True)
249 self.in_path.touch(exist_ok=True)
250 self._cursor = 0
251 self._stats = TransportStats()
252 self._lock = threading.Lock()
253 self._closed = False
254
255 def start(self) -> bool:
256 return True # rien à ouvrir — fichiers manipulés à chaque appel
257
258 def stop(self) -> None:
259 self._closed = True
260
261 @property
262 def connected(self) -> bool:
263 return not self._closed
264
265 @property
266 def stats(self) -> TransportStats:
267 """Statistiques du transport fichier (envois, réceptions, erreurs)."""
268 return self._stats
269
270 def send(self, message: dict) -> bool:
271 try:
272 with self._lock:
273 with open(self.out_path, "a", encoding="utf-8") as fh:
274 fh.write(json.dumps(message, ensure_ascii=False, separators=(",", ":")) + "\n")
275 self._stats.sent += 1
276 return True
277 except OSError as exc:
278 self._stats.errors += 1
279 logger.warning("écriture IPC fichier échouée", extra={"error": str(exc)})
280 return False
281
282 def poll(self, timeout_s: float = 0.0) -> Optional[dict]:
283 """Lit le message suivant de in.jsonl (curseur persistant) — None si rien."""
284 deadline = time.monotonic() + max(0.0, timeout_s)
285 while True:
286 message = self._read_next()
287 if message is not None:
288 return message
289 if time.monotonic() >= deadline:
290 return None
291 time.sleep(min(0.02, max(0.001, deadline - time.monotonic())))
292
293 def _read_next(self) -> Optional[dict]:
294 try:
295 size = self.in_path.stat().st_size
296 except OSError:
297 return None
298 if size <= self._cursor:
299 return None
300 with self._lock:
301 with open(self.in_path, "rb") as fh:
302 fh.seek(self._cursor)
303 data = fh.read()
304 # ne consomme que des lignes complètes (écrivain non tronqué)
305 nl = data.rfind(b"\n")
306 if nl < 0:
307 return None
308 chunk = data[:nl]
309 self._cursor += nl + 1
310 for line in chunk.split(b"\n"):
311 line = line.strip()
312 if not line:
313 continue
314 try:
315 self._stats.received += 1
316 return json.loads(line.decode("utf-8"))
317 except (ValueError, UnicodeDecodeError):
318 self._stats.errors += 1
319 logger.warning("ligne IPC illisible — ignorée")
320 return None

323def create_transport(config: Optional[Dict] = None) -> Transport:
324 """Fabrique de transport : 'ws' si URL configurée et websockets présent, sinon FileIPC.
325
326 config : {"mode": "auto"|"ws"|"file", "url": str, "dir": str, "suffix": str}
327 Aucun appel réseau ici — la connexion n'a lieu qu'au `start()`.
328 """
329 config = config or {}
330 mode = config.get("mode", "auto")
331 settings = get_settings("pcb_plugin")
332 url = config.get("url") or DEFAULT_WS_URL.format(port=settings.gateway_port)
333 exchange_dir = Path(config.get("dir", os.environ.get("PCB_IPC_DIR", "/tmp/pcb_ipc")))
334 suffix = config.get("suffix", "")
335
336 if mode in ("ws", "auto"):
337 client = WebSocketClient(url)
338 if mode == "ws":
339 return client
340 # auto : WebSocket seulement si la librairie est réellement importable
341 try:
342 import websockets # noqa: F401 — sondage d'import gardé
343 return client
344 except ImportError:
345 logger.info("websockets absent — repli IPC fichiers (mode local)")
346 return FileIPC(exchange_dir, suffix=suffix)
