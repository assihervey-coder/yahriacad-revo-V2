1"""Découpage de datasheets en chunks métadonnés pour l'index vectoriel.
2
3Stratégie déterministe : découpe par pages (sauts de formulaire ``\\f`` ou
4marqueurs « Page N »), puis par sections (titres Markdown ou lignes numérotées
5type « 3.2 Electrical Characteristics »), puis accumulation de paragraphes
6jusqu'à une taille cible de ~512 tokens (≈ 4 caractères/token), avec un
7chevauchement de 50 tokens entre chunks consécutifs pour ne jamais couper un
8contexte au milieu d'un tableau de caractéristiques.
9"""

45def _hard_split(paragraph: str, max_chars: int, overlap_chars: int) -> List[str]:
46 """Découpe forcée d'un paragraphe trop long, avec recouvrement."""
47 parts: List[str] = []
48 start = 0
49 while start < len(paragraph):
50 end = min(start + max_chars, len(paragraph))
51 # coupe de préférence sur une fin de phrase ou d'espace
52 if end < len(paragraph):
53 dot = paragraph.rfind(". ", start, end)
54 if dot > start + max_chars // 2:
55 end = dot + 1
56 parts.append(paragraph[start:end].strip())
57 if end >= len(paragraph):
58 break
59 start = max(end - overlap_chars, start + 1)
60 return [p for p in parts if p]

63def chunk_datasheet(text: str, source: str,
64 target_tokens: int = DEFAULT_TARGET_TOKENS,
65 overlap_tokens: int = DEFAULT_OVERLAP_TOKENS) -> List[Document]:
66 """Découpe un document (datasheet) en :class:`Document` prêts à indexer.
67
68 ``target_tokens`` ~512 (≈ 2048 caractères) ; ``overlap_tokens`` ~50
69 (≈ 200 caractères) — chaque chunk porte {source, page, section}.
70 """
71 target_chars = max(200, target_tokens * CHARS_PER_TOKEN)
72 overlap_chars = max(0, overlap_tokens * CHARS_PER_TOKEN)
73 docs: List[Document] = []
74
75 # 1) pages : sauts de formulaire ou marqueurs « Page N »
76 pages: List[tuple] = [] # (page_number, text)
77 current_no, buffer = 1, []
78 for line in text.splitlines():
79 marker = _PAGE_MARKER_RE.match(line)
80 if line == "\f" or marker:
81 pages.append((current_no, "\n".join(buffer)))
82 if marker:
83 current_no = int(marker.group(1))
84 else:
85 current_no += 1
86 buffer = []
87 else:
88 buffer.append(line)
89 pages.append((current_no, "\n".join(buffer)))
90
91 # 2) sections puis accumulation de paragraphes
92 for page_no, page_text in pages:
93 section, chunk, chunk_len, section_name = "", [], 0, "général"
94 pending_first: str | None = None
95
96 def flush(first: str | None = None) -> None:
97 nonlocal chunk, chunk_len
98 body = "\n\n".join(chunk).strip()
99 if body:
100 content = body if first is None else f"{first}\n\n{body}"
101 docs.append(Document(content=content,
102 metadata={"source": source, "page": page_no,
103 "section": section_name}))
104 chunk, chunk_len = [], 0
105
106 for paragraph in _split_paragraphs(page_text):
107 first_line = paragraph.splitlines()[0].strip()
108 if _SECTION_RE.match(first_line):
109 flush(pending_first if chunk else None)
110 pending_first, section_name = first_line, first_line[:80]
111 paragraph = "\n".join(paragraph.splitlines()[1:]).strip() or first_line
112 if len(paragraph) > target_chars:
113 flush(pending_first if chunk else None)
114 pending_first = None
115 for piece in _hard_split(paragraph, target_chars, overlap_chars):
116 docs.append(Document(content=piece,
117 metadata={"source": source, "page": page_no,
118 "section": section_name}))
119 continue
120 if chunk_len + len(paragraph) > target_chars and chunk:
121 flush(pending_first)
122 pending_first = None
123 # chevauchement : on ré-amorce avec la fin du chunk précédent
124 tail = "\n\n".join(chunk)[-overlap_chars:]
125 if tail.strip():
126 chunk, chunk_len = [tail], len(tail)
127 chunk.append(paragraph)
128 chunk_len += len(paragraph) + 2
129 flush(pending_first if chunk else None)
130 return docs
