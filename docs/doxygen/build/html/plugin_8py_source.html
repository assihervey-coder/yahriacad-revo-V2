1"""Plugin KiCad de routage en direct — application des segments moteur dans l'éditeur.
2
3Principe (section 6.3, brique DeepPCB) : le routeur IA pousse ses Segments via
4le canal WebSocket/IPC ; le plugin les applique dans la carte ouverte de KiCad
5par l'API de scripting `pcbnew` (création TRACK/VIA) **sans export/import** —
6l'utilisateur voit le routage se dessiner en direct.
7
8`pcbnew` est importé GARDÉ (installation : KiCad ≥ 7 avec bindings Python ;
9TODO déploiement — pointer PYTHONPATH vers /usr/lib/kicad/share/kicad/...).
10Sans pcbnew, le plugin tourne en « modèle fantôme » : les segments sont
11validés puis mémorisés localement, ce qui permet les tests et le mode
12observateur.
13
14Conflits d'édition : verrous par zone réutilisant la sémantique
15`Placement.locked` du modèle commun — un segment qui traverserait l'emprise
16d'un composant verrouillé (surgical_edit humain) est rejeté avec motif.

58class ZoneLockManager:
59 """Verrous par zone — même sémantique que `Placement.locked`.
60
61 Deux niveaux :
62 - verrous statiques : emprises des composants `locked=True` de la carte ;
63 - verrous dynamiques : zones réservées temporairement par un éditeur
64 (`acquire`/`release`, TTL) pour les éditions concurrentes.
65 """
66
67 def __init__(self, board: Optional[Board] = None) -> None:
68 self._board = board
69 self._dynamic: Dict[str, Tuple[float, float, float, float, str, float]] = {}
70 # zone_id -> (x0, y0, x1, y1, owner, expires_ts)
71
72 def attach_board(self, board: Board) -> None:
73 self._board = board

75 def locked_component_at(self, x_mm: float, y_mm: float,
76 margin_mm: float = LOCK_MARGIN_MM) -> Optional[str]:
77 """Référence du composant verrouillé couvrant le point — None si libre."""
78 if self._board is None:
79 return None
80 for ref, placement in self._board.placements.items():
81 if not placement.locked:
82 continue
83 comp = self._board.components.get(ref)
84 if comp is None:
85 continue
86 x0, y0, x1, y1 = comp.bounding_box(placement)
87 if (x0 - margin_mm <= x_mm <= x1 + margin_mm
88 and y0 - margin_mm <= y_mm <= y1 + margin_mm):
89 return ref
90 return None
91
92 def acquire(self, zone_id: str, bbox: Tuple[float, float, float, float],
93 owner: str, ttl_s: float = 30.0) -> bool:
94 """Réserve dynamique d'une zone — False si déjà détenue par un autre."""
95 now = time.time()
96 self._gc(now)
97 for zid, (_x0, _y0, _x1, _y1, held_owner, expires) in self._dynamic.items():
98 if expires > now and held_owner != owner and zid != zone_id:
99 if _overlaps(bbox, (_x0, _y0, _x1, _y1)):
100 return False
101 self._dynamic[zone_id] = (*bbox, owner, now + ttl_s)
102 return True
103
104 def release(self, zone_id: str, owner: str) -> bool:
105 entry = self._dynamic.get(zone_id)
106 if entry and entry[4] == owner:
107 del self._dynamic[zone_id]
108 return True
109 return False
110
111 def dynamic_holder_at(self, x_mm: float, y_mm: float) -> Optional[str]:
112 now = time.time()
113 self._gc(now)
114 for (_x0, _y0, _x1, _y1, owner, expires) in self._dynamic.values():
115 if expires > now and _x0 <= x_mm <= _x1 and _y0 <= y_mm <= _y1:
116 return owner
117 return None
118
119 def _gc(self, now: float) -> None:
120 expired = [zid for zid, entry in self._dynamic.items() if entry[5] <= now]
121 for zid in expired:
122 del self._dynamic[zid]

125def _overlaps(a: Tuple[float, float, float, float],
126 b: Tuple[float, float, float, float]) -> bool:
127 return not (a[2] <= b[0] or b[2] <= a[0] or a[3] <= b[1] or b[3] <= a[1])

169 # ---- application des segments -------------------------------------------
170 def apply_segments(self, segments: List[dict]) -> ApplyReport:
171 """Applique un lot de segments (dicts wire-format ou Segment du modèle).
172
173 Rejets : composant verrouillé traversé (Placement.locked) ou zone
174 dynamiquement réservée. Ack renvoyé à la plateforme.
175 """
176 report = ApplyReport()
177 for index, seg in enumerate(segments):
178 reason = self._rejection_reason(seg)
179 if reason:
180 report.rejected.append({"index": index, "reason": reason,
181 "net": seg.get("net", "")})
182 continue
183 if self._pcbnew_board is not None:
184 kind = self._pcbnew_create(seg)
185 if kind == "via":
186 report.created_vias += 1
187 elif kind == "track":
188 report.created_tracks += 1
189 else:
190 report.rejected.append({"index": index, "reason": "pcbnew_error",
191 "net": seg.get("net", "")})
192 continue
193 else:
194 self._shadow_segments.append(self._to_segment(seg))
195 if seg.get("is_via"):
196 report.created_vias += 1
197 else:
198 report.created_tracks += 1
199 report.applied += 1
200
201 self._events.append(make_event(
202 EventType.NET_ROUTED, self.project_id, "kicad_live_host",
203 self.design_version, applied=report.applied, rejected=len(report.rejected),
204 ))
205 self.transport.send({
206 "type": MSG_SEGMENT_ACK,
207 "project_id": self.project_id,
208 "design_version": self.design_version,
209 **report.to_dict(),
210 })
211 return report

213 def _rejection_reason(self, seg: dict) -> Optional[str]:
214 """Motif de rejet éventuel — verrou statique ou réservation dynamique."""
215 for x, y in ((seg.get("x1_mm", 0.0), seg.get("y1_mm", 0.0)),
216 (seg.get("x2_mm", 0.0), seg.get("y2_mm", 0.0))):
217 locked_ref = self.locks.locked_component_at(x, y)
218 if locked_ref is not None:
219 return f"component_locked:{locked_ref}"
220 holder = self.locks.dynamic_holder_at(x, y)
221 if holder is not None:
222 return f"zone_reserved:{holder}"
223 return None

234 def _pcbnew_create(self, seg: dict) -> str:
235 """Création TRACK/VIA dans la carte KiCad ouverte — renvoie 'track'|'via'|''.
236
237 Import pcbnew fait ici (déjà validé par attach_pcbnew) ; toute erreur
238 d'API retourne '' → rejet compté, jamais d'exception vers le routeur.
239 """
240 try:
241 import pcbnew # type: ignore — import gardé
242 board = self._pcbnew_board
243 if seg.get("is_via"):
244 via = pcbnew.PCB_VIA(board)
245 via.SetPosition(pcbnew.VECTOR2I(pcbnew.FromMM(seg["x1_mm"]),
246 pcbnew.FromMM(seg["y1_mm"])))
247 via.SetDrill(pcbnew.FromMM(0.3))
248 via.SetWidth(pcbnew.FromMM(seg.get("width_mm", 0.4)))
249 board.Add(via)
250 return "via"
251 track = pcbnew.PCB_TRACK(board)
252 track.SetStart(pcbnew.VECTOR2I(pcbnew.FromMM(seg["x1_mm"]),
253 pcbnew.FromMM(seg["y1_mm"])))
254 track.SetEnd(pcbnew.VECTOR2I(pcbnew.FromMM(seg["x2_mm"]),
255 pcbnew.FromMM(seg["y2_mm"])))
256 track.SetWidth(pcbnew.FromMM(seg.get("width_mm", 0.2)))
257 track.SetLayer(pcbnew.F_Cu if int(seg.get("layer", 0)) == 0 else pcbnew.B_Cu)
258 track.SetNetCode(_netcode_for(board, seg.get("net", "")))
259 board.Add(track)
260 pcbnew.Refresh()
261 return "track"
262 except Exception as exc: # pragma: no cover - nécessite KiCad réel
263 logger.warning("création pcbnew échouée", extra={"error": str(exc)})
264 return ""

274 def sync_loop(self, stop: Optional[threading.Event] = None) -> None:
275 """Boucle bloquante : messages entrants → apply_segments → acks.
276
277 Lancée dans un thread daemon par `start()` ; les messages de type
278 MSG_SEGMENT_APPLY déclenchent l'application dans KiCad. Arrêt propre
279 via `stop()`.
280 """
281 stop = stop or self._stop
282 while not stop.is_set():
283 message = self.transport.poll(timeout_s=0.25)
284 if message is None:
285 continue
286 if message.get("type") == MSG_SEGMENT_APPLY:
287 self.apply_segments(message.get("segments", []))
288 elif message.get("type") == "stop":
289 break
290
291 def start(self) -> None:
292 """Démarre la boucle de sync dans un thread daemon + le transport."""
293 if not self.transport.start():
294 logger.warning("transport indisponible — plugin en mode dégradé local")
295 self._stop.clear()
296 self._thread = threading.Thread(target=self.sync_loop, name="kicad_sync", daemon=True)
297 self._thread.start()
298
299 def stop(self) -> None:
300 self._stop.set()
301 if self._thread is not None:
302 self._thread.join(timeout=2.0)
303 self._thread = None
304 self.transport.stop()
