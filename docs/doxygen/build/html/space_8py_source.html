3Discrétisation : pas de 0.5 mm en x/y, rotations limitées à {0, 90, 180, 270}°,
4couche ∈ {0=top, 1=bottom}. Fournit l'échantillonnage (autour d'une ancre ou

54 # ---- utilitaires ------------------------------------------------------------
55 def snap(self, value_mm: float) -> float:
56 """Aligne une coordonnée sur la grille de 0.5 mm."""
57 return round(round(value_mm / self.step_mm) * self.step_mm, 3)

59 def encode(self, action: Action) -> np.ndarray:
60 """Encode une action en vecteur numpy (x, y, rotation/90, couche)."""
61 return np.array([action.x_mm, action.y_mm,
62 action.rotation_deg / 90.0, float(action.layer)], dtype=np.float64)

64 def decode(self, vector: Sequence[float], ref: str) -> Action:
65 """Reconstruit une action depuis un vecteur encodé."""
66 x, y, rot_idx, layer = vector
67 rotation = ROTATIONS[int(round(rot_idx)) % len(ROTATIONS)]
68 return Action(ref=ref, x_mm=self.snap(float(x)), y_mm=self.snap(float(y)),
69 rotation_deg=float(rotation), layer=int(layer) % len(LAYERS))

71 # ---- échantillonnage -------------------------------------------------------------
72 def sample(self, bounds: Tuple[float, float], n: int, ref: str,
73 anchor: Optional[Tuple[float, float]] = None,
74 spread_mm: float = 12.0,
75 rng: Optional[random.Random] = None) -> List[Action]:
76 """Échantillonne n actions : autour de l'ancre (jitter ±spread) si fournie,
77 sinon uniformément sur la carte. Toutes les coordonnées sont snappées."""
78 rng = rng or random.Random(42)
79 width, height = bounds
80 actions: List[Action] = []
81 for _ in range(max(0, n)):
82 if anchor is not None:
83 x = anchor[0] + rng.uniform(-spread_mm, spread_mm)
84 y = anchor[1] + rng.uniform(-spread_mm, spread_mm)
85 else:
86 x, y = rng.uniform(0.0, width), rng.uniform(0.0, height)
87 actions.append(Action(
88 ref=ref, x_mm=self.snap(min(max(x, 0.0), width)),
89 y_mm=self.snap(min(max(y, 0.0), height)),
90 rotation_deg=rng.choice(ROTATIONS), layer=rng.choice(LAYERS)))
91 return actions

93 # ---- légalité --------------------------------------------------------------------------
94 def legal_mask(self, board: Any, actions: Sequence[Action]) -> np.ndarray:
95 """Masque booléen : True = action légalement plaçable sur ce board."""
96 mask = np.zeros(len(actions), dtype=bool)
97 comp = board.components.get(actions[0].ref) if actions else None
98 for i, action in enumerate(actions):
99 if comp is None:
100 continue
101 placement = action.to_placement()
102 x1, y1, x2, y2 = comp.bounding_box(placement)
103 # 1) intégralement sur la carte
104 if x1 < 0.0 or y1 < 0.0 or x2 > board.width_mm or y2 > board.height_mm:
105 continue
106 # 2) hors des zones keepout (marge de clearance incluse)
107 margin = DEFAULT_CLEARANCE_MM
108 hit_keepout = any(
109 not (x2 <= z.x_min_mm - margin or z.x_max_mm + margin <= x1
110 or y2 <= z.y_min_mm - margin or z.y_max_mm + margin <= y1)
111 for z in board.zones if z.kind == "keepout")
112 if hit_keepout:
113 continue
114 # 3) pas de superposition grossière avec les autres composants
115 collision = False
116 for other_ref, other_placement in board.placements.items():
117 if other_ref == action.ref or other_ref not in board.components:
118 continue
119 other = board.components[other_ref]
120 ox1, oy1, ox2, oy2 = other.bounding_box(other_placement)
121 if not (x2 <= ox1 + margin or ox2 + margin <= x1
122 or y2 <= oy1 + margin or oy2 + margin <= y1):
123 collision = True
124 break
125 mask[i] = not collision
126 return mask

128 def anchored_candidates(self, board: Any, ref: str, n: int,
129 anchor: Tuple[float, float],
130 rng: Optional[random.Random] = None) -> List[Action]:
131 """Échantillonne puis filtre : ne garde que les actions légales."""
132 candidates = self.sample((board.width_mm, board.height_mm), n, ref,
133 anchor=anchor, rng=rng)
134 mask = self.legal_mask(board, candidates)
135 return [action for action, ok in zip(candidates, mask) if ok]
