35def _parse_sexprs(tokens: List[str]) -> List[Any]:
36 """Parse les s-expressions en listes imbriquées (atomes = chaînes)."""
37 position = 0
38
39 def parse_one() -> Any:
40 nonlocal position
41 token = tokens[position]
42 if token == "(":
43 position += 1
44 node: List[Any] = []
45 while position < len(tokens) and tokens[position] != ")":
46 node.append(parse_one())
47 position += 1 # saute ')' (tolérant aux déséquilibres)
48 return node
49 position += 1
50 if token.startswith('"'):
51 return token[1:-1].replace('\\"', '"').replace("\\\\", "\\")
52 return token
53
54 trees: List[Any] = []
55 while position < len(tokens):
56 if tokens[position] == ")": # parenthèse fermante orpheline — ignorée
57 position += 1
58 continue
59 trees.append(parse_one())
60 return trees

113def _symbol_to_instance(node: List[Any], seen_refs: Dict[str, int],
114 warnings: List[str]) -> Optional[SymbolInstance]:
115 """Convertit un nœud (symbol (lib_id ...)) en SymbolInstance."""
116 lib_id_nodes = _children(node, "lib_id")
117 if not lib_id_nodes:
118 return None # définition de lib_symbols (sans lib_id) — ignorée
119 lib_id = str(lib_id_nodes[0][1]) if len(lib_id_nodes[0]) > 1 else ""
120 at_nodes = _children(node, "at")
121 atoms = _atoms(at_nodes[0]) if at_nodes else []
122 x = float(atoms[0]) if len(atoms) > 0 else 0.0
123 y = -float(atoms[1]) if len(atoms) > 1 else 0.0 # KiCad : Y vers le bas → inversion
124 rotation = float(atoms[2]) if len(atoms) > 2 else 0.0
125
126 props: Dict[str, str] = {}
127 for prop in _children(node, "property"):
128 values = _atoms(prop)
129 if len(values) >= 2:
130 props[values[0]] = values[1]
131
132 ref = props.get("Reference", "")
133 if not ref:
134 ref = f"X?{len(warnings) + 1}"
135 warnings.append(f"symbole sans référence ({lib_id}) — ref générée « {ref} »")
136 if ref in seen_refs:
137 seen_refs[ref] += 1
138 warnings.append(f"référence dupliquée « {ref} » renommée en « {ref}_{seen_refs[ref]} »")
139 ref = f"{ref}_{seen_refs[ref]}"
140 else:
141 seen_refs[ref] = 0
142
143 prefix = ref[:1].upper()
144 footprint, width, height, price = default_package_for(prefix)
145 if props.get("Footprint"):
146 footprint = props["Footprint"]
147 pins = _estimate_pins(lib_id, _children(node, "pin"))
148 component = Component(
149 ref=ref,
150 mpn=props.get("Value", "") if prefix in {"U", "X", "Q", "M"} else "",
151 value=props.get("Value", ""),
152 footprint=footprint,
153 pins=pins,
154 width_mm=width,
155 height_mm=height,
156 price_usd=price,
157 functional_block=guess_functional_block(prefix, props.get("Value", "")),
158 )
159 return SymbolInstance(component=component, x_mm=x, y_mm=y, rotation_deg=rotation)

162def parse_kicad_sch(text: str) -> KiCadSchematic:
163 """Parse un .kicad_sch minimal : symboles → composants, wires → indices."""
164 result = KiCadSchematic()
165 tokens = _tokenize(text)
166 if "(" not in tokens:
167 raise ValueError("contenu illisible : aucune s-expression trouvée (format .kicad_sch attendu)")
168 trees = _parse_sexprs(tokens)
169 if not trees or _head(trees[0]) != "kicad_sch":
170 result.warnings.append("en-tête (kicad_sch ...) absent — lecture opportuniste tentée")
171
172 seen_refs: Dict[str, int] = {}
173 for tree in trees:
174 for node in _walk(tree):
175 if isinstance(node, list) and _head(node) == "symbol":
176 instance = _symbol_to_instance(node, seen_refs, result.warnings)
177 if instance is not None:
178 result.components.append(instance)
179 if not result.components:
180 raise ValueError(
181 "format non reconnu : aucun (symbol (lib_id ...)) trouvé — "
182 "s'agit-il bien d'un fichier .kicad_sch ?"
183 )
184
185 for tree in trees:
186 for node in _walk(tree):
187 if not (isinstance(node, list) and _head(node) == "wire"):
188 continue
189 pts = _children(node, "pts")
190 if not pts:
191 continue
192 coords = [_children(pts[0], "xy")]
193 endpoints: List[Tuple[float, float]] = []
194 for xy in coords[0]:
195 values = _atoms(xy)
196 if len(values) >= 2:
197 endpoints.append((float(values[0]), -float(values[1])))
198 if len(endpoints) == 2:
199 result.wire_hints.append((endpoints[0], endpoints[1]))
200
201 result.warnings.append(
202 "schéma sans netlist : connectivité non résolue "
203 f"({len(result.wire_hints)} wires conservés comme indices de routage)"
204 )
205 logger.info("schéma KiCad analysé", extra={"components": len(result.components),
206 "wire_hints": len(result.wire_hints)})
207 return result
