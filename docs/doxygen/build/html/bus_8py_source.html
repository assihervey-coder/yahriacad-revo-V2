77class InMemoryConstraintBus(ConstraintBus):
78 """Implémentation en mémoire — utilisée en mono-processus et en tests.
79
80 `publish()` mesure sa propre latence et journalise un avertissement si la
81 cible des 50 ms est dépassée : la diffusion doit rester temps réel même sur
82 des designs denses.
83 """
84
85 def __init__(self, target_latency_ms: float = 50.0) -> None:
86 self._subs: Dict[str, tuple[List[ConstraintKind] | None, Subscriber]] = {}
87 self._latest: Dict[str, ConstraintMessage] = {}
88 self._target_ms_target_ms = target_latency_ms
89 self._revisions: Dict[str, int] = {}
90 self._counter = 0
91
92 def publish(self, message: ConstraintMessage) -> float:
93 start = time.perf_counter()
94 message.revision = self._revisions.get(message.key, 0) + 1
95 self._revisions[message.key] = message.revision
96 self._latest[message.key] = message
97 self._counter += 1
98 for kinds, callback in list(self._subs.values()):
99 if kinds is None or message.kind in kinds:
100 try:
101 callback(message)
102 except Exception: # un abonné défaillant ne bloque jamais le bus
103 logger.exception("abonné constraint_bus en erreur", extra={"key": message.key})
104 elapsed_ms = (time.perf_counter() - start) * 1000.0
105 if elapsed_ms > self._target_ms_target_ms:
106 logger.warning(
107 "latence constraint_bus au-dessus de la cible",
108 extra={"elapsed_ms": round(elapsed_ms, 2), "target_ms": self._target_ms_target_ms, "key": message.key},
109 )
110 return elapsed_ms
111
112 def subscribe(self, kinds: Optional[List[ConstraintKind]], callback: Subscriber) -> str:
113 self._counter += 1
114 token = f"sub-{self._counter}"
115 self._subs[token] = (kinds, callback)
116 return token
117
118 def unsubscribe(self, token: str) -> None:
119 self._subs.pop(token, None)
120
121 def latest(self, kind: Optional[ConstraintKind] = None) -> Dict[str, ConstraintMessage]:
122 if kind is None:
123 return dict(self._latest)
124 return {k: v for k, v in self._latest.items() if v.kind == kind}
125
126 @property
127 def published_count(self) -> int:
128 return self._counter

131class RedisConstraintBus(InMemoryConstraintBus):
132 """Adaptateur multi-processus via Redis pub/sub (déploiement Kubernetes).
133
134 Réutilise la sémantique en mémoire comme cache de dernière valeur, et
135 propage chaque publication sur le canal Redis `constraint_bus`. L'import
136 est gardé : si `redis` est absent, `create_bus()` retombe sur la version
137 en mémoire — le service démarre toujours.
138 """
139
140 CHANNEL = "constraint_bus"
141
142 def __init__(self, redis_url: str, target_latency_ms: float = 50.0) -> None:
143 super().__init__(target_latency_ms)
144 import redis # import gardé — voir docstring de module
145
146 self._redis = redis.Redis.from_url(redis_url, decode_responses=True)
147 self._pubsub = self._redis.pubsub()
148 self._pubsub.subscribe(self.CHANNEL)
149
150 def publish(self, message: ConstraintMessage) -> float:
151 elapsed_ms = super().publish(message)
152 try:
153 self._redis.publish(self.CHANNEL, json.dumps(message.to_json()))
154 except Exception:
155 logger.exception("publication Redis échouée — cache local conservé", extra={"key": message.key})
156 return elapsed_ms
157

159def create_bus(redis_url: str | None = None, target_latency_ms: float = 50.0) -> ConstraintBus:
160 """Fabrique le bus adapté à l'environnement (Redis si disponible, sinon mémoire)."""

162 try:
163 return RedisConstraintBus(redis_url, target_latency_ms) # type: ignore[return-value]
164 except Exception:
165 logger.warning("Redis indisponible — bus en mémoire utilisé", extra={"redis_url": redis_url})
