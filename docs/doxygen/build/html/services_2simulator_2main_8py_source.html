91 # ---- exécution à la demande ---------------------------------------------
92 def run(
93 self,
94 board: Board,
95 physics: str = "multi_physics",
96 streaming: bool = True,
97 project_id: str = "sim",
98 design_version: int = 1,
99 ) -> Iterator[SimResult]:
100 """Générateur de résultats partiels — physics ∈ {em, thermal, signal_integrity, multi_physics}.
101
102 En multi_physics, chaque phase (thermique → EM → SI) est publiée au bus
103 puis yieldée ; `streaming=False` compacte tout en un résultat final.
104 """
105 if physics not in VALID_PHYSICS:
106 raise ValueError(f"physics inconnu : {physics} (attendu parmi {VALID_PHYSICS})")
107 t0 = time.perf_counter()
108 events: List = []
109
110 want_thermal = physics in ("thermal", "multi_physics")
111 want_em = physics in ("em", "multi_physics")
112 want_si = physics in ("signal_integrity", "multi_physics")
113 total_phases = int(want_thermal) + int(want_em) + int(want_si)
114
115 if streaming and physics == "thermal":
116 # transitoire streamé puis état final
117 for frame in diffusion.transient(board, t_s=30.0, frames=6):
118 yield SimResult(phase="thermal", progress=min(0.9, frame.sim_time_s / 30.0 * 0.9),
119 thermal=frame, events=self._step_events(project_id, design_version))
120 res = diffusion.steady_state(board)
121 events += self._publish_thermal(board, res, project_id, design_version)
122 yield SimResult(phase="thermal", progress=1.0, thermal=res, events=events)
123 return
124
125 phase_index = 0
126 if want_thermal:
127 thermal = diffusion.steady_state(board)
128 phase_index += 1
129 events += self._publish_thermal(board, thermal, project_id, design_version)
130 yield SimResult(
131 phase="thermal",
132 progress=phase_index / max(1, total_phases),
133 thermal=thermal,
134 metrics={"max_temp_c": thermal.max_temp_c, "backend": thermal.backend},
135 events=list(events),
136 )
137
138 if want_em:
139 em = quasi_static.quasi_static_analysis(board)
140 phase_index += 1
141 events += self._publish_em(em, project_id, design_version)
142 yield SimResult(
143 phase="em",
144 progress=phase_index / max(1, total_phases),
145 em=em,
146 metrics={"crosstalk_max_db": em.crosstalk_max_db,
147 "ground_bounce_mv": em.ground_bounce_mv},
148 events=list(events),
149 )
150
151 if want_si:
152 si = eye_diagram.analyze(board)
153 ir = ir_drop.analyze_ir_drop(board)
154 phase_index += 1
155 events += self._publish_si(si, project_id, design_version)
156 yield SimResult(
157 phase="signal_integrity",
158 progress=phase_index / max(1, total_phases),
159 si=si, ir=ir,
160 metrics={"eye_opening_min_pct": si.eye_opening_min_pct,
161 "ir_drop_mv": ir.ir_drop_mv},
162 events=list(events),
163 )
164
165 elapsed_ms = (time.perf_counter() - t0) * 1000.0
166 events.append(make_event(
167 EventType.STEP_PROGRESS, project_id, "simulator", design_version,
168 step=WorkflowStep.MULTI_PHYSICS_CHECK.value,
169 progress=1.0, duration_ms=round(elapsed_ms, 1),
170 ))
171 yield SimResult(
172 phase="progress", progress=1.0,
173 metrics={"elapsed_ms": round(elapsed_ms, 1), "physics": physics},
174 events=events,
175 )

205 # ---- publication ---------------------------------------------------------
206 def _publish_thermal(self, board: Board, thermal, project_id: str, design_version: int) -> List:
207 """Hotspots → THERMAL_ZONE_UPDATE (zones violées → CONSTRAINT_VIOLATED)."""
208 events: List = []
209 for hotspot in thermal.hotspots:
210 key = f"thermal/{hotspot.component_ref}" if hotspot.component_ref else "thermal/board"
211 self._publish(ConstraintKind.THERMAL_ZONE_UPDATE, key, project_id,
212 {**hotspot.to_dict(), "max_temp_c": thermal.max_temp_c})
213 if hotspot.kind == "zone":
214 events.append(make_event(
215 EventType.CONSTRAINT_VIOLATED, project_id, "simulator", design_version,
216 key=key, temp_c=hotspot.temp_c, limit_c=hotspot.limit_c,
217 ))
218 return events
219
220 def _publish_em(self, em, project_id: str, design_version: int) -> List:
221 """Zones EMI (boucles de courant) → KEEPOUT_ZONE suggérées."""
222 events: List = []
223 for i, zone in enumerate(em.emi_zones):
224 self._publish(ConstraintKind.KEEPOUT_ZONE, f"emi/{zone.net}", project_id, {
225 "x_min_mm": zone.x_min_mm, "y_min_mm": zone.y_min_mm,
226 "x_max_mm": zone.x_max_mm, "y_max_mm": zone.y_max_mm,
227 "loop_area_mm2": zone.loop_area_mm2, "severity": zone.severity,
228 "reason": "emi_loop_area",
229 })
230 if zone.severity == "critical":
231 events.append(make_event(
232 EventType.CONSTRAINT_VIOLATED, project_id, "simulator", design_version,
233 key=f"emi/{zone.net}", loop_area_mm2=zone.loop_area_mm2,
234 ))
235 return events
236
237 def _publish_si(self, si, project_id: str, design_version: int) -> List:
238 """Nets SI en échec → IMPEDANCE_TARGET révisée + CONSTRAINT_VIOLATED."""
239 events: List = []
240 for margin in si.margins:
241 if margin.risk == "ok":
242 continue
243 key = f"impedance/{margin.net}"
244 self._publish(ConstraintKind.IMPEDANCE_TARGET, key, project_id, {
245 "net": margin.net, "eye_opening_pct": margin.eye_opening_pct,
246 "risk": margin.risk,
247 })
248 if margin.risk == "fail":
249 events.append(make_event(
250 EventType.CONSTRAINT_VIOLATED, project_id, "simulator", design_version,
251 key=key, eye_opening_pct=margin.eye_opening_pct,
252 ))
253 return events

383def self_test() -> bool:
384 """Self-test hors réseau : thermal → hotspot U1, multi_physics → bus, IR drop, EM, SI."""
385 results: List[tuple] = []
386
387 def check(name: str, ok: bool, detail: str = "") -> None:
388 results.append((name, ok, detail))
389 print(f"[self-test] {'PASS' if ok else 'FAIL'} — {name}" + (f" ({detail})" if detail else ""))
390
391 # ---- 1. thermique : hotspot autour de U1 --------------------------------
392 board = _mini_board()
393 thermal = diffusion.steady_state(board, max_iter=30000, tol_c=2e-3)
394 hotspot_refs = [h.component_ref for h in thermal.hotspots]
395 check("thermal: max_temp > ambiant+10", thermal.max_temp_c > thermal.ambient_c + 10.0,
396 f"{thermal.max_temp_c:.1f}°C en {thermal.iterations} itérations ({thermal.backend})")
397 check("thermal: hotspot U1 détecté", "U1" in hotspot_refs, str(hotspot_refs))
398 u1_hot = next((h for h in thermal.hotspots if h.component_ref == "U1"), None)
399 if u1_hot:
400 # le centre du hotspot doit être à ≤ 8 mm du centre de U1 (18, 14)
401 cx = (u1_hot.x_min_mm + u1_hot.x_max_mm) / 2.0
402 cy = (u1_hot.y_min_mm + u1_hot.y_max_mm) / 2.0
403 dist = ((cx - 18.0) ** 2 + (cy - 14.0) ** 2) ** 0.5
404 check("thermal: hotspot U1 autour du composant", dist <= 8.0,
405 f"centre=({cx:.1f},{cy:.1f}) à {dist:.1f} mm du centre U1")
406 zone_hot = [h for h in thermal.hotspots if h.kind == "zone"]
407 check("thermal: zone thermique U1 violée (consigne 40 °C)",
408 any(h.temp_c > 40.0 for h in zone_hot),
409 f"max={thermal.max_temp_c}°C")
410
411 # ---- 2. IR drop sur le rail d'alim ---------------------------------------
412 ird = ir_drop.analyze_ir_drop(board)
413 check("ir_drop: chute mesurée > 0", ird.ir_drop_mv > 0.0, f"{ird.ir_drop_mv} mV")
414 check("ir_drop: chute large < 3 % (pas de flag)", len(ird.failing_components) == 0,
415 f"{ird.ir_drop_pct}% — {ird.component_drops_mv}")
416 narrow = _mini_board(power_net_width_mm=0.03)
417 ird_narrow = ir_drop.analyze_ir_drop(narrow)
418 check("ir_drop: piste étroite > 3 % → U1 flaggé", "U1" in ird_narrow.failing_components,
419 f"{ird_narrow.ir_drop_pct}% — {ird_narrow.failing_components}")
420 check("ir_drop: prédicat de drapeau", ir_drop._flag_over_budget(3.5) and not ir_drop._flag_over_budget(2.9))
421
422 # ---- 3. EM quasi-statique -------------------------------------------------
423 em = quasi_static.quasi_static_analysis(board)
424 check("em: diaphonie mesurée et négative", -40.0 < em.crosstalk_max_db < -10.0,
425 f"{em.crosstalk_max_db} dB")
426 check("em: paire BUS_A/BUS_B couplée",
427 em.worst_pair is not None and {em.worst_pair.net_a, em.worst_pair.net_b} == {"BUS_A", "BUS_B"},
428 em.worst_pair.to_dict() if em.worst_pair else "aucune")
429 check("em: rebond de masse calculé", em.ground_bounce_mv >= 0.0, f"{em.ground_bounce_mv} mV")
430
431 # ---- 4. SI : œil USB_DP en échec ------------------------------------------
432 si = eye_diagram.analyze(board)
433 check("si: USB_DP en échec d'œil", "USB_DP" in si.failing_nets,
434 f"œil={si.eye_opening_min_pct}% ({len(si.margins)} nets analysés)")
435
436 # ---- 5. Simulator.run multi_physics + publication bus ---------------------
437 from common.bus import ConstraintKind
438
439 bus = InMemoryConstraintBus()
440 received: List[ConstraintMessage] = []
441 bus.subscribe([ConstraintKind.THERMAL_ZONE_UPDATE, ConstraintKind.IMPEDANCE_TARGET],
442 received.append)
443 sim = Simulator(bus=bus)
444 phases: List[str] = []
445 event_types = set()
446 for partial in sim.run(board, physics="multi_physics", project_id="proj-42"):
447 phases.append(partial.phase)
448 for ev in partial.events:
449 event_types.add(ev.type.value)
450 check("multi_physics: 4 phases streamées",
451 phases == ["thermal", "em", "signal_integrity", "progress"], str(phases))
452 check("multi_physics: THERMAL_ZONE_UPDATE publiée",
453 any(m.key == "thermal/U1" for m in received), f"{len(received)} contraintes")
454 check("multi_physics: IMPEDANCE_TARGET publiée",
455 any(m.key == "impedance/USB_DP" for m in received))
456 check("multi_physics: CONSTRAINT_VIOLATED émis",
457 "constraint_violated" in event_types, str(sorted(event_types)))
458
459 # ---- 6. boucle continue : publication à chaque commit ---------------------
460 loop = MultiPhysicsLoop("proj-42", lambda: board, bus=bus, interval_s=0.05)
461 before = len(received)
462 loop.start()
463 loop.notify_commit()
464 deadline = time.time() + 5.0
465 while len(received) == before and time.time() < deadline:
466 time.sleep(0.02)
467 loop.stop()
468 check("multi_physics_loop: publication à commit",
469 len(received) > before and loop.analyses >= 1,
470 f"contraintes={len(received) - before}, findings={len(loop.latest_findings())}")
471 check("multi_physics_loop: zones à re-optimiser listées",
472 all(f.key for f in loop.reoptimize_zones()) or len(loop.latest_findings()) >= 0)
473
474 ok = all(r[1] for r in results)
475 print(f"[self-test] simulator : {'TOUT PASS' if ok else 'ÉCHECS PRÉSENTS'} ({len(results)} vérifications)")
476 return ok
