33def run_erc(board: Board, project_id: str = "demo", skidl_source: Optional[str] = None,
34 design_version: int = 1) -> List[Violation]:
35 """Exécute l'ERC complet : règles électriques sur Board + analyse SKiDL.
36
37 Chaque erreur est journalisée et émise comme événement ERC_ERROR — le
38 corrector_agent en dépend pour sa boucle de correction.
39 """
40 errors: List[Violation] = []
41 errors += _check_floating_nets(board)
42 errors += _check_multiple_drivers(board)
43 errors += _check_i2c_pullups(board)
44 errors += _check_supply_decoupling(board)
45 if skidl_source:
46 errors += _check_skidl_static(skidl_source)
47
48 for err in errors:
49 logger.info("erreur ERC détectée", extra={"rule": err.rule_id, "net": err.net})
50 make_event(EventType.ERC_ERROR, project_id, "erc_executor", design_version,
51 rule_id=err.rule_id, net=err.net, message=err.message)
52 return errors

55def _check_floating_nets(board: Board) -> List[Violation]:
56 """Nets avec moins de 2 connexions — fils suspendus."""
57 out: List[Violation] = []
58 for name, net in board.nets.items():
59 if len(net.connections) < 2:
60 out.append(Violation(
61 rule_id="ERC-floating-net", severity="error", net=name,
62 message=f"net {name} flottant ({len(net.connections)} connexion(s))",
63 ))
64 return out

67def _check_multiple_drivers(board: Board) -> List[Violation]:
68 """Deux sorties actives sur un même net — court-circuit potentiel.
69
70 Heuristique : un composant dont la référence commence par U et un pad
71 nommé OUT/TX/SW est considéré comme driver ; deux drivers sur un net =
72 erreur (hors bus explicitement classés CAN/LIN).
73 """
74 out: List[Violation] = []
75 for name, net in board.nets.items():
76 if any(k in (net.net_class or "").upper() for k in ("BUS", "CAN", "LIN", "RS485")):
77 continue
78 drivers = [f"{ref}/{pad}" for ref, pad in net.connections
79 if ref.startswith("U") and any(
80 tag in pad.upper() for tag in ("OUT", "TX", "SW", "DRV"))]
81 if len(drivers) > 1:
82 out.append(Violation(
83 rule_id="ERC-multi-driver", severity="error", net=name,
84 message=f"{len(drivers)} sorties actives sur le net {name} : {', '.join(drivers)}",
85 ))
86 return out

89def _check_i2c_pullups(board: Board) -> List[Violation]:
90 """Bus I2C sans résistance de pull-up sur SDA/SCL."""
91 out: List[Violation] = []
92 for name, net in board.nets.items():
93 if (net.net_class or "").upper() != "I2C" and not _looks_like_i2c(name):
94 continue
95 has_pullup = any(ref.startswith("R") for ref, _ in net.connections)
96 if not has_pullup:
97 out.append(Violation(
98 rule_id="ERC-i2c-pullup", severity="error", net=name,
99 message=f"bus I2C {name} sans résistance de pull-up (motif connu du graphe Neo4j)",
100 ))
101 return out

109def _check_supply_decoupling(board: Board) -> List[Violation]:
110 """Net d'alimentation sans condensateur de découplage associé."""
111 out: List[Violation] = []
112 for name, net in board.nets.items():

116 if not has_cap:
117 out.append(Violation(
118 rule_id="ERC-decoupling", severity="error", net=name,
119 message=f"rail {name} sans condensateur de découplage",
120 ))
121 return out

132def _check_skidl_static(source: str) -> List[Violation]:
133 """Analyse statique du script SKiDL généré (regex — sans exécution)."""
134 out: List[Violation] = []
135 nets = set(_RE_NET.findall(source))
136 parts = _RE_PART.findall(source)
137
138 if not parts:
139 out.append(Violation(
140 rule_id="ERC-skidl-empty", severity="error",
141 message="script SKiDL sans aucun Part() — génération défaillante",
142 ))
143 if "from skidl" not in source:
144 out.append(Violation(
145 rule_id="ERC-skidl-import", severity="error",
146 message="script SKiDL sans import skidl — non exécutable",
147 ))
148
149 # pull-up I2C : présence de SDA/SCL sans résistance déclarée proche
150 i2c_nets = [n for n in nets if any(t in n.upper() for t in ("SDA", "SCL", "I2C"))]
151 resistors = [p for p in parts if "resistor" in p.lower() or "R" in p]
152 if i2c_nets and not resistors:
153 out.append(Violation(
154 rule_id="ERC-skidl-i2c-pullup", severity="error",
155 message=f"nets I2C {i2c_nets} sans Part résistance dans le script SKiDL",
156 ))
157 return out

160def erc_summary(errors: List[Violation]) -> Tuple[int, int, Dict[str, int]]:
161 """(erreurs, avertissements, comptage par règle) — pour les rapports."""
162 counts: Dict[str, int] = {}
163 for e in errors:
164 counts[e.rule_id] = counts.get(e.rule_id, 0) + 1
165 err = sum(1 for e in errors if e.severity == "error")
166 warn = sum(1 for e in errors if e.severity == "warning")
167 return err, warn, counts
