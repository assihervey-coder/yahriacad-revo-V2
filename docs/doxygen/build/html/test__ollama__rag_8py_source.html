31 def _json(self, payload: dict, status: int = 200) -> None:
32 body = json.dumps(payload).encode("utf-8")
33 self.send_response(status)
34 self.send_header("Content-Type", "application/json")
35 self.send_header("Content-Length", str(len(body)))
36 self.end_headers()
37 self.wfile.write(body)

39 def do_GET(self): # health-check /api/tags
40 if self.path.startswith("/api/tags"):
41 self._json({"models": [{"name": "llama3.1:8b"}, {"name": "bge-m3"}]})
42 else:
43 self._json({"error": "not found"}, status=404)
44
45 def do_POST(self): # /api/chat
46 length = int(self.headers.get("Content-Length", 0))
47 payload = json.loads(self.rfile.read(length) or b"{}")
48 if self.path.startswith("/api/chat"):
49 question = ""
50 for message in payload.get("messages", []):
51 if message.get("role") == "user":
52 question = message.get("content", "")
53 self._json({"model": payload.get("model", "?"),
54 "message": {"role": "assistant",
55 "content": f"Réponse LLM local à : {question[:40]} [ds_stm32 p.2]"},
56 "done": True})
57 else:
58 self._json({"error": "not found"}, status=404)
