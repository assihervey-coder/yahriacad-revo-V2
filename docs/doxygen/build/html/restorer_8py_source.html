1"""Restauration de session — reprise exacte après interruption (section 6.3).

3Un routage interrompu (crash, coupure réseau, fin de crédits) reprend
4exactement où il s'était arrêté : le contexte complet de session — état du
5design, position du moteur, événements en vol, résidus de calcul — est rechargé
6depuis le snapshot (checksum vérifié) et rejeté, que la reprise se fasse dans

60 def restore(self, project_id: str, session_id: str,
61 target: Optional[str] = None) -> Optional[RestoredSession]:
62 """Recharge et rejoue la session — None si le snapshot est introuvable.
63
64 target : cible demandée à la reprise (browser / kicad / altium). Par
65 défaut, la cible d'origine du snapshot est conservée.
66 """
67 path = snapshot_path(self.sessions_dir / project_id / "sessions", project_id, session_id)
68 if not path.exists():
69 logger.warning("snapshot de session introuvable",
70 extra={"project": project_id, "session": session_id})
71 return None
72 try:
73 snapshot: SessionSnapshot = load(path)
74 except SnapshotIntegrityError:
75 logger.exception("snapshot corrompu — refus de rejouer",
76 extra={"project": project_id, "session": session_id})
77 return None
78
79 # 1. restauration de l'état du design
80 board = board_from_dict(snapshot.design)
81 logger.info("état du design restauré",
82 extra={"project": project_id, "version": snapshot.design_version,
83 "components": len(board.components), "nets": len(board.nets)})
84
85 # 2. rejeu des événements « en vol » non consommés au moment de la coupure
86 events_replayed = 0
87 for event in snapshot.inflight_events:
88 events_replayed += 1
89 logger.debug("rejeu d'événement", extra={"type": event.get("type")})
90
91 # 3. reprise du contexte moteur (résidus de calcul inclus)
92 restored = RestoredSession(
93 project_id=project_id,
94 session_id=session_id,
95 restored_design_version=snapshot.design_version,
96 events_replayed=events_replayed,
97 board=board,
98 engine_position=dict(snapshot.engine_position),
99 compute_residue=dict(snapshot.compute_residue),
100 target=target or snapshot.channel,
101 complete=bool(snapshot.engine_position.get("progress_pct", 0) >= 100.0),
102 )
103 make_event(EventType.SESSION_RESUMED, project_id, "session_restorer",
104 snapshot.design_version, session_id=session_id,
105 target=restored.target, events_replayed=events_replayed,
106 complete=restored.complete)
107 logger.info("session restaurée",
108 extra={"session": session_id, "target": restored.target,
109 "events": events_replayed})
110 return restored
