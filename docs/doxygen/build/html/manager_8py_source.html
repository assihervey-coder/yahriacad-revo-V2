1"""State manager — état global par projet, versions, verrous, diffusion.
2
3Responsabilités (section 04) :
4- `get_design()` → DesignState (Board + version + journal en mémoire) ;
5- `commit()` → version++ + snapshot + écriture du journal JSONL ;
6- verrous de concurrence par zone (le RL agent et un surgical_edit humain ne
7 peuvent JAMAIS écrire la même zone — voir locks.py) ;
8- diffusion des deltas aux abonnés (callbacks WebSocket), latence cible < 100 ms.
9"""

47 # ---- cycle de vie -------------------------------------------------------
48 def get_or_create(self, project_id: str, request_text: str = "") -> DesignState:
49 with self._mutex_mutex_mutex:
50 state = self._states.get(project_id)
51 if state is None:
52 state = DesignState(project_id=project_id, board=Board())
53 state.snapshot(author="state_manager", note="projet créé")
54 self._states[project_id] = state
55 self._snapshots[project_id] = {state.version: copy.deepcopy(state.board)}
56 self.journal.append(project_id, {
57 "version": state.version, "author": "state_manager",
58 "note": "projet créé", "request_text": request_text[:200],
59 })
60 logger.info("design initialisé", extra={"project_id": project_id})
61 return state

63 def get_design(self, project_id: str) -> DesignState:
64 """Design existant uniquement — KeyError si projet inconnu."""
65 state = self._states.get(project_id)
66 if state is None:
67 raise KeyError(f"projet inconnu : {project_id}")
68 return state

70 # ---- mutations versionnées ----------------------------------------------
71 def mutate(self, project_id: str, author: str, note: str,
72 mutator: Callable[[Board], Any]) -> dict[str, Any]:
73 """Applique `mutator(board)` puis commit + diffusion du delta."""
74 state = self.get_or_create(project_id)
75 before = copy.deepcopy(state.board)
76 result = mutator(state.board)
77 return self.commit_delta(project_id, before, author, note, result=result)

79 def commit_delta(self, project_id: str, before: Board, author: str, note: str,
80 result: Any = None) -> dict[str, Any]:
81 """Commit après mutation extérieure : delta → journal + événements."""
82 state = self.get_or_create(project_id)
83 delta = compute_board_delta(before, state.board)
84 entry = self.commit(project_id, note, author, delta_summary=delta["summary"])
85 for event in events_from_delta(delta, project_id, state.version, emitter=author):
86 self.publish_event(event)
87 return {"entry": entry, "delta": delta, "result": result}

89 def commit(self, project_id: str, note: str, author: str = "system",
90 delta_summary: str = "") -> dict[str, Any]:
91 """Version++ + snapshot + journal — point de passage obligé du pipeline."""
92 with self._mutex_mutex_mutex:
93 state = self.get_design(project_id)
94 state.version += 1
95 entry = state.snapshot(author, note)
96 entry["delta_summary"] = delta_summary
97 self._snapshots.setdefault(project_id, {})[state.version] = copy.deepcopy(state.board)
98 self.journal.append(project_id, entry)
99 logger.info("commit design", extra={"project_id": project_id,
100 "version": state.version, "author": author})
101 return entry

103 def journal_decision(self, project_id: str, agent: str, note: str,
104 justifications: list[str] | None = None) -> None:
105 """Journalise une décision d'agent SANS incrémenter la version."""
106 self.journal.append(project_id, {
107 "kind": "agent_decision", "agent": agent, "note": note,
108 "justifications": justifications or [], "version": self._peek_version(project_id),
109 })

111 def _peek_version(self, project_id: str) -> int:
112 state = self._states.get(project_id)
113 return state.version if state else 0

115 def rollback(self, project_id: str, to_version: int) -> dict[str, Any]:
116 """Restaure le Board depuis les snapshots mémoire + trace le rollback."""
117 with self._mutex_mutex_mutex:
118 state = self.get_design(project_id)
119 snapshot = self._snapshots.get(project_id, {}).get(to_version)
120 if snapshot is None:
121 raise KeyError(f"version {to_version} introuvable pour {project_id}")
122 state.board = copy.deepcopy(snapshot)
123 state.version = to_version
124 state.snapshot(author="state_manager", note=f"rollback vers v{to_version}")
125 entry = {"version": state.version, "author": "state_manager",
126 "note": f"rollback vers v{to_version}", "kind": "rollback"}
127 self.journal.append(project_id, entry)
128 self.publish_event(make_event(EventType.ROLLBACK_PERFORMED, project_id,
129 "state_manager", design_version=to_version,
130 to_version=to_version))
131 return entry

145 # ---- diffusion WebSocket ----------------------------------------------------
146 def publish_event(self, event: Event) -> None:
147 """Diffuse l'événement aux abonnés + buffer de reprise (delta)."""
148 self._buffer.append(event)
149 start = time.perf_counter()
150 for _token, (scope, callback) in list(self._subscribers.items()):
151 if scope and scope != event.project_id:
152 continue
153 try:
154 callback(event)
155 except Exception: # un abonné défaillant ne bloque jamais la diffusion
156 logger.exception("abonné WebSocket en erreur")
157 elapsed_ms = (time.perf_counter() - start) * 1000.0
158 if elapsed_ms > self.ws_target_latency_msws_target_latency_ms:
159 logger.warning("latence diffusion au-dessus de la cible",
160 extra={"elapsed_ms": round(elapsed_ms, 2), "seq": event.seq})
161
162 def subscribe(self, callback: Subscriber,
163 project_id: str | None = None) -> str:
164 """Abonne un canal WebSocket (project_id None = tous les projets)."""
165 self._counter += 1
166 token = f"ws-{self._counter}"
167 self._subscribers[token] = (project_id, callback)
168 return token
169
170 def unsubscribe(self, token: str) -> None:
171 self._subscribers.pop(token, None)

173 def events_since(self, since_seq: int, project_id: str | None = None) -> list[Event]:
174 """Reprise de session : événements bufferisés après `since_seq`."""
175 return events_since(self._buffer, since_seq, project_id)
