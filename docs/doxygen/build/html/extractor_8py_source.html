34def _deaccent(text: str) -> str:
35 """Minuscules sans accents + espaces normalisés pour les regex de détection."""
36 flat = "".join(char for char in unicodedata.normalize("NFD", (text or "").lower())
37 if unicodedata.category(char) != "Mn")
38 return " ".join(flat.split())

45def _infer_group(lowered: str, match_start: int) -> str:
46 """Rattache une règle de longueur appariée à une classe (fenêtre avant le match)."""
47 window = lowered[max(0, match_start - 80):match_start]
48 for hint in _CLASS_HINTS:
49 if hint in window:
50 return hint.upper()
51 return "default"

54def extract_constraints(spec_text: str, project_id: str = "") -> List[ConstraintMessage]:
55 """Extrait les contraintes d'un cahier des charges NL ou d'une netlist."""
56 lowered = _deaccent(spec_text)
57 messages: List[ConstraintMessage] = []
58
59 # 1) classes de signaux → impédance cible + skew + largeur minimale
60 specs: List[SignalClassSpec] = match_signal_classes(lowered)
61 for spec in specs:
62 messages.append(_msg(
63 ConstraintKind.IMPEDANCE_TARGET, f"impedance/{spec.net_class}",
64 {
65 "impedance_target_ohm": spec.impedance_target_ohm,
66 "differential": spec.differential,
67 "skew_tolerance_mm": spec.skew_tolerance_mm,
68 "min_trace_width_mm": spec.min_trace_width_mm,
69 "class_name": spec.name,
70 },
71 project_id,
72 ))
73
74 # 2) longueurs appariées : « appariées ±5mm » / « matched within 5 mm »
75 for match in re.finditer(
76 r"(?:appari\w*|matched(?:\s+within)?|length\s+match\w*)[^.0-9]{0,30}?±?\s*"
77 r"(\d+(?:[.,]\d+)?)\s*mm", lowered):
78 tolerance = float(match.group(1).replace(",", "."))
79 group = _infer_group(lowered, match.start())
80 messages.append(_msg(ConstraintKind.LENGTH_MATCH_RULE, f"length_match/{group}",
81 {"tolerance_mm": tolerance, "group": group}, project_id))
82
83 # 3) zones interdites : « keepout de 3 mm près de l'antenne »
84 for match in re.finditer(r"keepout[^.;]{0,90}", lowered):
85 span = match.group(0)
86 reason = next((word for word in ("antenne", "antenna", "rf", "cara", "thermique") if word in span),
87 "zone")
88 radius = re.search(r"(\d+(?:[.,]\d+)?)\s*mm", span)
89 clearance = float(radius.group(1).replace(",", ".")) if radius else 5.0
90 messages.append(_msg(ConstraintKind.KEEPOUT_ZONE, f"keepout/{reason}",
91 {"reason": reason, "clearance_mm": clearance, "hint": span.strip()},
92 project_id))
93
94 # 4) budget thermique : « température max 80°C » / « max 85 »
95 for match in re.finditer(
96 r"(?:max(?:imum)?|limite)[^.0-9]{0,25}?(\d+(?:[.,]\d+)?)\s*(?:degres\s*)?(?:°\s*)?c\b"
97 r"|temperature[^.0-9]{0,20}?(\d+(?:[.,]\d+)?)", lowered):
98 raw = match.group(1) or match.group(2)
99 if raw:
100 messages.append(_msg(ConstraintKind.THERMAL_ZONE_UPDATE, "thermal/global",
101 {"max_temp_c": float(raw.replace(",", "."))}, project_id))
102
103 # 5) budgets de courant : « jusqu'à 3A », « 2 A max »
104 for match in re.finditer(r"(\d+(?:[.,]\d+)?)\s*a\b(?![a-z])", lowered):
105 amps = float(match.group(1).replace(",", "."))
106 if 0.05 <= amps <= 50.0:
107 key = f"current/rail_{len(messages)}"
108 messages.append(_msg(ConstraintKind.CURRENT_BUDGET, key,
109 {"max_current_a": amps}, project_id))
110
111 # 6) clearances explicites : « isolement de 0,3 mm »
112 for match in re.finditer(r"(?:clearance|isolement|distance)[^.0-9]{0,25}?(\d+(?:[.,]\d+)?)\s*mm",
113 lowered):
114 value = float(match.group(1).replace(",", "."))
115 if 0.05 <= value <= 5.0:
116 messages.append(_msg(ConstraintKind.CLEARANCE, "clearance/explicit",
117 {"min_clearance_mm": value}, project_id))
118
119 logger.info("contraintes extraites", extra={"count": len(messages),
120 "classes": [s.net_class for s in specs]})
121 return messages
