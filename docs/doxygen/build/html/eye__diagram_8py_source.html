1"""Ouverture d'œil des buses rapides — heuristique calibrée sur la perte d'insertion.
2
3Modèle (par net rapide) :
4 - perte d'insertion diélectrique : IL ≈ 0.5 dB/in/GHz × longueur(in) × f(GHz) ;
5 - pénalité par via : 0.6·√f dB (discontinuité d'impédance) ;
6 - fermeture d'œil verticale ≈ amplitude résiduelle 10^(−IL_tot/20), amputée
7 d'un facteur ISI dépendant de la longueur, puis de la diaphonie fournie ;
8 - dépassement (overshoot) : réflexions ∝ nombre de vias + longueur.
9
10Nets considérés « rapides » : classe connue du tableau ci-dessous ou cible
11d'impédance déclarée. Les nets à risque sont listés (risk = ok | watch | fail).
12"""

89def _net_frequency(net) -> Optional[float]:

97def analyze_net(net, crosstalk_db: float = -40.0) -> Optional[NetMargin]:
98 """Heuristique d'ouverture d'œil pour un net — None si le net n'est pas analysé."""
99 f_ghz = _net_frequency(net)
100 if f_ghz is None or not net.routed_segments:
101 return None
102
103 length_mm = net.routed_length_mm
104 length_in = length_mm / MM_PER_INCH
105 vias = net.via_count
106
107 il_diell_db = LOSS_DB_PER_IN_GHZ * length_in * f_ghz
108 il_via_db = VIA_LOSS_DB * math.sqrt(max(f_ghz, 1e-3)) * vias
109 il_total = il_diell_db + il_via_db
110
111 amplitude = 10.0 ** (-il_total / 20.0) # 0..1
112 isi_factor = min(0.5, 0.02 * f_ghz * length_in) # interférence inter-symboles
113 xt_factor = min(0.25, 0.5 * (10.0 ** (crosstalk_db / 20.0)))
114 eye_pct = max(0.0, 100.0 * amplitude * (1.0 - isi_factor) * (1.0 - xt_factor))
115
116 # overshoot : réflexion des discontinuités (vias) + ringing proportionnel à la longueur
117 overshoot_mv = min(OVERSHOOT_LIMIT_MV, 60.0 + 40.0 * vias + 8.0 * length_in * f_ghz)
118
119 if eye_pct < FAIL_EYE_PCT:
120 risk = "fail"
121 elif eye_pct < WATCH_EYE_PCT:
122 risk = "watch"
123 else:
124 risk = "ok"
125
126 return NetMargin(
127 net=net.name, net_class=net.net_class,
128 length_mm=round(length_mm, 1), f_ghz=f_ghz, vias=vias,
129 insertion_loss_db=round(il_total, 2),
130 eye_opening_pct=round(eye_pct, 1),
131 overshoot_mv=round(overshoot_mv, 1), risk=risk,
132 )

135def analyze(board, crosstalk_db: float = -40.0) -> SIResult:
136 """Analyse SI complète de la carte → SIResult (nets à risque listés)."""
137 t0 = time.perf_counter()
138 margins: List[NetMargin] = []
139 for net in board.nets.values():
140 margin = analyze_net(net, crosstalk_db=crosstalk_db)
141 if margin is not None:
142 margins.append(margin)
143
144 failing = [m.net for m in margins if m.risk == "fail"]
145 eye_min = min((m.eye_opening_pct for m in margins), default=100.0)
146 overshoot_max = max((m.overshoot_mv for m in margins), default=0.0)
147 if failing:
148 logger.warning(
149 "nets SI en échec d'ouverture d'œil", extra={"nets": failing},
150 )
151 return SIResult(
152 eye_opening_min_pct=round(eye_min, 1),
153 overshoot_max_mv=round(overshoot_max, 1),
154 failing_nets=failing,
155 margins=sorted(margins, key=lambda m: m.eye_opening_pct),
156 analyzed_nets=len(margins),
157 elapsed_s=round(time.perf_counter() - t0, 4),
158 )
