52def check(board: Board, project_id: str = "demo", factory_profile: str = "internal",
53 include_erc: bool = False, skidl_source: Optional[str] = None,
54 design_version: int = 1) -> CheckReply:
55 """Vérification complète : design + manufacturing (+ ERC optionnel).
56
57 Le score composite (0..100) est la métrique officielle consommée par le
58 keeper_logic de l'optimiseur nocturne et par le benchmark vs Quilter.
59 """
60 get_settings("drc_dfm_engine")
61 profile: FactoryProfile = get_profile(factory_profile)
62 report: DrcReport = check_design_rules(board)
63 factory_violations = check_factory(board, profile)
64 report.violations.extend(factory_violations)
65
66 erc_errors: List[Violation] = []
67 if include_erc:
68 erc_errors = run_erc(board, project_id, skidl_source, design_version)
69
70 has_error = any(v.severity == "error" for v in report.violations) or \
71 any(v.severity == "error" for v in erc_errors)
72 score = report.score if not erc_errors else max(
73 0.0, report.score - 6.0 * len([e for e in erc_errors if e.severity == "error"]))
74
75 reply = CheckReply(pass_=not has_error, score=round(score, 2),
76 violations=report.violations, erc_errors=erc_errors,
77 factory_profile=profile.name)
78 logger.info("vérification DRC/DFM terminée",
79 extra={"project": project_id, "pass": reply.pass_, "score": reply.score,
80 "violations": len(reply.violations), "erc": len(reply.erc_errors)})
81 return reply
