15def _fmt(value: float) -> str:
16 if isinstance(value, int) or float(value).is_integer():

21def _fmt_gain(gain_pct: float, better: str) -> str:
22 """Gain lisible : +X % (nous) / -X % (Quilter) / =."""
23 if better == "tie":
24 return "="
25 sign = "+" if gain_pct > 0 else ""
26 arrow = "notre moteur" if better == "ours" else "Quilter"
27 return f"{sign}{gain_pct:.2f} % ({arrow})"

30def render_design_table(run: DesignRun) -> str:
31 """Tableau comparatif d'un design — section du rapport."""
32 comp: ComparisonResult = run.comparison
33 lines = [
34 f"### {run.design.id} — {run.design.name}",
35 "",
36 run.design.description,
37 "",
38 f"Replays déterministes : {'oui' if run.replays_identical else 'NON (échec Circuitron)'}"
39 f" — vias flashés à l'export : {run.gerber_summary.get('vias_flashed', '?')}",
40 "",
41 "| Métrique | Notre moteur | Quilter (stub) | Gain |",
42 "|---|---|---|---|",
43 ]
44 for row in comp.rows:
45 lines.append(f"| {row.label} | {_fmt(row.ours)} | {_fmt(row.quilter)} "
46 f"| {_fmt_gain(row.gain_pct, row.better)} |")
47 lines += [
48 f"| **Gain moyen** | | | **{comp.mean_gain_pct:+.2f} %** |",
49 "",
50 ]
51 if run.gate_violations:
52 lines.append(f"**Gate : ÉCHEC** — régression vs baseline :")
53 for v in run.gate_violations:
54 lines.append(f"- `{v.metric}` : {v.detail}")
55 else:
56 lines.append("**Gate : OK** — métriques conformes à la baseline certifiée.")
57 lines.append("")
58 # Métriques « cible de revue » du corpus (documentaires).
59 if run.design.expected_metrics:
60 lines.append("Cibles de revue du corpus : "
61 + ", ".join(f"{k}={v}" for k, v in sorted(run.design.expected_metrics.items()))
62 + ".")
63 lines.append("")
64 return "\n".join(lines)
