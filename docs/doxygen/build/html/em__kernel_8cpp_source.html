2// em_kernel.cpp — couplage capacitif quasi-statique entre nets (C++17, OpenMP).
3//
4// Miroir du calcul numpy de em_sim/quasi_static.py : pour chaque paire de
5// segments parallèles appartenant à deux nets différents, le couplage
6// capacitif vaut C ≈ ε₀·ε_r·A/d avec
7// A = longueur de recouvrement × largeur moyenne (m²),
8// d = distance perpendiculaire entre axes (m).
9// La matrice de couplage (pF, symétrique, diagonale nulle) est agrégée par
10// paire de nets — consommée par le calcul de diaphonie côté Python.
11//
12// API C exportée (ctypes — signatures dans README.md) :
13// int em_coupling_matrix(const double* segs, // 6 doubles par segment :
14// // x1,y1,x2,y2,width,net_id
15// int n_seg,
16// int n_nets,
17// double eps_r,
18// double* c_matrix_pf); // n_nets × n_nets (pF), row-major
19// int em_parallel_pairs(const double* segs, int n_seg,
20// int n_nets, double eps_r, double max_dist_mm,
21// double* pairs_out, int max_pairs); // option diagnostic
22// Codes de retour : 0 = succès, -1 = arguments invalides.
23//
24// Politique d'échelle : segments plats dans un seul tableau (6 doubles chacun)
25// — zéro allocation côté kernel, OpenMP sur les paires de nets (parallel for).

35// ε₀ en pF/mm : 8.8541878128e-12 F/m = 8.8541878128e-6 pF/mm... on garde SI
36// puis conversion : C[F] = ε₀[SI]·A[m²]/d[m] ; ×1e12 → pF.
37static const double EPS_0_SI = 8.8541878128e-12;

39typedef struct {

45static inline double seg_length(const EmSegment& s)
46{
47 const double dx = s.x2 - s.x1;
48 const double dy = s.y2 - s.y1;
49 return std::sqrt(dx * dx + dy * dy);
50}

52// Métriques d'une paire : renvoie true si quasi-parallèles ; calcule la
53// distance perpendiculaire (mm) et le recouvrement axial (mm).
54static inline bool parallel_metrics(const EmSegment& a, const EmSegment& b,
55 double& dist_mm, double& overlap_mm)
56{
57 const double la = seg_length(a);
58 const double lb = seg_length(b);
59 if (la < 1e-9 || lb < 1e-9)
60 return false;
61
62 const double ux = (a.x2 - a.x1) / la;
63 const double uy = (a.y2 - a.y1) / la;
64 const double vx = (b.x2 - b.x1) / lb;
65 const double vy = (b.y2 - b.y1) / lb;
66
67 // quasi-parallélisme : |sin(θ)| ≤ 0.05 (~±2.9°)
68 const double sin_theta = std::fabs(ux * vy - uy * vx);
69 if (sin_theta > 0.05)
70 return false;
71
72 // distance perpendiculaire du milieu de b à la droite portée par a
73 const double ma_x = (a.x1 + a.x2) * 0.5;
74 const double ma_y = (a.y1 + a.y2) * 0.5;
75 const double mb_x = (b.x1 + b.x2) * 0.5;
76 const double mb_y = (b.y1 + b.y2) * 0.5;
77 dist_mm = std::fabs(ux * (mb_y - ma_y) - uy * (mb_x - ma_x));
78
79 // recouvrement des projections de b sur l'axe de a
80 const double ta = (b.x1 - ma_x) * ux + (b.y1 - ma_y) * uy;
81 const double tb = (b.x2 - ma_x) * ux + (b.y2 - ma_y) * uy;
82 const double lo = std::max(-la * 0.5, std::min(ta, tb));
83 const double hi = std::min(la * 0.5, std::max(ta, tb));
84 overlap_mm = std::max(0.0, hi - lo);
85 return true;
86}

88// ---------------------------------------------------------------------------
89// Matrice de couplage capacitif (pF) agrégée par paire de nets.
90// ---------------------------------------------------------------------------
91extern "C" int em_coupling_matrix(const double* segs_flat, int n_seg,
92 int n_nets, double eps_r,
93 double* c_matrix_pf)
94{
95 if (segs_flat == nullptr || c_matrix_pf == nullptr || n_seg < 0 || n_nets <= 0)
96 return -1;
97
98 // désérialisation plate → structure (sans allocation GPU, cache-friendly)
99 EmSegment* segs = (EmSegment*)std::malloc((size_t)n_seg * sizeof(EmSegment));
100 if (segs == nullptr && n_seg > 0)
101 return -2;
102 for (int i = 0; i < n_seg; ++i) {
103 const double* s = segs_flat + 6 * (size_t)i;
104 segs[i].x1 = s[0]; segs[i].y1 = s[1];
105 segs[i].x2 = s[2]; segs[i].y2 = s[3];
106 segs[i].width = s[4];
107 segs[i].net = (int)s[5];
108 }
109
110 for (int i = 0; i < n_nets * n_nets; ++i)
111 c_matrix_pf[i] = 0.0;
112
113#ifdef PCB_EM_HAVE_OPENMP
114#pragma omp parallel for schedule(dynamic, 1)
115#endif
116 for (int net_a = 0; net_a < n_nets; ++net_a) {
117 // accumulation locale par net_a pour éviter les atomics
118 double* row = c_matrix_pf + (size_t)net_a * (size_t)n_nets;
119 for (int i = 0; i < n_seg; ++i) {
120 if (segs[i].net != net_a)
121 continue;
122 for (int j = 0; j < n_seg; ++j) {
123 if (segs[j].net <= net_a) // paire ordonnée net_a < net_b uniquement
124 continue;
125 double dist, overlap;
126 if (!parallel_metrics(segs[i], segs[j], dist, overlap))
127 continue;
128 if (dist <= 1e-6 || overlap <= 0.5) // collées ou recouvrement négligeable
129 continue;
130 // C = ε₀·ε_r·A/d (SI) → pF ; A en m² = (overlap·w_moy)·1e-6
131 const double area_m2 = (overlap * (segs[i].width + segs[j].width) * 0.5) * 1e-6;
132 const double d_m = dist * 1e-3;
133 const double c_pf = EPS_0_SI * eps_r * area_m2 / d_m * 1e12;
134#ifdef PCB_EM_HAVE_OPENMP
135#pragma omp atomic
136#endif
137 row[segs[j].net] += c_pf;
138 }
139 }
140 }
141
142 // symétrisation (la boucle ordonnée ne remplit que le triangle supérieur)
143 for (int a = 0; a < n_nets; ++a) {
144 for (int b = a + 1; b < n_nets; ++b) {
145 const double v = c_matrix_pf[(size_t)a * n_nets + b];
146 c_matrix_pf[(size_t)b * n_nets + a] = v;
147 }
148 }
149
150 std::free(segs);
151 return 0;
152}

154// ---------------------------------------------------------------------------
155// Diagnostic : liste plate des paires de segments couplées (net_a, net_b, pF).
156// Renvoie le nombre de paires écrites (≤ max_pairs).
157// ---------------------------------------------------------------------------
158extern "C" int em_parallel_pairs(const double* segs_flat, int n_seg,
159 int n_nets, double eps_r, double max_dist_mm,
160 double* pairs_out, int max_pairs)
161{
162 if (segs_flat == nullptr || pairs_out == nullptr || n_seg < 0 || n_nets <= 0)
163 return -1;
164
165 int written = 0;
166 for (int i = 0; i < n_seg && written < max_pairs; ++i) {
167 for (int j = i + 1; j < n_seg && written < max_pairs; ++j) {
168 EmSegment a, b;
169 const double* sa = segs_flat + 6 * (size_t)i;
170 const double* sb = segs_flat + 6 * (size_t)j;
171 a.x1 = sa[0]; a.y1 = sa[1]; a.x2 = sa[2]; a.y2 = sa[3];
172 a.width = sa[4]; a.net = (int)sa[5];
173 b.x1 = sb[0]; b.y1 = sb[1]; b.x2 = sb[2]; b.y2 = sb[3];
174 b.width = sb[4]; b.net = (int)sb[5];
175 if (a.net == b.net)
176 continue;
177 double dist, overlap;
178 if (!parallel_metrics(a, b, dist, overlap))
179 continue;
180 if (dist > max_dist_mm || overlap <= 0.5)
181 continue;
182 const double area_m2 = (overlap * (a.width + b.width) * 0.5) * 1e-6;
183 const double c_pf = EPS_0_SI * eps_r * area_m2 / (dist * 1e-3) * 1e12;
184 double* row = pairs_out + 3 * (size_t)written;
185 row[0] = (double)a.net;
186 row[1] = (double)b.net;
187 row[2] = c_pf;
188 ++written;
189 }
190 }
191 return written;
192}
