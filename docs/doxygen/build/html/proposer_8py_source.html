1"""Proposer de modifications candidates — 100 % heuristiques déterministes.
2
3Familles de propositions, motivées par le score courant + graphe d'intention :
4 * rotate — rotations 90/180/270 des composants les plus encombrants ;
5 * swap — permutation de deux composants de blocs différents quand
6 elle réduit la HPWL totale ;
7 * spread_hot — écartement des zones chaudes (composants de puissance) ;
8 * regroup — rapprochement des membres d'un même bloc fonctionnel ;
9 * compact_net — rapprochement du centroïde des partenaires de ses nets.
10Le hook ``llm_refine`` (callable optionnel) peut réordonner/filtrer les
11propositions ; en son absence l'ordre heuristique est conservé — aucune
12proposition « non structurée » n'entre jamais dans la boucle.
13"""

36def apply_proposal(board: Any, proposal: Proposal) -> bool:
37 """Applique une proposition au board (move/rotate/swap) — False si refusée."""
38 payload = proposal.json
39 action = payload.get("action")
40 try:
41 if action == "rotate":
42 board.move(payload["ref"], board.placements[payload["ref"]].x_mm,
43 board.placements[payload["ref"]].y_mm,
44 rotation_deg=float(payload["rotation_deg"]))
45 elif action == "move":
46 board.move(payload["ref"], float(payload["x"]), float(payload["y"]))
47 elif action == "swap":
48 ref_a, ref_b = payload["ref_a"], payload["ref_b"]
49 pa, pb = board.placements[ref_a], board.placements[ref_b]
50 board.move(ref_b, pa.x_mm, pa.y_mm)
51 board.move(ref_a, pb.x_mm, pb.y_mm)
52 else:
53 return False
54 except (KeyError, PermissionError):
55 return False
56 return True

59def hpwl_mm(board: Any) -> float:
60 """Longueur totale estimée des nets (Half-Perimeter Wire Length)."""
61 total = 0.0
62 for net in board.nets.values():
63 points = [(board.placements[r].x_mm, board.placements[r].y_mm)
64 for r, _p in net.connections if r in board.placements]
65 if len(points) >= 2:
66 xs = [p[0] for p in points]
67 ys = [p[1] for p in points]
68 total += (max(xs) - min(xs)) + (max(ys) - min(ys))
69 return total

78 # ---- API -----------------------------------------------------------------
79 def propose(self, board: Any, intent_graph: Any = None, n: int = 6) -> List[Proposal]:
80 """Jusqu'à n propositions candidates, motivées et sérialisables.
81
82 Les familles sont entrelacées (rotate, swap, spread_hot, regroup,
83 compact_net) pour qu'un lot de 3 ne soit jamais monoculture — la
84 boucle nocturne explore ainsi toutes les stratégies à chaque passe.
85 """
86 buckets: List[List[Proposal]] = [
87 self._rotate_big(board),
88 self._swap_for_hpwl(board),
89 self._spread_hot(board),
90 self._regroup_blocks(board, intent_graph),
91 self._compact_net(board),
92 ]
93 proposals: List[Proposal] = []
94 index = 0
95 while len(proposals) < max(0, n) and any(buckets):
96 bucket = buckets[index % len(buckets)]
97 index += 1
98 if bucket:
99 proposals.append(bucket.pop(0))
100 if self.llm_refine is not None:
101 try:
102 proposals = self.llm_refine(proposals) or proposals
103 except Exception:
104 pass # le hook ne doit jamais casser la boucle
105 return proposals

108 def _rotate_big(self, board: Any) -> List[Proposal]:
109 """Rotation des composants non carrés à grande empreinte — peut débloquer
110 le routage (une rotation d'empreinte carrée est un no-op exclu)."""
111 out: List[Proposal] = []
112 ranked = sorted(
113 (c for c in board.components.values() if c.width_mm != c.height_mm),
114 key=lambda c: -(c.width_mm * c.height_mm * (1.0 + c.power_w)))
115 for comp in ranked[:2]:
116 for rotation in (90.0, 180.0):
117 out.append(Proposal(
118 json={"action": "rotate", "ref": comp.ref, "rotation_deg": rotation},
119 description=(f"Rotation {rotation:.0f}° de {comp.ref} "
120 f"({comp.width_mm}x{comp.height_mm} mm) pour libérer le corridor"),
121 targets=[comp.ref], kind="rotate"))
122 return out

124 def _swap_for_hpwl(self, board: Any) -> List[Proposal]:
125 """Permutation des deux composants qui réduit le plus la HPWL totale."""
126 best: Optional[Tuple[float, str, str]] = None
127 refs = [r for r in board.placements if r in board.components]
128 baseline = hpwl_mm(board)
129 for i, ref_a in enumerate(refs):
130 for ref_b in refs[i + 1:]:
131 pa, pb = board.placements[ref_a], board.placements[ref_b]
132 if pa.locked or pb.locked:
133 continue
134 board.placements[ref_a], board.placements[ref_b] = pb, pa # swap éphémère
135 delta = hpwl_mm(board) - baseline
136 board.placements[ref_a], board.placements[ref_b] = pa, pb
137 if delta < -1e-6 and (best is None or delta < best[0]):
138 best = (delta, ref_a, ref_b)
139 if best is None:
140 return []
141 delta, ref_a, ref_b = best
142 return [Proposal(
143 json={"action": "swap", "ref_a": ref_a, "ref_b": ref_b},
144 description=(f"Permutation {ref_a} ↔ {ref_b} : HPWL réduite de {-delta:.1f} mm"),
145 targets=[ref_a, ref_b], kind="swap")]

147 def _spread_hot(self, board: Any) -> List[Proposal]:
148 """Écarte le composant le plus chaud de son voisin puissant le plus proche."""
149 hot = [c for c in board.components.values() if c.power_w > 0.2]
150 if len(hot) < 2:
151 return []
152 source = max(hot, key=lambda c: c.power_w)
153 p0 = board.placements[source.ref]
154 nearest, nearest_d = None, float("inf")
155 for comp in hot:
156 if comp.ref == source.ref:
157 continue
158 p = board.placements[comp.ref]
159 d = math.hypot(p.x_mm - p0.x_mm, p.y_mm - p0.y_mm)
160 if d < nearest_d:
161 nearest, nearest_d = comp, d
162 if nearest is None or nearest_d < 1e-3:
163 return []
164 q = board.placements[nearest.ref]
165 dx, dy = p0.x_mm - q.x_mm, p0.y_mm - q.y_mm
166 norm = math.hypot(dx, dy) or 1.0
167 step = 5.0
168 new_x = min(max(p0.x_mm + dx / norm * step, 0.0), board.width_mm)
169 new_y = min(max(p0.y_mm + dy / norm * step, 0.0), board.height_mm)
170 return [Proposal(
171 json={"action": "move", "ref": source.ref, "x": round(new_x, 2), "y": round(new_y, 2)},
172 description=(f"Écartement de {source.ref} ({source.power_w:.1f} W) de "
173 f"{nearest.ref} : espacement thermique +{step:.0f} mm"),
174 targets=[source.ref], kind="spread_hot")]

176 def _regroup_blocks(self, board: Any, intent_graph: Any) -> List[Proposal]:
177 """Rapproche un composant du centroïde des membres de son bloc fonctionnel."""
178 blocks: Dict[str, List[str]] = {}
179 for comp in board.components.values():
180 if comp.functional_block:
181 blocks.setdefault(comp.functional_block, []).append(comp.ref)
182 out: List[Proposal] = []
183 for block, refs in blocks.items():
184 if len(refs) < 2:
185 continue
186 cx = sum(board.placements[r].x_mm for r in refs) / len(refs)
187 cy = sum(board.placements[r].y_mm for r in refs) / len(refs)
188 farthest = max(refs, key=lambda r: math.hypot(
189 board.placements[r].x_mm - cx, board.placements[r].y_mm - cy))
190 p = board.placements[farthest]
191 if math.hypot(p.x_mm - cx, p.y_mm - cy) < 3.0:
192 continue
193 out.append(Proposal(
194 json={"action": "move", "ref": farthest,
195 "x": round(cx, 2), "y": round(cy, 2)},
196 description=(f"Regroupement du bloc « {block} » : {farthest} vers le "
197 f"centroïde du bloc ({cx:.1f}, {cy:.1f})"),
198 targets=[farthest], kind="regroup"))
199 return out[:2]

201 def _compact_net(self, board: Any) -> List[Proposal]:
202 """Rapproche un composant du centroïde des partenaires de ses nets."""
203 partners: Dict[str, List[str]] = {}
204 for net in board.nets.values():
205 refs = [r for r, _p in net.connections if r in board.placements]
206 for ref in refs:
207 partners.setdefault(ref, []).extend(r for r in refs if r != ref)
208 out: List[Proposal] = []
209 for ref, others in partners.items():
210 if not others or board.placements[ref].locked:
211 continue
212 cx = sum(board.placements[r].x_mm for r in others) / len(others)
213 cy = sum(board.placements[r].y_mm for r in others) / len(others)
214 p = board.placements[ref]
215 if math.hypot(p.x_mm - cx, p.y_mm - cy) < 2.0:
216 continue
217 out.append(Proposal(
218 json={"action": "move", "ref": ref, "x": round(cx, 2), "y": round(cy, 2)},
219 description=(f"Rapprochement de {ref} vers le centroïde de ses "
220 f"{len(set(others))} partenaires de nets"),
221 targets=[ref], kind="compact_net"))
222 return out[:2]
