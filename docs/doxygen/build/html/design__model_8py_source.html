61 def center(self) -> Tuple[float, float]:
62 if self.padspads:
63 xs = [p.x_mm for p in self.padspads]
64 ys = [p.y_mm for p in self.padspads]
65 return (sum(xs) / len(xs), sum(ys) / len(ys))
66 return (0.0, 0.0)

68 def bounding_box(self, placement: "Placement") -> Tuple[float, float, float, float]:
69 """(x_min, y_min, x_max, y_max) en mm une fois le placement appliqué."""
70 cx, cy = placement.x_mm, placement.y_mm
71 hw, hh = self.width_mm / 2.0, self.height_mm / 2.0
72 if abs(math.sin(math.radians(placement.rotation_deg))) > 0.5:
73 hw, hh = hh, hw
74 return (cx - hw, cy - hh, cx + hw, cy + hh)

164 def move(self, ref: str, x_mm: float, y_mm: float, rotation_deg: Optional[float] = None) -> Placement:
165 """Déplacement atomique — rejeté si le composant est verrouillé."""
166 if ref not in self.placements:
167 raise KeyError(f"composant inconnu : {ref}")
168 current = self.placements[ref]
169 if current.locked:
170 raise PermissionError(f"{ref} est verrouillé (surgical_edit humain)")
171 new_rot = current.rotation_deg if rotation_deg is None else rotation_deg
172 new = replace(current, x_mm=x_mm, y_mm=y_mm, rotation_deg=new_rot)
173 self.placements[ref] = new
174 return new

176 def bounding_box_overlap(self, ref_a: str, ref_b: str) -> bool:
177 pa, pb = self.placements[ref_a], self.placements[ref_b]
178 ca, cb = self.components[ref_a], self.components[ref_b]
179 ax1, ay1, ax2, ay2 = ca.bounding_box(pa)
180 bx1, by1, bx2, by2 = cb.bounding_box(pb)
181 return not (ax2 <= bx1 or bx2 <= ax1 or ay2 <= by1 or by2 <= ay1)

193 def drc_score(self) -> float:
194 """Score composite 0..100 provisoire : complétude + pénalités de vias.
195
196 Le score définitif est calculé par le drc_dfm_engine ; celui-ci sert
197 d'objectif interne au rl_agent et au keeper_logic (AutoPCB).
198 """
199 total = len(self.nets) or 1
200 routed_ratio = (total - len(self.unrouted_nets())) / total
201 via_penalty = min(0.2, self.via_count() / 2000.0)
202 return round(100.0 * routed_ratio * (1.0 - via_penalty), 2)
