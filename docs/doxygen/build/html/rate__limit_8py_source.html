49 def check(self, token_id: str, tier: str = "free", action: str = "default",
50 cost: int = 1) -> QuotaDecision:
51 """Compte `cost` unités pour l'action — lève RateLimitExceeded si quota."""
52 limit, window_s = self._quota_for(tier, action)
53 now = time.monotonic()
54 with self._mutex_mutex:
55 hits = self._hits.setdefault((token_id, action), deque())
56 self._evict(hits, now, window_s)
57 if len(hits) + cost > limit:
58 retry_after = max(0.0, window_s - (now - hits[0])) if hits else window_s
59 raise RateLimitExceeded(
60 f"quota « {action} » atteint pour le palier {tier} "
61 f"({limit}/{_label(window_s)}) — réessayez dans {int(retry_after) + 1}s",
62 retry_after_s=retry_after)
63 for _ in range(max(1, cost)):
64 hits.append(now)
65 return QuotaDecision(action=action, limit=limit, remaining=max(0, limit - len(hits)),
66 window_s=window_s)
