1"""Adaptateurs inter-services — point d'extension unique de l'orchestrateur.
2
3Tout appel vers un service externe (parser, ai_engine, rl_agent, exporteur…)
4passe par `call_adapter()` — trois voies, essayées dans cet ordre :
5
61. gRPC réel (mode distribué OPTIONNEL) : stubs générés depuis
7 `proto/<service>/v1/*.proto` (`make proto`), endpoints décrits dans
8 `super_agent/resource_allocator.py` (`SERVICE_ENDPOINTS`, ports
9 50051-50058) — activé par ORCH_DISTRIBUTED=1, voir
10 `orchestrator/grpc_transport.py` ; la valeur suit alors le contrat proto v1 ;
112. import in-process (voie PAR DÉFAUT) : import direct du package cible —
12 aucun appel réseau ;
133. fallback de simulation locale : si l'import échoue, un résultat plausible
14 + un avertissement — le pipeline reste exécutable hors ligne.
15
16En mode distribué, un RPC en échec bascule automatiquement sur la voie
17in-process puis sur le fallback (`AdapterResult.source` trace la voie
18empruntée : « grpc », « service » ou « fallback »).
19"""

112class AdapterResult:
113 """Résultat normalisé d'un appel d'adaptateur (source traçable)."""
114
115 key: str
116 value: Any
117 source: str = "service" # "service" (module importé) | "fallback" (simulation)
118 elapsed_s: float = 0.0
119 warnings: list[str] = field(default_factory=list)
120
121
122def call_adapter(key: str, *args: Any, **kwargs: Any) -> AdapterResult:
123 """Appelle le service `key` — gRPC (optionnel), in-process, puis fallback.
124
125 L'interface est inchangée pour tous les appelants depuis la version
126 initiale ; seule la VOIE empruntée change selon l'environnement
127 (`AdapterResult.source` : « grpc » | « service » | « fallback »).
128 """
129 contract = ADAPTER_CONTRACTS[key]
130 start = time.perf_counter()
131 # 1) mode distribué (optionnel) — RPC proto v1 via les stubs générés
132 grpc_value = grpc_transport.try_grpc(key, args, kwargs)
133 if grpc_value is not None:
134 return AdapterResult(key=key, value=grpc_value, source="grpc",
135 elapsed_s=round(time.perf_counter() - start, 4))
136 # 2) voie in-process (défaut hors ligne) puis 3) fallback de simulation
137 try:
138 module = importlib.import_module(contract["module"])
139 fn: Callable[..., Any] | None = getattr(module, contract["function"], None)
140 if fn is None:
141 raise ImportError(f"fonction absente : {contract['function']}")
142 value = fn(*args, **kwargs)
143 return AdapterResult(key=key, value=value, source="service",
144 elapsed_s=round(time.perf_counter() - start, 4))
145 except Exception as exc: # service écrit en parallèle — fallback résilient
146 warning = f"adaptateur {key} indisponible ({type(exc).__name__}) — simulation locale"
147 logger.warning(warning, extra={"target_module": contract["module"]})
148 fallback = _FALLBACKS[key]
149 value = fallback(*args, **kwargs) if callable(fallback) else fallback
150 return AdapterResult(key=key, value=value, source="fallback",
151 elapsed_s=round(time.perf_counter() - start, 4),
152 warnings=[warning])

172def _fb_rl_place_route(board) -> dict: # noqa: ANN001 — Board (common.design_model)
173 """Placement en grille + routage en L (2 segments par net), hors ligne."""
174 moved: list[str] = []
175 x, y = 8.0, 8.0
176 for ref, comp in board.components.items():
177 place = board.placements.get(ref)
178 try:
179 if place is None or not place.locked:
180 board.move(ref, x, y)
181 moved.append(ref)
182 except (KeyError, PermissionError):
183 continue
184 x += comp.width_mm + 4.0
185 if x > board.width_mm - 10.0:
186 x, y = 8.0, y + 12.0
187 for name, net in board.nets.items():
188 if net.is_routed or len(net.connections) < 2:
189 continue
190 from common.design_model import Segment
191
192 (ref_a, _pad_a), (ref_b, _pad_b) = net.connections[0], net.connections[-1]
193 pa = board.placements.get(ref_a)
194 pb = board.placements.get(ref_b)
195 if pa is None or pb is None:
196 continue
197 net.routed_segments = [
198 Segment(net=name, x1_mm=pa.x_mm + 2, y1_mm=pa.y_mm,
199 x2_mm=pa.x_mm + 2, y2_mm=pb.y_mm, layer=0),
200 Segment(net=name, x1_mm=pa.x_mm + 2, y1_mm=pb.y_mm,
201 x2_mm=pb.x_mm - 2, y2_mm=pb.y_mm, layer=0),
202 ]
203 return {"score": board.drc_score(), "moved": moved}

206def _fb_scoped_edit(board, targets: list[str], transformation: dict) -> dict: # noqa: ANN001
207 """Applique move/rotate aux cibles — borné à la bounding box d'impact."""
208 move = transformation.get("move", {})
209 rotation = float(transformation.get("rotate_deg", 0.0) or 0.0)
210 changed: list[str] = []
211 for ref in targets:
212 place = board.placements.get(ref)
213 if place is None:
214 continue
215 try:
216 board.move(ref, place.x_mm + float(move.get("dx_mm", 0.0)),
217 place.y_mm + float(move.get("dy_mm", 0.0)),
218 rotation if rotation else None)
219 changed.append(ref)
220 except (KeyError, PermissionError):
221 continue
222 return {"changed_refs": changed, "new_bbox": None}
