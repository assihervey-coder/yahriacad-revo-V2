89class FastAPIStub:
90 """Gateway sans serveur : `handle(method, path, payload)` → (status, body).
91
92 Toujours disponible (même si fastapi est installé) : sert au self-test

102 """Enregistre les routes d'un router — tuples du stub OU APIRoute fastapi."""

120 def handle(self, method: str, path: str, payload: dict[str, Any] | None = None,
121 headers: dict[str, str] | None = None) -> tuple[int, Any]:
122 """Dispatch synchrone : → (status_code, body) — pour les tests."""
123 context = {"method": method.upper(), "path": path,
124 "payload": payload or {}, "headers": headers or {}}
125 try:
126 for guard in self._guards:
127 guard(context)
128 method_up = method.upper()

155def _invoke(fn: Callable[..., Any], path_params: dict[str, str],
156 payload: dict[str, Any], headers: dict[str, str]) -> Any:
157 """Appelle un handler avec ses paramètres (path / body pydantic / query).
158
159 `eval_str=True` résout les annotations paresseuses (`from __future__ import
160 annotations`) — sans cela les annotations arrivent en str et le body
161 pydantic ne serait jamais injecté.
162 """
163 try:
164 signature = inspect.signature(fn, eval_str=True)
165 except (ValueError, NameError):
166 signature = inspect.signature(fn)
167 kwargs: dict[str, Any] = {}
168 for name, parameter in signature.parameters.items():
169 annotation = parameter.annotation
170 if name in path_params:
171 kwargs[name] = _coerce(annotation, path_params[name])
172 elif isinstance(annotation, type) and issubclass(annotation, BaseModel):
173 kwargs[name] = annotation(**(payload or {}))
174 elif name in (payload or {}):
175 kwargs[name] = _coerce(annotation, payload[name])
176 elif name in headers:
177 kwargs[name] = headers[name]
178 return fn(**kwargs)
