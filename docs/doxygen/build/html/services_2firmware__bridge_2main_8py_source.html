35def export_pins_for(board: Board, project_id: str = "demo",
36 target: str = "raw", design_version: int = 1) -> List[PinAssignment]:
37 """Exporte les broches et détecte les changements de brochage.
38
39 Si l'assignation diffère du dernier export (fichier pin_state.json), un
40 événement FIRMWARE_REGENERATED est émis — le firmware_preview rafraîchit
41 ses diffs et le développeur embarqué ne peut plus écrire un firmware
42 désynchronisé du matériel.
43 """
44 get_settings("firmware_bridge")
45 assignments = export_pins(board)
46 signature = [a.to_dict() for a in assignments]
47
48 state_path = STATE_DIR / project_id / "firmware" / "pin_state.json"
49 previous = _load_state(state_path)
50 if previous is not None and previous != signature:
51 changed = _diff_count(previous, signature)
52 make_event(EventType.FIRMWARE_REGENERATED, project_id, "firmware_bridge",
53 design_version, changed=changed, target=target)
54 logger.info("brochage modifié — régénération déclenchée",
55 extra={"project": project_id, "changed": changed})
56 _save_state(state_path, signature)
57 return assignments

89def _diff_count(previous: List[Dict], current: List[Dict]) -> int:
90 """Nombre d'assignations modifiées — charge utile de FIRMWARE_REGENERATED."""
91 prev_index = {(p.get("component_ref"), p.get("pin")): p for p in previous}
92 changed = 0
93 for cur in current:
94 key = (cur.get("component_ref"), cur.get("pin"))
95 if prev_index.get(key) != cur:
96 changed += 1
97 return changed + abs(len(previous) - len(current))

100def _self_test() -> int:
101 """Self-test : carte STM32 + LoRa → broches → headers Zephyr/Arduino → diff."""
102 from common.design_model import Board, Component, Net, Pad, Placement
103
104 board = Board(width_mm=40.0, height_mm=30.0)
105 defs = {
106 "U1": (8.0, 15.0, 8.0, 8.0, {"1": "VDD", "2": "GND", "3": "SWDIO", "4": "SWCLK",
107 "5": "SPI2_SCK", "6": "SPI2_MOSI", "7": "I2C1_SDA",
108 "8": "I2C1_SCL"}),
109 "U2": (30.0, 15.0, 6.0, 4.0, {"1": "SPI2_SCK", "2": "SPI2_MOSI", "3": "ANT"}),
110 "R1": (20.0, 8.0, 2.0, 1.0, {"1": "I2C1_SDA", "2": "VDD"}),
111 "C1": (20.0, 22.0, 2.0, 1.0, {"1": "VDD", "2": "GND"}),
112 }
113 for ref, (x, y, w, h, pads_def) in defs.items():
114 comp = Component(ref=ref, width_mm=w, height_mm=h)
115 comp.pads = [Pad(name=name, x_mm=x + i * 1.0, y_mm=y, net=net)
116 for i, (name, net) in enumerate(pads_def.items())]
117 board.add_component(comp, Placement(ref=ref, x_mm=x, y_mm=y))
118 for name, net_class in {"VDD": "POWER", "GND": "POWER", "SPI2_SCK": "SPI",
119 "SPI2_MOSI": "SPI", "I2C1_SDA": "I2C", "I2C1_SCL": "I2C",
120 "SWDIO": "DEBUG", "SWCLK": "DEBUG", "ANT": "RF"}.items():
121 board.nets[name] = Net(name=name, net_class=net_class)
122
123 assignments = export_pins_for(board, "selftest", design_version=3)
124 print(f"[firmware] broches exportées : {len(assignments)}")
125 for a in assignments[:5]:
126 print(f" {a.component_ref}/{a.pin:<4s} → {a.function:<12s} (net {a.net})")
127
128 headers = generate_headers_for(board, "selftest", target="both", design_version=3)
129 for header in headers:
130 print(f"[firmware] {header.filename} : {len(header.c_source.splitlines())} lignes")
131 zephyr_header = headers[0].c_source
132 ok_zephyr = "#ifndef PCB_AI_DESIGNER_PIN_MAP_ZEPHYR_H" in zephyr_header
133 ok_pins = len(assignments) >= 14
134
135 # simulation d'un changement de brochage → FIRMWARE_REGENERATED
136 board.components["U1"].pads[2].net = "SPI3_SCK" # remap pin 3
137 assignments2 = export_pins_for(board, "selftest", design_version=4)
138 changed = sum(1 for a in assignments2 if a.pin.endswith("3") and a.component_ref == "U1")
139 print(f"[firmware] changement détecté sur {changed} broche(s) → FIRMWARE_REGENERATED")
140
141 ok = ok_zephyr and ok_pins and changed >= 1
142 print("[firmware] SELF-TEST", "OK" if ok else "FAIL")
143 return 0 if ok else 1
