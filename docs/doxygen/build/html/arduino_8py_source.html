63def _symbol(a: PinAssignment) -> str:
64 base = (a.net or f"PIN_{a.pin}").upper()
65 cleaned = "".join(c if c.isalnum() else "_" for c in base)
66 return "_".join(part for part in cleaned.split("_") if part)
