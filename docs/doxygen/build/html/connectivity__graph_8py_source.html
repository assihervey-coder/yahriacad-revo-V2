39 def __init__(self, board: Board) -> None:
40 self.board = board
41 self._pads: Dict[str, List[PadNode]] = {}
42 for ref, comp in board.components.items():
43 placement = board.placements.get(ref)
44 layer = placement.layer if placement else 0
45 for pad in comp.pads:
46 net = self._resolve_net(pad.net, ref, pad.name)
47 if net is None:
48 continue
49 self._pads.setdefault(net, []).append(
50 PadNode(ref, pad.name, pad.x_mm, pad.y_mm, layer))

52 def _resolve_net(self, declared: Optional[str], ref: str, pad: str) -> Optional[str]:
53 """Net d'une pastille : champ direct, sinon résolution via net.connections."""
54 if declared:
55 return declared
56 for name, net in self.board.nets.items():
57 if (ref, pad) in net.connections:
58 return name
59 return None

61 def net_endpoints(self, net_name: str) -> List[Tuple[str, str, Tuple[float, float]]]:
62 """Points d'ancrage d'un net : [(ref, pad, (x_mm, y_mm)), ...]."""
63 return [(node.ref, node.pad, (node.x_mm, node.y_mm))
64 for node in self._pads.get(net_name, [])]

66 def nodes(self, net_name: str) -> List[PadNode]:
67 """Nœuds complets (avec couche) attachés à un net."""

70 def mst_pairs(self, net_name: str) -> List[Tuple[PadNode, PadNode]]:
71 """Arbre couvrant minimal (Prim) — paires de pads à relier par le router."""
72 nodes = self._pads.get(net_name, [])
73 if len(nodes) < 2:
74 return []
75 in_tree = [nodes[0]]
76 remaining = nodes[1:]
77 pairs: List[Tuple[PadNode, PadNode]] = []
78 while remaining:
79 best = min(
80 ((a, b) for a in in_tree for b in remaining),
81 key=lambda pair: (pair[0].x_mm - pair[1].x_mm) ** 2
82 + (pair[0].y_mm - pair[1].y_mm) ** 2,
83 )
84 pairs.append(best)
85 in_tree.append(best[1])
86 remaining.remove(best[1])
87 return pairs

89 def unrouted_net_names(self) -> List[str]:
90 """Nets à router : non routés et possédant au moins deux pastilles."""
91 return [name for name, net in self.board.nets.items()
92 if not net.is_routed and len(self._pads.get(name, [])) >= 2]
93
94 def layer_allowlist(self, net_name: str) -> List[int]:
95 """Couches cuivre autorisées pour un net (intersection avec la pile)."""
96 net = self.board.nets.get(net_name)
97 allowed = net.layer_allowlist if net and net.layer_allowlist else list(range(len(self.board.layers)))
98 valid = [layer for layer in allowed if 0 <= layer < len(self.board.layers)]
99 return valid or [0]
