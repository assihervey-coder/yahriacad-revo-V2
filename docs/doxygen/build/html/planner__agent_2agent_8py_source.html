43def _detect_blocks(request_text: str) -> list[str]:
44 """Détecte les blocs fonctionnels par mots-clés (mcu toujours présent)."""
45 text = (request_text or "").lower()
46 detected = [block for block, keywords in _BLOCK_KEYWORDS.items()
47 if any(k in text for k in keywords)]

64 def execute(self, context: AgentContext) -> AgentResult:
65 blocks = _detect_blocks(context.request_text)
66 layer_budget = _layer_budget(blocks)
67 # Contraintes via adaptateur (kind/key/value — cf. common.bus)
68 extracted = call_adapter("constraint_extractor", context.request_text)
69 constraints = list(extracted.value or [])
70 justifications = [
71 f"blocs détectés : {', '.join(blocks)} (mots-clés de la requête)",
72 f"budget de couches = {layer_budget} "
73 f"({'RF/actionneurs ⇒ empilement 4 couches' if layer_budget >= 4 else 'design simple'})",
74 ]
75 if extracted.source == "fallback":
76 justifications.append("contraintes par défaut (constraint_extractor indisponible)")
77
78 plan = PlanDocument(
79 project_id=context.project_id,
80 functional_blocks=[{"name": b, "refs": [], "priority": i}
81 for i, b in enumerate(blocks)],
82 placement_zones=self._zones(blocks),
83 layer_budget=layer_budget,
84 constraints=constraints,
85 sub_objectives=([{"objective": f"placer le bloc {name}", "block": name,
86 "verify": "zones sans chevauchement (validator)"}
87 for name in blocks] + [
88 {"objective": "router 100 % des nets",
89 "verify": "board.unrouted_nets() vide"},
90 {"objective": "respecter le budget de couches",
91 "verify": f"layers <= {layer_budget}"},
92 ]),
93 )
94 context.plan_document = plan
95 self.log.info("plan produit", extra={"blocks": blocks, "layers": layer_budget})
96 return AgentResult(agent=self.name, status="done",
97 outputs={"plan": plan.to_json()},
98 justifications=justifications)

100 def _zones(self, blocks: list[str]) -> list[dict[str, Any]]:
101 """Zones de placement — mcu au centre, RF en bord (antenne), power au bord."""
102 zones = []
103 for i, block in enumerate(blocks):
104 y = 8.0 + i * 20.0
105 zones.append({
106 "block": block,
107 "bbox": [8.0, y, 60.0, y + 18.0],
108 "note": "RF en bord de carte (antenne dégagée)" if block == "rf" else "",
109 })
110 return zones
