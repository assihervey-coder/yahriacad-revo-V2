64def load_mapping(path: Optional[Path] = None) -> Dict[str, Dict[str, Any]]:
65 """Charge altium_mapping.yaml s'il existe (import yaml gardé), sinon défauts.
66
67 Un fichier partiellement renseigné est fusionné section par section avec
68 les défauts — jamais de remplacement total silencieux.
69 """
70 mapping = {section: dict(values) for section, values in DEFAULT_MAPPING.items()}
71 candidates = [path] if path else [
72 Path.cwd() / MAPPING_FILE,
73 Path(__file__).resolve().parents[1] / MAPPING_FILE,
74 ]
75 for candidate in candidates:
76 if candidate is None or not candidate.exists():
77 continue
78 try:
79 import yaml # import gardé — dépendance optionnelle
80
81 with open(candidate, "r", encoding="utf-8") as fh:
82 override = yaml.safe_load(fh) or {}
83 except ImportError:
84 logger.warning(
85 "pyyaml absent — altium_mapping.yaml ignoré, mapping par défaut utilisé",
86 extra={"path": str(candidate)},
87 )
88 continue
89 except Exception as exc:
90 logger.warning("lecture altium_mapping.yaml échouée", extra={"error": str(exc)})
91 continue
92 for section, values in override.items():
93 if isinstance(values, dict) and isinstance(mapping.get(section), dict):
94 mapping[section].update(values)
95 logger.info("mapping Altium chargé", extra={"path": str(candidate)})
96 break
97 return mapping

124def rule_to_constraint(rule: dict, mapping: Optional[dict] = None) -> Optional[dict]:
125 """Règle Altium → contrainte interne {kind, key, value} — None si ignorée.
126
127 `rule` : {"kind": "WidthRule", "scope": "+3V3", "value": 0.5, "unit": "mm"}
128 Les unités non supportées sont converties quand c'est trivial (mil → mm),
129 sinon journalisées et la règle est ignorée (écart de sémantique assumé).
130 """
131 mapping = mapping or DEFAULT_MAPPING
132 kind = mapping["rule_kinds"].get(rule.get("kind", ""))
133 if kind is None:
134 logger.info("règle Altium sans équivalent — ignorée", extra={"rule": rule.get("kind")})
135 return None
136 if kind == "functional_zone":
137 return {
138 "kind": kind, "key": f"zone/{rule.get('scope', 'room')}",
139 "value": {"zone_kind": mapping["zone_kinds"].get(rule.get("kind"), "functional"),
140 "name": rule.get("scope", "room")},
141 }
142
143 value = rule.get("value")
144 if value is None:
145 value = mapping["defaults"].get(_default_key_for(kind))
146 if value is None:
147 logger.warning("règle Altium sans valeur ni défaut — ignorée", extra={"rule": rule})
148 return None
149 unit = (rule.get("unit") or "mm").lower()
150 if unit == "mil":
151 value = float(value) * 0.0254
152 elif unit not in ("mm", "ohm", ""):
153 logger.warning("unité non supportée — règle ignorée",
154 extra={"unit": unit, "rule": rule.get("kind")})
155 return None
156
157 key = f"{_key_prefix(kind)}/{rule.get('scope', 'board')}"
158 return {"kind": kind, "key": key, "value": {"source": "altium", "value": value}}

161def arbitrate(existing: Optional[dict], incoming: Optional[dict], context: str) -> Optional[dict]:
162 """Arbitrage entre deux contraintes concurrentes — « au plus strict », journalisé.
163
164 Pour IMPEDANCE_TARGET / CLEARANCE / CURRENT_BUDGET (impédance et clearance :
165 garder la contrainte la plus exigeante) ; pour les autres, la plus récente
166 gagne. Retourne la contrainte retenue (peut être `incoming`).
167 """
168 if existing is None:
169 return incoming
170 if incoming is None:
171 return existing
172 kind = str(existing.get("kind", ""))
173 strictness = {"IMPEDANCE_TARGET": "lower", "CLEARANCE": "lower", "CURRENT_BUDGET": "higher"}
174 if kind in strictness:
175 mode = strictness[kind]
176 ev = float(existing.get("value", {}).get("value", 0.0))
177 iv = float(incoming.get("value", {}).get("value", 0.0))
178 winner = existing if (mode == "lower" and ev <= iv) or (mode == "higher" and ev >= iv) else incoming
179 logger.info(
180 "arbitrage de mapping : contrainte concurrente résolue au plus strict",
181 extra={"context": context, "kind": kind,
182 "kept": winner.get("value", {}).get("value"), "mode": mode},
183 )
184 return winner
185 logger.info("arbitrage de mapping : plus récente retenue", extra={"context": context})
186 return incoming
