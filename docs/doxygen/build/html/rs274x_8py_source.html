16_FMT_HEADER = (
17 "%FSLAX36Y36*%\n" # format 3.6 — coordonnées en µm implicites
18 "%MOMM*%\n" # unités métriques
19 "%ADD10C,0.200*%\n" # aperture 10 : piste standard 0,2 mm
20 "%ADD11C,0.400*%\n" # aperture 11 : piste puissance 0,4 mm
21 "%ADD12C,0.300*%\n" # aperture 12 : via 0,3 mm
22 "%ADD13C,0.600*%\n" # aperture 13 : pastille standard 0,6 mm
23)
24_APERTURE = {0.2: "10", 0.4: "11", 0.3: "12", 0.6: "13"}

27def _coord(value_mm: float) -> str:
28 """Coordonnée RS-274X : entier 3.6 (µm), sans point décimal."""
29 return f"{int(round(value_mm * 1e6)):06d}"

32def _aperture_for(width_mm: float) -> str:
33 widths = sorted(_APERTURE)
34 best = min(widths, key=lambda w: abs(w - width_mm))
35 return _APERTURE[best]

38def render_copper_layer(board: Board, layer_index: int) -> str:
39 """RS-274X d'une couche cuivre : segments + vias + pastilles de la couche."""
40 lines: List[str] = ["*RS274X generated by pcb_ai_designer_v2/exporter*"]
41 lines.append(_FMT_HEADER)
42 lines.append("G01*") # interpolation linéaire
43
44 current_aperture = None
45 for net in board.nets.values():
46 for seg in net.routed_segments:
47 if seg.layer != layer_index:
48 continue
49 ap = _aperture_for(seg.width_mm) # via 0,3 mm → aperture 12 via la table
50 if ap != current_aperture:
51 lines.append(f"D{ap}*")
52 current_aperture = ap
53 x1, y1 = _coord(seg.x1_mm), _coord(seg.y1_mm)
54 x2, y2 = _coord(seg.x2_mm), _coord(seg.y2_mm)
55 if seg.is_via:
56 # via : flash d'une pastille circulaire
57 lines.append(f"X{x1}Y{y1}D03*")
58 else:
59 lines.append(f"X{x1}Y{y1}D02*")
60 lines.append(f"X{x2}Y{y2}D01*")
61
62 # pastilles des composants présents sur cette couche (aperture 13 : 0,6 mm)
63 pad_ap = _APERTURE[0.6]
64 lines.append(f"D{pad_ap}*")
65 for ref, comp in board.components.items():
66 placement = board.placements.get(ref)
67 if placement is None or placement.layer != layer_index:
68 continue
69 for pad in comp.pads:
70 lines.append(f"X{_coord(pad.x_mm)}Y{_coord(pad.y_mm)}D03*")
71
72 lines.append("M02*") # fin de fichier
73 return "\n".join(lines)

76def render_edge_cuts(board: Board) -> str:
77 """Contour de la carte (couche Edge.Cuts) : rectangle aux dimensions réelles."""
78 w, h = board.width_mm, board.height_mm
79 corners: List[Tuple[float, float]] = [(0, 0), (w, 0), (w, h), (0, h), (0, 0)]
80 lines = ["*RS274X edge cuts — pcb_ai_designer_v2*", _FMT_HEADER, "G01*", "D10*"]
81 prev = corners[0]
82 lines.append(f"X{_coord(prev[0])}Y{_coord(prev[1])}D02*")
83 for corner in corners[1:]:
84 lines.append(f"X{_coord(corner[0])}Y{_coord(corner[1])}D01*")
85 lines.append("M02*")
86 return "\n".join(lines)
