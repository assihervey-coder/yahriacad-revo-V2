1"""REINFORCE avec baseline — la passe de policy-gradient de l'agent [AutoPCB].
2
3Algorithme : pour chaque épisode, on accumule les (état, action, récompense)
4de l'environnement de placement ; à l'issue, on calcule les retours actualisés
5(return-to-go), on les centre par une baseline EMA (moyenne mobile des retours
6d'épisode) et on minimise :
7
8 L = −moyenne(log π(a_t|s_t) · (G_t − b)) − β · moyenne(H[π])
9
10Le bonus d'entropie (β) maintient l'exploration ; la baseline EMA réduit la
11variance sans introduction de critic (les épisodes sont courts : ≤ 12 pas).
12C'est le choix assumé REINFORCE plutôt que PPO : 3 ordres de grandeur moins
13de calcul par mise à jour, suffisant pour un espace d'action de dimension 4.
14
15torch est importé de façon gardée — sans torch, le module reste importable
16mais les classes ne sont pas définies (cf. torch_modules.py).
17"""

67 # ---- un épisode ------------------------------------------------------------------------
68 def _rollout(self) -> Dict[str, Any]:
69 """Un épisode complet : états, log-probs, entropies, rewards."""
70 state = torch.tensor(np.asarray(self.env.reset(), dtype=np.float32),
71 device=self.cfg.device)
72 log_probs, entropies, rewards = [], [], []
73 done, info_last = False, {}
74 while not done:
75 action_vec, aux = self.policy.act(state.unsqueeze(0),
76 deterministic=self.cfg.deterministic_eval)
77 next_state, reward, done, info = self.env.step(action_vec.detach().cpu().numpy()[0])
78 log_probs.append(aux["log_prob"].squeeze(0))
79 entropies.append(aux["entropy"].squeeze(0))
80 rewards.append(float(reward))
81 info_last = info
82 state = torch.tensor(np.asarray(next_state, dtype=np.float32),
83 device=self.cfg.device)
84 return {"log_probs": log_probs, "entropies": entropies,
85 "rewards": rewards, "final_info": info_last}

87 @staticmethod
88 def _returns(rewards: List[float], gamma: float) -> "torch.Tensor":
89 """Return-to-go actualisé : G_t = r_t + γ·G_{t+1} (Monte-Carlo)."""
90 returns, accumulator = [], 0.0
91 for reward in reversed(rewards):
92 accumulator = reward + gamma * accumulator
93 returns.insert(0, accumulator)
94 return torch.tensor(returns, dtype=torch.float32)

96 def train_episode(self) -> Dict[str, float]:
97 """Un épisode = rollout + mise à jour policy-gradient. Retour : métriques."""
98 rollout = self._rollout()
99 returns = self._returns(rollout["rewards"], self.cfg.gamma)
100 episode_return = float(returns[0].item())
101 self.baselinebaseline = (self.cfg.baseline_beta * self.baselinebaseline
102 + (1.0 - self.cfg.baseline_beta) * episode_return)
103 advantages = returns - self.baselinebaseline # baseline EMA — variance réduite
104 if advantages.numel() > 1:
105 std = advantages.std().item()
106 if std > 1e-6: # normalisation légère
107 advantages = (advantages - advantages.mean()) / (std + 1e-8)
108
109 log_probs = torch.stack(rollout["log_probs"])
110 entropies = torch.stack(rollout["entropies"])
111 policy_loss = -(log_probs * advantages).mean()
112 entropy_bonus = entropies.mean()
113 loss = policy_loss - self.cfg.entropy_beta * entropy_bonus
114
115 self.optimizer.zero_grad()
116 loss.backward()
117 torch.nn.utils.clip_grad_norm_(self.policy.parameters(), max_norm=5.0)
118 self.optimizer.step()
119
120 metrics = {"episode_return": round(episode_return, 4),
121 "baseline": round(self.baselinebaseline, 4),
122 "loss": round(float(loss.item()), 6),
123 "entropy": round(float(entropy_bonus.item()), 4),
124 "steps": len(rollout["rewards"]),
125 "hpwl_mm": rollout["final_info"].get("hpwl_mm", 0.0)}
126 self.history.append(metrics)
127 return metrics

144 mean_reward = (sum(m["episode_return"] for m in self.history) / len(self.history)
