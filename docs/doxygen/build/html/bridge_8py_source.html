79 """Sérialise la carte au format ASCII Altium simplifié (1 clé=valeur/ligne)."""
80 lines: List[str] = [
81 "[PCB]",
82 f"boardW={_fmt(board.width_mm)}",
83 f"boardH={_fmt(board.height_mm)}",
84 f"layers={len(board.layers)}",
85 ]
86 for ref, comp in board.components.items():
87 placement = board.placements.get(ref, Placement(ref=ref))
88 lines += [
89 "[COMPONENT]",
90 f"ref={comp.ref}",
91 f"footprint={(comp.footprint or comp.mpn or 'UNKNOWN').replace(' ', '_')}",
92 f"value={comp.value.replace(' ', '_') if comp.value else ''}",
93 f"w={_fmt(comp.width_mm)}",
94 f"h={_fmt(comp.height_mm)}",
95 f"pins={comp.pins}",
96 f"power={_fmt(comp.power_w)}",
97 f"functional={comp.functional_block or ''}",
98 f"x={_fmt(placement.x_mm)}",
99 f"y={_fmt(placement.y_mm)}",
100 f"rot={_fmt(placement.rotation_deg)}",
101 f"layer={placement.layer}",
102 f"locked={1 if placement.locked else 0}",
103 ]
104 for net in board.nets.values():
105 lines += [
106 "[NET]",
107 f"name={net.name}",
108 f"class={internal_class_to_altium(net.net_class, self.mapping)}",
109 f"impedance={_fmt(net.impedance_target_ohm) if net.impedance_target_ohm else '0'}",
110 f"conn={';'.join(f'{ref}/{pad}' for ref, pad in net.connections)}",
111 ]
112 for seg in net.routed_segments:
113 lines += [
114 "[TRACK]",
115 f"net={net.name}",
116 f"x1={_fmt(seg.x1_mm)}",
117 f"y1={_fmt(seg.y1_mm)}",
118 f"x2={_fmt(seg.x2_mm)}",
119 f"y2={_fmt(seg.y2_mm)}",
120 f"layer={seg.layer}",
121 f"w={_fmt(seg.width_mm)}",
122 f"via={1 if seg.is_via else 0}",
123 ]
124 for zone in board.zones:
125 lines += [
126 "[ZONE]",
127 f"name={zone.name.replace(' ', '_')}",
128 f"kind={zone.kind}",
129 f"x0={_fmt(zone.x_min_mm)}",
130 f"y0={_fmt(zone.y_min_mm)}",
131 f"x1={_fmt(zone.x_max_mm)}",
132 f"y1={_fmt(zone.y_max_mm)}",
133 f"maxtemp={_fmt(zone.max_temp_c) if zone.max_temp_c is not None else ''}",
134 ]
135 out_path = Path(out_path)
136 out_path.parent.mkdir(parents=True, exist_ok=True)
137 out_path.write_text("\n".join(lines) + "\n", encoding="utf-8")
138 logger.info("export Altium écrit", extra={"path": str(out_path),
139 "components": len(board.components)})
140 return out_path

143 def import_pcbdoc(self, path: Path) -> ImportReport:
144 """Parse un fichier ASCII Altium simplifié → carte interne + écarts journalisés."""
145 warnings: List[str] = []
146 board = Board()
147 state = "none"
148 comp: Optional[Component] = None
149 placement: Optional[Placement] = None
150 net: Optional[Net] = None

154 def finalize_component() -> None:
155 nonlocal comp, placement
156 if comp is not None and comp.ref:
157 board.add_component(comp, placement or Placement(ref=comp.ref))
158 elif comp is not None:
159 warnings.append("composant sans ref ignoré")
160 comp, placement = None, None
161
162 def finalize_segment() -> None:
163 nonlocal seg
164 if seg is not None and net is not None:
165 net.routed_segments.append(seg)
166 seg = None
167
168 def finalize_zone() -> None:
169 nonlocal zone
170 if zone:
171 try:
172 board.zones.append(Zone(
173 name=zone.get("name", "zone"),
174 kind=zone.get("kind", "keepout") or "keepout",
175 x_min_mm=_float(zone.get("x0"), 0.0),
176 y_min_mm=_float(zone.get("y0"), 0.0),
177 x_max_mm=_float(zone.get("x1"), 0.0),
178 y_max_mm=_float(zone.get("y1"), 0.0),
179 max_temp_c=(_float(zone["maxtemp"], 0.0)
180 if zone.get("maxtemp") else None),
181 ))
182 except Exception as exc: # zone malformée → écart journalisé
183 warnings.append(f"zone ignorée ({exc})")
184 logger.info("écart de sémantique : zone malformée", extra={"error": str(exc)})
185 zone = None
186
187 for raw_line in Path(path).read_text(encoding="utf-8").splitlines():
188 line = raw_line.strip()
189 if not line:
190 continue
191 if line.startswith("[") and line.endswith("]"):
192 section = line[1:-1].upper()
193 finalize_component()
194 finalize_segment()
195 finalize_zone()
196 if section == "COMPONENT":
197 comp, placement = Component(ref=""), None
198 state = "component"
199 elif section == "NET":
200 net = None
201 state = "net"
202 elif section == "TRACK":
203 state = "track"
204 elif section == "ZONE":
205 zone = {}
206 state = "zone"
207 else:
208 state = "none"
209 if section != "PCB":
210 warnings.append(f"section inconnue [{section}] ignorée")
211 logger.info("écart de sémantique : section inconnue",
212 extra={"section": section})
213 continue
214
215 kv = _parse_kv(line)
216 if kv is None:
217 warnings.append(f"ligne non structurée ignorée : {line[:40]!r}")
218 logger.info("écart de sémantique : ligne illisible ignorée",
219 extra={"line": line[:40]})
220 continue
221 key, value = kv
222
223 if state == "pcb" or state == "none":
224 if key == "boardW":
225 board.width_mm = _float(value, 100.0)
226 elif key == "boardH":
227 board.height_mm = _float(value, 80.0)
228 elif key == "layers" and int(_float(value, 4)) != len(board.layers):
229 warnings.append(
230 f"nb de couches Altium ({value}) ≠ modèle ({len(board.layers)}) — pile conservée")
231 elif state == "component" and comp is not None:
232 comp, placement = _feed_component(comp, placement, key, value, warnings)
233 elif state == "net":
234 if key == "name":
235 net = board.nets.get(value) or Net(name=value)
236 board.nets[value] = net
237 elif net is None:
238 warnings.append(f"clé NET '{key}' avant name — ignorée")
239 elif key == "class":
240 net.net_class = altium_class_to_internal(value, self.mapping)
241 elif key == "impedance":
242 net.impedance_target_ohm = _float(value, 0.0) or None
243 elif key == "conn":
244 for item in value.split(";"):
245 if "/" in item:
246 ref, pad = item.split("/", 1)
247 net.connections.append((ref, pad))
248 elif item:
249 warnings.append(f"connexion malformée '{item}' ignorée")
250 else:
251 warnings.append(f"clé NET inconnue '{key}' ignorée")
252 elif state == "track":
253 if key == "net":
254 finalize_segment()
255 net = board.nets.get(value) or board.nets.setdefault(value, Net(name=value))
256 elif seg is None and net is not None:
257 seg = Segment(net=net.name, x1_mm=0, y1_mm=0, x2_mm=0, y2_mm=0, layer=0)
258 if seg is not None:
259 if key == "x1":
260 seg.x1_mm = _float(value, 0.0)
261 elif key == "y1":
262 seg.y1_mm = _float(value, 0.0)
263 elif key == "x2":
264 seg.x2_mm = _float(value, 0.0)
265 elif key == "y2":
266 seg.y2_mm = _float(value, 0.0)
267 elif key == "layer":
268 seg.layer = int(_float(value, 0))
269 elif key == "w":
270 seg.width_mm = _float(value, 0.2)
271 elif key == "via":
272 seg.is_via = value.strip() in ("1", "true", "True")
273 finalize_segment()
274 else:
275 warnings.append(f"clé TRACK inconnue '{key}' ignorée")
276 elif state == "zone":
277 zone = zone or {}
278 zone[key] = value
279
280 finalize_component()
281 finalize_segment()
282 finalize_zone()
283
284 segments_total = sum(len(n.routed_segments) for n in board.nets.values())
285 report = ImportReport(
286 board=board, warnings=warnings,
287 components=len(board.components), nets=len(board.nets),
288 segments=segments_total, zones=len(board.zones),
289 )
290 if warnings:
291 logger.warning("import Altium avec écarts de sémantique",
292 extra={"warnings": len(warnings), "path": str(path)})
293 return report

296 def diff(self, board_a: Board, board_b: Board) -> List[str]:
297 """Écarts de sémantique entre deux cartes (après aller-retour attendu : vide)."""
298 gaps: List[str] = []
299 refs_a, refs_b = set(board_a.components), set(board_b.components)
300 for ref in sorted(refs_a - refs_b):
301 gaps.append(f"composant absent en B : {ref}")
302 for ref in sorted(refs_b - refs_a):
303 gaps.append(f"composant absent en A : {ref}")
304 for ref in sorted(refs_a & refs_b):
305 pa, pb = board_a.placements[ref], board_b.placements[ref]
306 if abs(pa.x_mm - pb.x_mm) > 1e-3 or abs(pa.y_mm - pb.y_mm) > 1e-3:
307 gaps.append(f"placement {ref} : ({pa.x_mm},{pa.y_mm}) ≠ ({pb.x_mm},{pb.y_mm})")
308 if abs(pa.rotation_deg - pb.rotation_deg) > 1e-3:
309 gaps.append(f"rotation {ref} : {pa.rotation_deg} ≠ {pb.rotation_deg}")
310 names_a, names_b = set(board_a.nets), set(board_b.nets)
311 for name in sorted(names_a - names_b):
312 gaps.append(f"net absent en B : {name}")
313 for name in sorted(names_b - names_a):
314 gaps.append(f"net absent en A : {name}")
315 for name in sorted(names_a & names_b):
316 la = board_a.nets[name].routed_length_mm
317 lb = board_b.nets[name].routed_length_mm
318 if abs(la - lb) > 1e-3:
319 gaps.append(f"longueur routée {name} : {la:.2f} ≠ {lb:.2f} mm")
320 return gaps
