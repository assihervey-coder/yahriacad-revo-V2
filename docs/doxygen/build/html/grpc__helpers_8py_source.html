38 """Démarre un serveur gRPC standardisé pour un microservice.
39
40 `add_servicer(servicer, server)` doit appeler
41 `pb2_grpc.add_XServicer_to_server(servicer, server)`.
42 Intercepte SIGINT/SIGTERM pour un arrêt propre (arrêt gracieux Kubernetes).
43 """
44 settings: Settings = get_settings(service_name)
45 configure_logging(service_name, settings.log_level)
46 logger = get_logger(service_name)
47
48 if not proto_available():
49 logger.error(
50 "stubs gRPC absents — exécuter « make proto » avant de lancer le service",
51 extra={"service": service_name},
52 )
53 sys.exit(2)
54
55 import grpc
56 from grpc_health.v1 import health, health_pb2, health_pb2_grpc
57
58 server = grpc.server(
59 futures.ThreadPoolExecutor(max_workers=max_workers),
60 options=[
61 ("grpc.keepalive_time_ms", 30_000),
62 ("grpc.keepalive_timeout_ms", 10_000),
63 ("grpc.max_message_length", 64 * 1024 * 1024), # designs denses
64 ],
65 )
66 add_servicer(None, server) # placeholder réel fourni par le main() du service
67
68 health_servicer = health.HealthServicer()
69 health_pb2_grpc.add_HealthServicer_to_server(health_servicer, server)
70 health_servicer.set("", health_pb2.HealthCheckResponse.SERVING)
71
72 bound_port = server.add_insecure_port(f"{settings.grpc_host}:{port or settings.grpc_port}")
73 server.start()
74 logger.info("service gRPC démarré", extra={"port": bound_port, "service": service_name})
75
76 stop = signal.signal(signal.SIGTERM, signal.SIG_IGN)
77 try:
78 signal.signal(signal.SIGTERM, lambda *_: server.stop(grace=5.0))
79 server.wait_for_termination()
80 except KeyboardInterrupt:
81 logger.info("arrêt demandé (SIGINT)", extra={"service": service_name})
82 server.stop(grace=5.0)
