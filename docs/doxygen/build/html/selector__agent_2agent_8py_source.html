75 def _pick(self, candidates: list[dict[str, Any]], block: str,
76 rejected: list[dict[str, str]]) -> tuple[dict[str, Any] | None, str]:
77 for record in candidates:
78 voltage = float(record.get("voltage_v", 0.0))
79 if block not in ("passive", "connector") and voltage and voltage < _RAIL_VOLTAGE_V:
80 rejected.append({"mpn": str(record.get("mpn")),
81 "reason": f"tension {voltage}V < rail {_RAIL_VOLTAGE_V}V"})
82 continue
83 if not record.get("footprint"):
84 rejected.append({"mpn": str(record.get("mpn")), "reason": "empreinte inconnue"})
85 continue
86 min_stock = _MIN_STOCK.get(block, _MIN_STOCK["default"])
87 if int(record.get("stock", 0)) < min_stock:
88 rejected.append({"mpn": str(record.get("mpn")),
89 "reason": f"stock < {min_stock}"})
90 continue
91 return record, (f"meilleur score coût/stock du bloc {block} "
92 f"({record.get('price_usd')} USD, stock {record.get('stock')})")
93 return None, "aucun candidat compatible"
