22def _wire(board: Board, net_name: str, refs: list[str]) -> Net:
23 """Net câblé : segments en étoile depuis le premier composant + 1 via."""
24 net = Net(name=net_name, connections=[(ref, "1") for ref in refs])
25 first = board.placements[refs[0]]
26 for ref in refs[1:]:
27 other = board.placements[ref]
28 net.routed_segments.append(
29 Segment(net=net_name, x1_mm=first.x_mm, y1_mm=first.y_mm,
30 x2_mm=other.x_mm, y2_mm=other.y_mm, layer=0)
31 )
32 net.routed_segments.append(
33 Segment(net=net_name, x1_mm=first.x_mm, y1_mm=first.y_mm,
34 x2_mm=first.x_mm + 1.0, y2_mm=first.y_mm, layer=1, is_via=True)
35 )
36 return net

49 """Tous les nets routés, une poignée de vias -> score quasi parfait.

51 Formule contractuelle de Board.drc_score() : 100 x complétude x

104def test_bus_publish_subscribe_latest_and_latency() -> None:
105 """Le bus retient la dernière valeur par clé, filtre par kind et reste
106 sous la cible de 50 ms (spécification bus de contraintes)."""
107 bus = InMemoryConstraintBus(target_latency_ms=50.0)
108 received: list[ConstraintMessage] = []
109 token_all = bus.subscribe(None, received.append)
110 token_thermal = bus.subscribe([ConstraintKind.THERMAL_ZONE_UPDATE], received.append)
111
112 for temp in (42.0, 55.0, 61.0):
113 latency_ms = bus.publish(ConstraintMessage(
114 kind=ConstraintKind.THERMAL_ZONE_UPDATE,
115 key="thermal/U12",
116 value={"temp_c": temp},
117 source="multi_physics_loop",
118 ))
119 assert latency_ms < 50.0
120
121 # 3 publications, 3 appels du subscriber global + 3 du thermal-only.
122 assert len(received) == 6
123 # Dernière valeur retenue par clé :
124 latest = bus.latest(ConstraintKind.THERMAL_ZONE_UPDATE)
125 assert latest["thermal/U12"].value["temp_c"] == 61.0
126 assert latest["thermal/U12"].revision == 3
127
128 bus.unsubscribe(token_all)
129 bus.unsubscribe(token_thermal)
130 received.clear()
131 bus.publish(ConstraintMessage(kind=ConstraintKind.KEEPOUT_ZONE, key="keepout/ANT",
132 value={"x_min": 0, "y_min": 0}))
133 assert received == [] # désinscrit -> plus rien ne parvient
