83 def execute_step(self, plan: ExecutionPlan, step: PlanStep,
84 handler: Callable[[dict[str, Any]], Any],
85 context: dict[str, Any],
86 priority: Priority = Priority.INTERACTIVE) -> dict[str, Any]:
87 """Alloue les ressources puis exécute l'étape sous budget de temps.
88
89 Le handler reçoit le contexte enrichi de l'allocation ; la durée est
90 comparée au budget — un dépassement est audité (sans bloquer).
91 """
92 allocation: Allocation = self.allocator.allocate(
93 step.agent, step.service, priority=priority, budget_s=step.budget_s)
94 run_context = {**context, "allocation": asdict(allocation)}
95 started = time.perf_counter()
96 try:
97 result = handler(run_context)
98 finally:
99 elapsed = time.perf_counter() - started
100 over_budget = elapsed > allocation.budget_s
101 if over_budget:
102 self.record_decision(
103 intention=f"exécuter l'étape {step.index} ({step.name}) dans le budget",
104 chosen=f"résultat accepté hors budget ({elapsed:.1f}s > {allocation.budget_s:.1f}s)",
105 rejected_alternatives=["relancer l'étape avec un budget réduit"],
106 rationale="le résultat est disponible — la relance coûterait plus cher")
107 self._audit_log.append({"action": "execute_step", "step": step.index,
108 "agent": step.agent, "endpoint": allocation.endpoint,
109 "elapsed_s": round(elapsed, 3),
110 "over_budget": over_budget, "ts": time.time()})
111 return {"result": result, "elapsed_s": round(elapsed, 3),
112 "allocation": allocation, "over_budget": over_budget}

114 def handle_escalation(self, err: Exception, context: dict[str, Any],
115 attempts: int = 1) -> dict[str, Any]:
116 """Escalade d'un agent : retry → dégradation → abort (dans cet ordre)."""
117 step = int(context.get("step", 0) or 0)
118 project_id = str(context.get("project_id", ""))
119 if attempts <= 1:
120 decision, rationale = "retry", "première erreur — une relance est peu coûteuse"
121 elif step not in CRITICAL_STEPS:
122 decision = "degrade"
123 rationale = ("étape non critique — poursuite avec le fallback de simulation, "
124 "le pipeline reste utilisable")
125 else:
126 decision, rationale = "abort", f"étape critique {step} en échec — arrêt propre"
127 payload = {"decision": decision, "attempts": attempts, "step": step,
128 "error": f"{type(err).__name__}: {err}", "rationale": rationale}
129 self.record_decision(
130 intention=f"résoudre l'échec de l'étape {step} de {project_id}",
131 chosen=decision, rejected_alternatives=["ignorer l'erreur (silencieux)"],
132 rationale=rationale)
133 self._audit_log.append({"action": "escalation", **payload, "ts": time.time()})
134 logger.warning("escalade traitée", extra=payload)
135 return payload
