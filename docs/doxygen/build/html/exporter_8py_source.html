66def _deduce_function(net_name: str, net_class: str, index: int) -> str:
67 """Fonction logicielle déduite : nom de net > classe de signaux > GPIO."""
68 for pattern, function in _FUNCTIONS:
69 if pattern.search(net_name or ""):
70 return f"{function}" if index < 1 else f"{function}_{index + 1}"
71 if (net_class or "").upper() in {"USB", "DDR", "MIPI", "ETHERNET"}:
72 return f"{net_class.upper()}_{index + 1}"
73 return "GPIO"

76def export_pins(board: Board, target: str = "raw") -> List[PinAssignment]:
77 """Exporte toutes les assignations de broches du design.
78
79 target : "raw" (liste brute), "zephyr" ou "arduino" (filtres futurs — la
80 génération des en-têtes est dans header_generator).
81 """
82 assignments: List[PinAssignment] = []
83 for ref in sorted(board.components):
84 comp = board.components[ref]
85 if not comp.pads:
86 continue
87 net_of_pad = {p.name: (p.net or "") for p in comp.pads}
88 # complète avec les connexions déclarées dans les nets
89 for net_name, net in board.nets.items():
90 for conn_ref, pad_name in net.connections:
91 if conn_ref == ref and not net_of_pad.get(pad_name):
92 net_of_pad[pad_name] = net_name
93 for index, pad in enumerate(comp.pads):
94 net_name = pad.net or net_of_pad.get(pad.name, "")
95 function = _deduce_function(net_name, _class_of(board, net_name), index)
96 pin_label = _physical_pin(comp.ref, pad.name, index)
97 assignments.append(PinAssignment(component_ref=ref, pin=pin_label,
98 function=function, net=net_name))
99 logger.info("broches exportées", extra={"components": len(board.components),
100 "pins": len(assignments), "target": target})
101 return assignments
