1"""Checker déterministe — géométrie, électrique, routabilité, contraintes du bus.
2
3Dernier rempart avant engagement [Siemens Fuse] : aucune heuristique
4apprisent, uniquement de la physique reproductible. Trois familles de tests :
5 1. GÉOMÉTRIE — bounding boxes (collision, clearance 0.2 mm par défaut,
6 surchargée par la contrainte CLEARANCE du bus), zones keepout (du board
7 ET du bus), limites de carte ;
8 2. ÉLECTRIQUE — chute de tension sur les nets d'alim : R ≈ ρ·L/A
9 (cuivre 1.724e-5 Ω·mm, section = largeur × 35 µm), seuil 3 % ;
10 3. ROUTABILITÉ — A* 8-directions sur grille clairsemée (cellule 2 mm) :
11 l'action ne doit murer aucun composant.
12Le verdict cite les clés du constraint_bus violées (ex. ``keepout/antenna``).
13"""

58 def __init__(self, bus: Any = None, rules: Optional[DesignRules] = None) -> None:
59 self.bus = bus # ProjectConstraintBus optionnel

62 # ---- lecture des contraintes du bus ----------------------------------------------
63 def _clearance(self) -> float:
64 """Clearance effective = max(défaut, dernière contrainte CLEARANCE)."""
65 clearance = self.rules.min_clearance_mm
66 if self.bus is not None:
67 for message in self.bus.latest(ConstraintKind.CLEARANCE).values():
68 clearance = max(clearance, float(message.value.get("min_clearance_mm", 0.0)))
69 return clearance

71 def _keepout_rects(self, board: Any) -> List[Tuple[str, Tuple[float, float, float, float]]]:
72 """Rects keepout du board + ceux publiés sur le bus (source externe)."""
73 rects = [(f"keepout/{z.name}", (z.x_min_mm, z.y_min_mm, z.x_max_mm, z.y_max_mm))
74 for z in board.zones if z.kind == "keepout"]
75 if self.bus is not None:
76 for key, message in self.bus.latest(ConstraintKind.KEEPOUT_ZONE).items():
77 v = message.value
78 rects.append((key, (float(v.get("x_min_mm", 0.0)), float(v.get("y_min_mm", 0.0)),
79 float(v.get("x_max_mm", 0.0)), float(v.get("y_max_mm", 0.0)))))
80 return rects

82 # ---- API principale -----------------------------------------------------------------
83 def check(self, board: Any, action: Any) -> Verdict:
84 """Vérifie une action (ref, x_mm, y_mm, rotation_deg, layer) sur le board."""
85 comp = board.components.get(action.ref)
86 if comp is None:
87 return Verdict(False, [f"component/{action.ref}"],
88 f"composant inconnu : {action.ref}", "geometrie")
89 x1, y1, x2, y2 = comp.bounding_box(
90 Placement(ref=action.ref, x_mm=action.x_mm, y_mm=action.y_mm,
91 rotation_deg=getattr(action, "rotation_deg", 0.0),
92 layer=getattr(action, "layer", 0)))
93 violated: List[str] = []
94 # ---- 1) géométrie -------------------------------------------------------------
95 clearance = self._clearance()
96 if x1 < -0.01 or y1 < -0.01 or x2 > board.width_mm + 0.01 or y2 > board.height_mm + 0.01:
97 violated.append(f"board_bounds/{action.ref}")
98 for key, rect in self._keepout_rects(board):
99 kx1, ky1, kx2, ky2 = rect
100 if not (x2 <= kx1 + clearance or kx2 - clearance <= x1
101 or y2 <= ky1 + clearance or ky2 - clearance <= y1):
102 if key not in violated:
103 violated.append(key)
104 for other_ref, other_p in board.placements.items():
105 if other_ref == action.ref or other_ref not in board.components:
106 continue
107 ox1, oy1, ox2, oy2 = board.components[other_ref].bounding_box(other_p)
108 gap = max(ox1 - x2, x1 - ox2, oy1 - y2, y1 - oy2)
109 if gap < clearance:
110 key = f"collision/{other_ref}"
111 if key not in violated:
112 violated.append(key)
113 if violated:
114 return Verdict(False, violated, "chevauchement géométrique ou keepout", "geometrie")
115 # ---- 2) électrique ----------------------------------------------------------------
116 drop = self._worst_voltage_drop_pct(board, action)
117 if drop is not None and drop > self.rules.max_voltage_drop_pct:
118 return Verdict(False, [f"voltage_drop/{drop:.1f}pct"],
119 f"chute de tension estimée {drop:.1f} % > "
120 f"{self.rules.max_voltage_drop_pct} %", "electrique")
121 # ---- 3) routabilité -------------------------------------------------------------------
122 if not self._routable(board, action, (x1, y1, x2, y2)):
123 return Verdict(False, ["routability_blocked"],
124 "l'action mure une connexion (A* sans chemin)", "routabilite")
125 return Verdict(True, [], "action physiquement valide", "")

127 # ---- électrique -----------------------------------------------------------------------
128 def _worst_voltage_drop_pct(self, board: Any, action: Any) -> Optional[float]:
129 """Chute de tension max (%) sur les nets d'alim touchés par l'action.
130
131 R = ρ·L/A avec L = distance Manhattan estimée entre les deux charges
132 les plus éloignées du net, A = largeur × épaisseur (0.2 × 0.035 mm²).
133 """
134 worst: Optional[float] = None
135 for net in board.nets.values():
136 if net.net_class != "power" or len(net.connections) < 2:
137 continue
138 refs = {ref for ref, _pad in net.connections if ref in board.placements}
139 if action.ref not in refs:
140 continue
141 if self.bus is not None:
142 current = 0.0
143 for message in self.bus.latest(ConstraintKind.CURRENT_BUDGET).values():
144 current = max(current, float(message.value.get("budget_a", 0.0)))
145 current = current or self.rules.default_current_a
146 else:
147 current = self.rules.default_current_a
148 points = []
149 for ref in refs:
150 p = board.placements[ref]
151 points.append((p.x_mm, p.y_mm))
152 span = max(math.hypot(px - qx, py - qy)
153 for i, (px, py) in enumerate(points)
154 for (qx, qy) in points[i + 1:]) if len(points) > 1 else 0.0
155 area = self.rules.trace_width_mm * _TRACE_THICKNESS_MM
156 resistance = _RHO_CU_OHM_MM * span / max(area, 1e-9)
157 drop_pct = 100.0 * current * resistance / _DEFAULT_V_RAIL
158 worst = drop_pct if worst is None else max(worst, drop_pct)
159 return worst

161 # ---- routabilité (A* 8 directions) --------------------------------------------------------
162 def _routable(self, board: Any, action: Any, bbox: Tuple[float, float, float, float]) -> bool:
163 """Vérifie qu'un chemin libre relie le composant déplacé à son partenaire
164 de net le plus éloigné — grille clairsemée, obstacles = keepouts + autres
165 composants. Sans partenaire : la cellule centre de carte fait office de but."""
166 cell = self.rules.cell_mm
167 cols = max(2, int(math.ceil(board.width_mm / cell)))
168 rows = max(2, int(math.ceil(board.height_mm / cell)))
169 blocked = [[False] * cols for _ in range(rows)]
170
171 def block_rect(x1: float, y1: float, x2: float, y2: float, margin: float) -> None:
172 c1, r1 = max(0, int((x1 - margin) / cell)), max(0, int((y1 - margin) / cell))
173 c2 = min(cols - 1, int((x2 + margin) / cell))
174 r2 = min(rows - 1, int((y2 + margin) / cell))
175 for r in range(r1, r2 + 1):
176 for c in range(c1, c2 + 1):
177 blocked[r][c] = True
178
179 for key, (kx1, ky1, kx2, ky2) in self._keepout_rects(board):
180 block_rect(kx1, ky1, kx2, ky2, 0.0)
181 for other_ref, other_p in board.placements.items():
182 if other_ref == action.ref or other_ref not in board.components:
183 continue
184 ox1, oy1, ox2, oy2 = board.components[other_ref].bounding_box(other_p)
185 block_rect(ox1, oy1, ox2, oy2, 0.0)
186 # le composant déplacé reste praticable autour de sa nouvelle bbox
187 bx1, by1, bx2, by2 = bbox
188 for r in range(max(0, int(by1 / cell)), min(rows, int(by2 / cell) + 1)):
189 for c in range(max(0, int(bx1 / cell)), min(cols, int(bx2 / cell) + 1)):
190 blocked[r][c] = False
191
192 start = (min(rows - 1, int((by1 + by2) / 2.0 / cell)),
193 min(cols - 1, int((bx1 + bx2) / 2.0 / cell)))
194 goal = self._goal_cell(board, action, rows, cols, cell)
195 # Le but (centre du partenaire) peut tomber DANS sa propre bbox bloquée :
196 # on vise la cellule libre la plus proche (périphérie du composant).
197 goal = self._nearest_free(blocked, rows, cols, goal)
198 if goal is None:
199 return False
200 return self._astar(blocked, rows, cols, start, goal) is not None

202 def _goal_cell(self, board: Any, action: Any, rows: int, cols: int,
203 cell: int) -> Tuple[int, int]:
204 """Cellule but : partenaire de net le plus éloigné, sinon centre de carte."""
205 best_xy = (board.width_mm / 2.0, board.height_mm / 2.0)
206 best_dist = -1.0
207 for net in board.nets.values():
208 refs = {ref for ref, _pad in net.connections if ref in board.placements}
209 if action.ref not in refs:
210 continue
211 p = board.placements[action.ref]
212 for ref in refs - {action.ref}:
213 q = board.placements[ref]
214 dist = math.hypot(p.x_mm - q.x_mm, p.y_mm - q.y_mm)
215 if dist > best_dist:
216 best_dist, best_xy = dist, (q.x_mm, q.y_mm)
217 return (min(rows - 1, int(best_xy[1] / cell)), min(cols - 1, int(best_xy[0] / cell)))

219 @staticmethod
220 def _nearest_free(blocked: List[List[bool]], rows: int, cols: int,
221 cell: Tuple[int, int]) -> Optional[Tuple[int, int]]:
222 """Cellule libre la plus proche (recherche en anneaux BFS) — le but du
223 A* doit être atteignable sans traverser l'obstacle qu'est le composant
224 partenaire lui-même."""
225 if not blocked[cell[0]][cell[1]]:
226 return cell
227 seen = {cell}
228 frontier = [cell]
229 while frontier:
230 next_ring: List[Tuple[int, int]] = []
231 for (r, c) in frontier:
232 for dr in (-1, 0, 1):
233 for dc in (-1, 0, 1):
234 nr, nc = r + dr, c + dc
235 if 0 <= nr < rows and 0 <= nc < cols and (nr, nc) not in seen:
236 if not blocked[nr][nc]:
237 return (nr, nc)
238 seen.add((nr, nc))
239 next_ring.append((nr, nc))
240 frontier = next_ring
241 return None

243 @staticmethod
244 def _astar(blocked: List[List[bool]], rows: int, cols: int,
245 start: Tuple[int, int], goal: Tuple[int, int]) -> Optional[Tuple[int, int]]:
246 """A* 8-directions classique (coûts 1 / √2, heuristique octile)."""
247
248 def heuristic(r: int, c: int) -> float:
249 dr, dc = abs(r - goal[0]), abs(c - goal[1])
250 return (dr + dc) + (math.sqrt(2.0) - 2.0) * min(dr, dc)
251
252 frontier = [(0.0, start)]
253 cost = {start: 0.0}
254 while frontier:
255 _f, (r, c) = heapq.heappop(frontier)
256 if (r, c) == goal:
257 return goal
258 for dr in (-1, 0, 1):
259 for dc in (-1, 0, 1):
260 if dr == 0 and dc == 0:
261 continue
262 nr, nc = r + dr, c + dc
263 if 0 <= nr < rows and 0 <= nc < cols and not blocked[nr][nc]:
264 new_cost = cost[(r, c)] + math.hypot(dr, dc)
265 if new_cost < cost.get((nr, nc), float("inf")):
266 cost[(nr, nc)] = new_cost
267 heapq.heappush(frontier, (new_cost + heuristic(nr, nc), (nr, nc)))
268 return None
