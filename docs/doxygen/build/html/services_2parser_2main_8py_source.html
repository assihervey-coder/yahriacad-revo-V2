37def import_file(design_id: str, filename: str, content_bytes: bytes,
38 format_hint: str = "auto") -> ImportReport:
39 """Importe un fichier (SPICE ou schéma KiCad) et le normalise vers un Board."""
40 fmt = detect_format(filename, content_bytes, format_hint)
41 text = content_bytes.decode("utf-8", errors="replace")
42 try:
43 if fmt == "kicad_sch":
44 parsed = parse_kicad_sch(text)
45 report = normalize_import([inst.component for inst in parsed.components],
46 parsed.net_connections, parsed.wire_hints,
47 parsed.warnings)
48 else:
49 parsed = parse_spice(text)
50 report = normalize_import(parsed.components, parsed.net_connections, None,
51 parsed.warnings)
52 except Exception as exc: # entrée invalide : rapport explicite plutôt qu'exception
53 logger.warning("import échoué", extra={"source_file": filename, "format": fmt,
54 "error": str(exc)})
55 return ImportReport(False, 0, 0, [f"parse {fmt} impossible : {exc}"], Board())
56
57 event = make_event(EventType.STEP_PROGRESS, design_id, emitter="parser",
58 step=WorkflowStep.NL_TO_SKIDL.value,
59 detail=f"import {fmt} : {report.components_imported} composants")
60 logger.info("événement émis", extra=event.to_json())
61 return report
