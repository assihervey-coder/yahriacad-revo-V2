40# Valeurs SPICE : 10k, 100n, 1meg, 4k7, 2R2, 3.3 (m = milli, meg = méga)
41_VALUE_RE = re.compile(r"^(?P<mant>[0-9]*\.?[0-9]+)\s*(?P<suf>meg|[TGKMUNPFRtgkmunpfr])?(?P<dec>[0-9]+)?$")
42_MULT = {"t": 1e12, "g": 1e9, "meg": 1e6, "k": 1e3, "m": 1e-3,
43 "u": 1e-6, "n": 1e-9, "p": 1e-12, "f": 1e-15, "r": 1.0}

46def parse_spice_value(token: str) -> float:
47 """Convertit une valeur SPICE en ohms/farads/henrys (4k7 → 4700.0)."""
48 match = _VALUE_RE.match(token.strip())
49 if not match:
50 raise ValueError(f"valeur SPICE illisible : {token!r}")
51 mant, suffix, dec = match.group("mant"), (match.group("suf") or "").lower(), match.group("dec")
52 value = float(mant)
53 if dec: # notation « 4k7 » : décimales après le suffixe
54 value = float(f"{mant}.{dec}")
55 return value * _MULT.get(suffix, 1.0)

63def guess_functional_block(prefix: str, value: str) -> str:
64 """Bloc fonctionnel heuristique — consommé par le BOM et le placement RL."""
65 upper = (value or "").upper()
66 if any(k in upper for k in ("RFM", "SX1", "SX6", "LORA", "NRF", "CC11")):
67 return "rf"
68 if prefix == "U" and any(k in upper for k in ("STM", "ESP", "PIC", "ATMEGA", "GD32", "MK6")):
69 return "mcu"
70 if prefix in {"X", "J"}:
71 return "connector"
72 if prefix in {"Q", "M"}:
73 return "power"
74 if prefix in {"R", "C", "L"}:
75 return "passive"
76 return "logic"

117def _handle_directive(line: str, subckt_stack: List[str], result: SpiceNetlist) -> None:
118 """Traite les directives pointées (.subckt/.ends/.include/.model/.end)."""
119 tokens = line.split()
120 keyword = tokens[0].lower()
121 if keyword == ".subckt":
122 name = tokens[1] if len(tokens) > 1 else "?"
123 subckt_stack.append(name)
124 result.warnings.append(f"sous-circuit '{name}' intégré (composants internes préfixés)")
125 elif keyword == ".ends":
126 if subckt_stack:
127 subckt_stack.pop()
128 else:
129 result.warnings.append("directive .ends orpheline ignorée")
130 elif keyword == ".include":
131 target = tokens[1] if len(tokens) > 1 else "?"
132 result.warnings.append(f"directive .include ignorée ({target})")
133 elif keyword in {".model", ".end", ".param", ".option", ".global"}:
134 logger.debug("directive SPICE ignorée", extra={"directive": keyword})
135 else:
136 result.warnings.append(f"directive SPICE inconnue ignorée : {keyword}")

139def parse_spice(text: str) -> SpiceNetlist:
140 """Parse une netliste SPICE en composants + connexions (ref, pad) par net."""
141 result = SpiceNetlist()
142 comps: Dict[str, Component] = {}
143
144 # 1) fusion des continuations « + » et retrait des commentaires (* et ;)
145 logical: List[str] = []
146 for raw in text.splitlines():
147 line = raw.split(";", 1)[0].rstrip()
148 if not line.strip() or line.lstrip().startswith("*"):
149 continue
150 if line.startswith("+") and logical:
151 logical[-1] += " " + line[1:].strip()
152 continue
153 logical.append(line.strip())
154
155 subckt_stack: List[str] = []
156 for line in logical:
157 if line.lower().startswith("."):
158 _handle_directive(line, subckt_stack, result)
159 continue
160 tokens = line.split()
161 ref_raw = tokens[0]
162 prefix = ref_raw[0].upper()
163 # Convention de référence (alignée sur le benchmark) : la lettre SPICE
164 # "X" est un marqueur de sous-circuit — si elle est suivie du préfixe
165 # métier (XU1, XR1, XC1...), on la retire : XU1 → U1, XR1 → R1.
166 if (prefix == "X" and len(ref_raw) >= 3 and ref_raw[1].isalpha()
167 and ref_raw[2:].isdigit()):
168 ref_raw = ref_raw[1:]
169 prefix = ref_raw[0].upper()
170 scope = subckt_stack[-1] if subckt_stack else ""
171 ref = f"{scope}_{ref_raw}" if scope else ref_raw
172 if len(tokens) >= 4: # dernier token = valeur/modèle, le reste = nœuds
173 nodes, value = tokens[1:-1], tokens[-1]
174 else:
175 nodes, value = tokens[1:], ""
176 if prefix == "X" and "=" in value: # instance X : nom du sous-circuit seul
177 value = value.split("=", 1)[0]
178 if ref in comps:
179 result.warnings.append(f"référence dupliquée dans la netlist : {ref} (ignorée)")
180 continue
181 comps[ref] = _make_component(ref, prefix, value, len(nodes))
182 for index, node in enumerate(nodes):
183 net = _canonical_net(node)
184 result.net_connections.setdefault(net, []).append((ref, str(index + 1)))
185
186 result.components = list(comps.values())
187 if not result.components:
188 result.warnings.append("aucun composant détecté dans la netlist SPICE")
189 logger.info("netlist SPICE analysée",
190 extra={"components": len(result.components), "nets": len(result.net_connections)})
191 return result
