53def build_schema(runtime: Any):
54 """Construit le GraphQLSchema exécutable — None si graphql-core absent."""
55 try:
56 import graphql # noqa: F401 — disponibilité testée par la construction
57 from graphql import (GraphQLArgument, GraphQLField, GraphQLID,
58 GraphQLInt, GraphQLList, GraphQLObjectType,
59 GraphQLSchema, GraphQLString)
60 except ImportError:
61 return None
62
63 def resolve_design(_root: Any, _info: Any, id: str) -> dict[str, Any]: # noqa: A002
64 return {"id": id, "version": runtime.get_design_state(id)["version"],
65 "boardJson": _json(runtime.get_design_state(id)["board"]),
66 "metrics": _json(runtime.state_manager.metrics(id))}
67
68 def resolve_status(_root: Any, _info: Any, id: str) -> dict[str, Any]:
69 run = runtime.get_run(id)
70 return {"pipelineId": run["pipeline_id"], "projectId": run["project_id"],
71 "state": run["state"], "steps": _json(run["statuses"])}
72
73 def resolve_run_pipeline(_root: Any, _info: Any, id: str,
74 steps: list[int] | None = None) -> dict[str, Any]:
75 run = runtime.run_pipeline(id, steps=steps)
76 return {"pipelineId": run["pipeline_id"], "projectId": run["project_id"],
77 "state": run["state"], "steps": _json(run["statuses"])}
78
79 design_type = GraphQLObjectType(
80 "Design", lambda: {
81 "id": GraphQLField(GraphQLID),
82 "version": GraphQLField(GraphQLInt),
83 "boardJson": GraphQLField(GraphQLString),
84 "metrics": GraphQLField(GraphQLString),
85 })
86 status_type = GraphQLObjectType(
87 "PipelineStatus", lambda: {
88 "pipelineId": GraphQLField(GraphQLID),
89 "projectId": GraphQLField(GraphQLID),
90 "state": GraphQLField(GraphQLString),
91 "steps": GraphQLField(GraphQLString),
92 })
93 query = GraphQLObjectType("Query", lambda: {
94 "design": GraphQLField(design_type, args={"id": GraphQLArgument(GraphQLID)},
95 resolve=resolve_design),
96 "status": GraphQLField(status_type, args={"id": GraphQLArgument(GraphQLID)},
97 resolve=resolve_status),
98 })
99 mutation = GraphQLObjectType("Mutation", lambda: {
100 "runPipeline": GraphQLField(
101 status_type,
102 args={"id": GraphQLArgument(GraphQLID),
103 "steps": GraphQLArgument(GraphQLList(GraphQLInt))},
104 resolve=resolve_run_pipeline),
105 })
106 return GraphQLSchema(query=query, mutation=mutation)

109def execute_query(runtime: Any, query: str, variables: dict[str, Any] | None = None) -> dict[str, Any]:
110 """Exécute une requête GraphQL — erreur explicite si graphql-core absent."""
111 schema = build_schema(runtime)
112 if schema is None:
113 return {"errors": [{"message":
114 "graphql-core absent — schéma descriptif uniquement (voir SCHEMA_SDL/TYPES)"}]}
115 from graphql import graphql_sync
116 result = graphql_sync(schema, query, variable_values=variables)
117 return {"data": result.data,
118 "errors": [e.formatted for e in (result.errors or [])]}
