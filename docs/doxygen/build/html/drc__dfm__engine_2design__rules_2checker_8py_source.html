68def check_design_rules(board: Board, rules: Dict[str, DesignRule] | None = None) -> DrcReport:
69 """Applique toutes les règles géométriques — retourne rapport + score."""
70 rules = rules or default_rules()
71 violations: List[Violation] = []
72
73 violations += _check_placement_overlaps(board, rules["placement_overlap"])
74 violations += _check_keepouts(board, rules["keepout"])
75 violations += _check_edge_clearance(board, rules["edge_clearance"])
76 violations += _check_min_width(board, rules["min_width"])
77 violations += _check_vias(board, rules["via_diameter"], rules["annular_ring"])
78 violations += _check_unrouted(board, rules["unrouted"])
79
80 # score composite : 100 − 12 pts par erreur − 3 pts par avertissement, borné
81 score = max(0.0, 100.0 - 12.0 * len([v for v in violations if v.severity == "error"])
82 - 3.0 * len([v for v in violations if v.severity == "warning"]))
83 return DrcReport(pass_=not any(v.severity == "error" for v in violations),
84 score=round(score, 2), violations=violations)

93def _check_placement_overlaps(board: Board, rule: DesignRule) -> List[Violation]:
94 margin = rule.params.get("margin_mm", 0.1)
95 refs = list(board.components)
96 out: List[Violation] = []
97 for i, ref_a in enumerate(refs):
98 for ref_b in refs[i + 1:]:
99 try:
100 if board.bounding_box_overlap(ref_a, ref_b):
101 pa, pb = board.placements[ref_a], board.placements[ref_b]
102 ca, cb = board.components[ref_a], board.components[ref_b]
103 ba, bb = ca.bounding_box(pa), cb.bounding_box(pb)
104 out.append(Violation(
105 rule_id=rule.rule_id, severity=rule.severity,
106 component_ref=f"{ref_a}/{ref_b}",
107 area=(min(ba[0], bb[0]), min(ba[1], bb[1]),
108 max(ba[2], bb[2]), max(ba[3], bb[3])),
109 message=f"chevauchement d'empreintes {ref_a} / {ref_b} "
110 f"(marge {margin} mm non respectée)",
111 ))
112 except PermissionError:
113 pass # composant verrouillé — état transitoire autorisé
114 return out

117def _check_keepouts(board: Board, rule: DesignRule) -> List[Violation]:
118 out: List[Violation] = []
119 for zone in board.zones:
120 if zone.kind != "keepout":
121 continue
122 zbox = (zone.x_min_mm, zone.y_min_mm, zone.x_max_mm, zone.y_max_mm)
123 for ref, comp in board.components.items():
124 placement = board.placements.get(ref)
125 if placement is None:
126 continue
127 cbox = comp.bounding_box(placement)
128 if _boxes_intersect(cbox, zbox):
129 out.append(Violation(
130 rule_id=rule.rule_id, severity=rule.severity,
131 component_ref=ref, area=zbox,
132 message=f"{ref} empiète sur la zone interdite « {zone.name} »",
133 ))
134 # pistes traversant le keepout
135 for net in board.nets.values():
136 for seg in net.routed_segments:
137 if seg.is_via:
138 continue
139 sbox = (min(seg.x1_mm, seg.x2_mm) - seg.width_mm / 2,
140 min(seg.y1_mm, seg.y2_mm) - seg.width_mm / 2,
141 max(seg.x1_mm, seg.x2_mm) + seg.width_mm / 2,
142 max(seg.y1_mm, seg.y2_mm) + seg.width_mm / 2)
143 if _boxes_intersect(sbox, zbox):
144 out.append(Violation(
145 rule_id=rule.rule_id, severity=rule.severity,
146 net=net.name, area=zbox,
147 message=f"piste du net {net.name} traverse la zone « {zone.name} »",
148 ))
149 return out

152def _check_edge_clearance(board: Board, rule: DesignRule) -> List[Violation]:
153 min_edge = rule.params.get("min_edge_mm", 0.5)
154 out: List[Violation] = []
155 w, h = board.width_mm, board.height_mm
156 for ref, comp in board.components.items():
157 placement = board.placements.get(ref)
158 if placement is None:
159 continue
160 x1, y1, x2, y2 = comp.bounding_box(placement)
161 if x1 < min_edge or y1 < min_edge or x2 > w - min_edge or y2 > h - min_edge:
162 out.append(Violation(
163 rule_id=rule.rule_id, severity=rule.severity, component_ref=ref,
164 area=(x1, y1, x2, y2),
165 message=f"{ref} à moins de {min_edge} mm du bord de carte",
166 ))
167 return out

205def _boxes_intersect(a: Tuple[float, float, float, float],
206 b: Tuple[float, float, float, float]) -> bool:
