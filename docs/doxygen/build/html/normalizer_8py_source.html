41def detect_format(filename: str, content_bytes: bytes, format_hint: str = "auto") -> str:
42 """Détermine le format d'entrée : « spice » ou « kicad_sch »."""
43 hint = (format_hint or "auto").strip().lower()
44 if hint in {"spice", "kicad_sch"}:
45 return hint
46 name = (filename or "").lower()
47 head = content_bytes[:8192].decode("utf-8", errors="ignore")
48 lowered = head.lower()
49 if "kicad_sch" in lowered or name.endswith(".kicad_sch"):
50 return "kicad_sch"
51 if name.endswith((".net", ".cir", ".spice", ".sp", ".cki")) or ".subckt" in lowered:
52 return "spice"
53 if re.search(r"\‍(\s*kicad_sch", head):
54 return "kicad_sch"
55 if re.search(r"^[A-Za-z]{1,2}\S*\s+\S+\s+\S+", head, re.M):
56 return "spice"
57 return "spice" # repli : format le plus permissif

60def _fallback_footprint(pins: int) -> str:
61 """Empreinte générique quand le composant importé n'en définit aucune."""

63 return "Resistor_SMD:R_0603_1608Metric"

69def _default_pad_layout(n_pins: int, width_mm: float, height_mm: float) -> List[Pad]:
70 """Disposition DIP en coordonnées locales : moitié gauche, moitié droite."""
71 pads: List[Pad] = []
72 left_count = (n_pins + 1) // 2
73 right_count = n_pins - left_count
74 offset = width_mm / 2.0 + 0.95 # pastilles juste à l'extérieur du boîtier
75 for index in range(n_pins):
76 on_left = index < left_count
77 local_index = index if on_left else index - left_count
78 rows = max(left_count, right_count, 1)
79 y = -height_mm / 2.0 + (local_index + 1) * height_mm / (rows + 1)
80 pads.append(Pad(name=str(index + 1), x_mm=-offset if on_left else offset, y_mm=y,
81 diameter_mm=0.6))
82 return pads

85def _place_on_grid(components: List[Component]) -> Dict[str, Placement]:
86 """Placement initial en grille 4 colonnes, trié par bloc fonctionnel."""
87 ordered = sorted(components, key=lambda c: (_BLOCK_ORDER.get(c.functional_block, 9), c.ref))
88 placements: Dict[str, Placement] = {}
89 for index, comp in enumerate(ordered):
90 col, row = index % 4, index // 4
91 placements[comp.ref] = Placement(ref=comp.ref, x_mm=14.0 + col * 22.0, y_mm=12.0 + row * 18.0)
92 return placements

95def normalize_import(components: Sequence[Component],
96 net_connections: Dict[str, List[Tuple[str, str]]],
97 wire_hints: Optional[Sequence] = None,
98 warnings: Optional[List[str]] = None) -> ImportReport:
99 """Fusionne les sorties des parseurs en un Board normalisé + avertissements."""
100 warns: List[str] = list(warnings or [])
101 if wire_hints:
102 logger.debug("wire hints conservés pour le router", extra={"count": len(list(wire_hints))})
103
104 # 1) validation des composants : refs uniques + empreintes présentes
105 clean: List[Component] = []
106 for comp in components:
107 if any(comp.ref == other.ref for other in clean):
108 comp.ref = f"{comp.ref}_dup{len(clean)}"
109 warns.append(f"référence dupliquée renommée en « {comp.ref} »")
110 if not comp.footprint:
111 comp.footprint = _fallback_footprint(max(comp.pins, 2))
112 warns.append(f"empreinte manquante pour {comp.ref} — boîtier générique appliqué")
113 if comp.pins <= 0:
114 comp.pins = 2
115 clean.append(comp)
116
117 # 2) validation des nets : détecter les nets flottants (< 2 connexions)
118 for net_name, conns in sorted(net_connections.items()):
119 if len(conns) < 2:
120 refs = ", ".join(f"{r}.{p}" for r, p in conns) or "?"
121 warns.append(f"net flottant « {net_name} » ({len(conns)} connexion(s) : {refs})")
122
123 # 3) placement initial + carte dimensionnée
124 board = Board()
125 placements = _place_on_grid(clean) if clean else {}
126 if placements:
127 max_x = max(placements[c.ref].x_mm + c.width_mm / 2.0 for c in clean)
128 max_y = max(placements[c.ref].y_mm + c.height_mm / 2.0 for c in clean)
129 board.width_mm = max(40.0, math.ceil(max_x + 8.0))
130 board.height_mm = max(30.0, math.ceil(max_y + 8.0))
131
132 # 4) pads : disposition locale → coordonnées absolues, net attaché
133 conn_by_key: Dict[Tuple[str, str], str] = {}
134 for net_name, conns in net_connections.items():
135 for ref, pad in conns:
136 conn_by_key[(ref, str(pad))] = net_name
137 for comp in clean:
138 place = placements[comp.ref]
139 comp.pads = []
140 for local in _default_pad_layout(comp.pins, comp.width_mm, comp.height_mm):
141 local.x_mm = place.x_mm + local.x_mm # convention : pads en coordonnées absolues
142 local.y_mm = place.y_mm + local.y_mm
143 local.net = conn_by_key.get((comp.ref, local.name))
144 comp.pads.append(local)
145 board.add_component(comp, place)
146
147 # 5) nets du Board
148 for net_name in sorted(net_connections):
149 conns = [(ref, str(pad)) for ref, pad in net_connections[net_name]]
150 board.nets[net_name] = Net(name=net_name, connections=conns)
151
152 ok = bool(clean)
153 logger.info("import normalisé", extra={"ok": ok, "components": len(clean),
154 "nets": len(board.nets), "warnings": len(warns)})
155 return ImportReport(ok=ok, components_imported=len(clean),
156 nets_imported=len(board.nets), warnings=warns, board=board)
