58 def store_pattern(self, pattern: Dict[str, Any]) -> str:
59 """Persiste un motif (tags, parts, nets, métriques) ; retourne son id."""
60 pattern_id = str(pattern.get("id") or uuid.uuid4().hex[:12])
61 payload = dict(pattern, id=pattern_id)
62 if self._driver is not None:
63 try:
64 with self._driver.session() as session:
65 session.run(
66 "MERGE (p:Pattern {id: $id}) SET p += $props",
67 id=pattern_id, props=payload,
68 )
69 return pattern_id
70 except Exception as exc:
71 logger.warning("écriture neo4j échouée — miroir utilisé",
72 extra={"error": str(exc)[:120]})
73 self._memory[pattern_id] = payload
74 return pattern_id

76 def find_patterns(self, tags: List[str], limit: int = 10) -> List[Dict[str, Any]]:
77 """Motifs portant au moins un des tags demandés (triés par similarité)."""
78 if self._driver is not None:
79 try:
80 with self._driver.session() as session:
81 result = session.run(
82 "MATCH (p:Pattern) WHERE ANY(t IN p.tags WHERE t IN $tags) "
83 "RETURN p LIMIT $limit",
84 tags=tags, limit=limit,
85 )
86 return [dict(record["p"]) for record in result]
87 except Exception as exc:
88 logger.warning("lecture neo4j échouée — miroir utilisé",
89 extra={"error": str(exc)[:120]})
90 wanted = set(tags)
91 hits = [p for p in self._memory.values() if wanted & set(p.get("tags", []))]
92 hits.sort(key=lambda p: -len(wanted & set(p.get("tags", []))))
93 return hits[:limit]

105 """

107 try:
108 with self._driver.session() as session:

110 return 1
111 except Exception as exc:
112 logger.warning("seed cypher échoué — parse miroir",

114 imported = 0
115 for block in _PROPS_RE.findall(cypher_text):
116 candidate = "{" + block + "}"
117 try:
118 props = json.loads(candidate.replace("'", '"'))
119 except ValueError:
120 continue
121 self.store_pattern(props)
122 imported += 1
123 return imported
