111 def submit(self, payload: dict[str, Any],
112 priority: Priority = Priority.INTERACTIVE,
113 task_id: str | None = None) -> str:
114 """Enfile une tâche — les interactives sont dépilées avant le batch nuit."""
115 self._counter += 1
116 task = QueuedTask(task_id=task_id or f"task-{self._counter}",
117 priority=priority, submitted_at=time.time(),
