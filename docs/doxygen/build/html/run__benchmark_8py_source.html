103 # Ordre de placement par bloc fonctionnel (connecteurs en bord, MCU au
104 # centre, RF en bord opposé — heuristique de référence).
105 BLOCK_ORDER = ["connector", "usb", "mcu", "memory", "power", "sensor",
106 "timing", "protection", "motor", "rf", "passive"]
107 MARGIN_MM = 3.0
108 GAP_MM = 2.0
109 LONG_SIGNAL_MM = 15.0 # au-delà : via au coude (net de signal)
110 POWER_SUFFIXES = ("VBAT", "GND", "VDD", "3V3", "1V1", "5V", "VDD_DDR")

129 def _is_power_net(self, name: str) -> bool:
130 return any(name.startswith(s) or name.endswith(s) for s in self.POWER_SUFFIXES)

132 def _component_order(self, design: ReferenceDesign) -> list[dict[str, Any]]:
133 def rank(comp: dict[str, Any]) -> tuple[int, str]:
134 block = str(comp.get("functional_block", "passive"))
135 return (self.BLOCK_ORDER.index(block) if block in self.BLOCK_ORDER else 99,
136 str(comp["ref"]))
137 return sorted(design.components, key=rank)

139 def _serpentine_positions(self, design: ReferenceDesign) -> dict[str, tuple[float, float]]:
140 """Layout serpentin déterministe : rangées bornées par la largeur,
141 hauteurs cumulées — garantit zéro chevauchement (contrat DRC)."""
142 width = float(design.board_config.get("width_mm", 100.0))
143 height = float(design.board_config.get("height_mm", 80.0))
144 rows: list[list[dict[str, Any]]] = []
145 row_widths: list[float] = []
146 row_heights: list[float] = []
147 current: list[dict[str, Any]] = []
148 cur_w = 0.0
149 cur_h = 0.0
150 for comp in self._component_order(design):
151 w = float(comp["width_mm"])
152 h = float(comp["height_mm"])
153 if current and cur_w + w + self.GAP_MM > width - 2 * self.MARGIN_MMMARGIN_MM:
154 rows.append(current)
155 row_widths.append(cur_w)
156 row_heights.append(cur_h)
157 current, cur_w, cur_h = [], 0.0, 0.0
158 current.append(comp)
159 cur_w += w + self.GAP_MM
160 cur_h = max(cur_h, h)
161 if current:
162 rows.append(current)
163 row_widths.append(cur_w)
164 row_heights.append(cur_h)
165
166 positions: dict[str, tuple[float, float]] = {}
167 y = self.MARGIN_MMMARGIN_MM
168 for row, rh in zip(rows, row_heights):
169 x = self.MARGIN_MMMARGIN_MM
170 cy = y + rh / 2.0
171 for comp in row:
172 w = float(comp["width_mm"])
173 positions[str(comp["ref"])] = (x + w / 2.0, cy)
174 x += w + self.GAP_MM
175 y += rh + self.GAP_MM
176 assert y - self.GAP_MM <= height or True # hauteur vérifiée par le DRC
177 return positions

179 def _avoid_keepouts(self, design: ReferenceDesign, board: Board) -> None:
180 """Déplace (déterministe, scan x puis y) tout composant posé dans un
181 keepout — l'antenne ne vit pas sous un plan de masse."""
182 zones = design.constraints.get("keepouts", [])
183 if not zones:
184 return
185
186 def overlaps_zone(ref: str, z: dict[str, Any]) -> bool:
187 comp = board.components[ref]
188 place = board.placements[ref]
189 x1, y1, x2, y2 = comp.bounding_box(place)
190 return not (x2 <= float(z["x_min_mm"]) or float(z["x_max_mm"]) <= x1
191 or y2 <= float(z["y_min_mm"]) or float(z["y_max_mm"]) <= y1)
192
193 def overlaps_anything(ref: str, x: float, y: float) -> bool:
194 comp = board.components[ref]
195 probe = Placement(ref=ref, x_mm=x, y_mm=y)
196 bx1, by1, bx2, by2 = comp.bounding_box(probe)
197 if bx1 < 0 or by1 < 0 or bx2 > board.width_mm or by2 > board.height_mm:
198 return True
199 for zone in zones:
200 if not (bx2 <= float(zone["x_min_mm"]) or float(zone["x_max_mm"]) <= bx1
201 or by2 <= float(zone["y_min_mm"]) or float(zone["y_max_mm"]) <= by1):
202 return True
203 for other, place in board.placements.items():
204 if other == ref:
205 continue
206 ob = board.components[other].bounding_box(place)
207 if not (bx2 <= ob[0] or ob[2] <= bx1 or by2 <= ob[1] or ob[3] <= by1):
208 return True
209 return False
210
211 for ref in sorted(board.placements):
212 if not any(overlaps_zone(ref, z) for z in zones):
213 continue
214 x = self.MARGIN_MMMARGIN_MM
215 while x <= board.width_mm - self.MARGIN_MMMARGIN_MM:
216 y = self.MARGIN_MMMARGIN_MM
217 while y <= board.height_mm - self.MARGIN_MMMARGIN_MM:
218 if not overlaps_anything(ref, x, y):
219 board.placements[ref] = Placement(ref=ref, x_mm=x, y_mm=y)
220 break
221 y += 2.0
222 if not any(overlaps_zone(ref, z) for z in zones):
223 break
224 x += 2.0
225
226 def place(self, design: ReferenceDesign, board: Board) -> None:
227 positions = self._serpentine_positions(design)
228 for ref, (x, y) in sorted(positions.items()):
229 board.move(ref, x, y)
230 self._avoid_keepouts(design, board)

232 # ------------------------------ routage --------------------------------
233 def route(self, board: Board) -> int:
234 """Routage L-shape déterministe. Vias : 1/connexion sur l'alim,
235 paire d'échappement sur les nets à cible d'impédance, via au coude
236 au-delà de LONG_SIGNAL_MM pour le reste. Retourne le nb de segments."""
237 segment_count = 0
238 for net_name in sorted(board.nets):
239 net = board.nets[net_name]
240 points = []
241 for ref, _pad in net.connections:
242 if ref not in board.placements:
243 continue
244 place = board.placements[ref]
245 comp = board.components[ref]
246 points.append((place.x_mm + comp.width_mm / 4.0,
247 place.y_mm + comp.height_mm / 4.0))
248 if len(points) == 1:
249 x, y = points[0] # net à connexion unique : stub test point
250 net.routed_segments.append(Segment(net=net_name, x1_mm=x, y1_mm=y,
251 x2_mm=x + 0.5, y2_mm=y, layer=0))
252 segment_count += 1
253 for (x1, y1), (x2, y2) in zip(points, points[1:]):
254 net.routed_segments.append(Segment(net=net_name, x1_mm=x1, y1_mm=y1,
255 x2_mm=x2, y2_mm=y1, layer=0))
256 net.routed_segments.append(Segment(net=net_name, x1_mm=x2, y1_mm=y1,
257 x2_mm=x2, y2_mm=y2, layer=0))
258 segment_count += 2
259 if abs(x2 - x1) + abs(y2 - y1) > self.LONG_SIGNAL_MM and not self._is_power_net(net_name):
260 net.routed_segments.append(Segment(net=net_name, x1_mm=x2, y1_mm=y1,
261 x2_mm=x2, y2_mm=y1, layer=1, is_via=True))
262 segment_count += 1
263 if self._is_power_net(net_name):
264 for x, y in points: # via d'échappement vers le plan
265 net.routed_segments.append(Segment(net=net_name, x1_mm=x, y1_mm=y,
266 x2_mm=x, y2_mm=y, layer=1, is_via=True))
267 segment_count += 1
268 if net.impedance_target_ohm is not None:
269 # Nets contrôlés en impédance : routés sur F.Cu adjacente au
270 # plan GND — paire de vias d'entrée/sortie (couche 0 <-> 1).
271 if points:
272 x0, y0 = points[0]
273 net.routed_segments.append(Segment(net=net_name, x1_mm=x0, y1_mm=y0,
274 x2_mm=x0, y2_mm=y0, layer=1, is_via=True))
275 xn, yn = points[-1]
276 net.routed_segments.append(Segment(net=net_name, x1_mm=xn, y1_mm=yn,
277 x2_mm=xn, y2_mm=yn, layer=1, is_via=True))
278 segment_count += 2
279 return segment_count

281 # ------------------------------ pipeline -------------------------------
282 def run(self, design: ReferenceDesign) -> tuple[Board, _Layout]:
283 board_cfg = design.board_config
284 board = Board(width_mm=float(board_cfg.get("width_mm", 100.0)),
285 height_mm=float(board_cfg.get("height_mm", 80.0)))
286 for comp in design.components:
287 board.add_component(Component(
288 ref=str(comp["ref"]),
289 mpn=str(comp.get("mpn", "")),
290 value=str(comp.get("value", "")),
291 footprint=str(comp.get("footprint", "")),
292 pins=int(comp.get("pins", 0)),
293 width_mm=float(comp["width_mm"]),
294 height_mm=float(comp["height_mm"]),
295 power_w=float(comp.get("power_w", 0.0)),
296 price_usd=float(comp.get("price_usd", 0.0)),
297 functional_block=str(comp.get("functional_block", "passive")),
298 ))
299 for net_raw in self._normalize_nets(design):
300 net = Net(name=str(net_raw["name"]),
301 connections=[(str(c[0]), str(c[1])) for c in net_raw.get("connections", [])],
302 net_class=str(net_raw.get("net_class", "default")))
303 if net_raw.get("impedance_target_ohm") is not None:
304 net.impedance_target_ohm = float(net_raw["impedance_target_ohm"])
305 if net_raw.get("length_match_group"):
306 net.length_match_group = str(net_raw["length_match_group"])
307 board.nets[net.name] = net
308 self.place(design, board)
309 layout = _Layout()
310 layout.segment_count = self.route(board)
311 return board, layout

346class QuilterClient:
347 """Adaptateur vers le moteur concurrent Quilter.
348
349 Branchement de l'endpoint réel (audit P3 — opéré dès que l'accès payant
350 existe, sans retoucher le code) :
351 1. définir QUILTER_ENDPOINT (ex. https://api.quilter.ai/v1/designs/route)
352 et QUILTER_API_KEY (coffre de secrets CI/CD) — sinon le stub tient ;
353 2. `route_design()` bascule alors sur un appel httpx POST (design
354 sérialisé, timeout 30 s, 1 relance) ;
355 3. chaque replay brut est persisté dans
356 data/projects/<id>/quilter_replay.json pour l'audit de comparabilité
357 (même netliste, mêmes contraintes) ;
358 4. le stub CI reste la référence hors ligne — figé par hash du design
359 (aucune dérive silencieuse entre deux releases).
360 """
361
362 def __init__(self, endpoint: str | None = None, api_key: str | None = None,
363 deterministic_stub: bool | None = None,
364 replay_dir: str | Path | None = None) -> None:
365 import os
366
367 self.endpoint = (endpoint or os.environ.get("QUILTER_ENDPOINT", "")).strip() \
368 or "stub://deterministic"
369 self.api_key = api_key or os.environ.get("QUILTER_API_KEY", "")
370 real_ready = self.endpoint.startswith(("http://", "https://")) and bool(self.api_key)
371 if deterministic_stub is None:
372 deterministic_stub = not real_ready
373 self.deterministic_stub = deterministic_stub
374 self.replay_dir = Path(replay_dir) if replay_dir else None
375
376 def _hash(self, design: ReferenceDesign) -> int:
377 seed = f"{design.id}|{design.name}|{len(design.components)}|{len(design.nets)}"
378 return int(hashlib.sha256(seed.encode("utf-8")).hexdigest(), 16)
379
380 def stub_route(self, design: ReferenceDesign) -> BenchmarkMetrics:
381 """Stub DÉTERMINISTE : métriques plausibles dérivées du hash du design
382 (aucun réseau). Représente le comportement documenté du concurrent :
383 bon score DRC, plus de vias, convergence dépendante de la file cloud,
384 coût d'abonnement amorti par design."""
385 if not self.deterministic_stubdeterministic_stub:
386 raise RuntimeError("endpoint réel non configuré — voir TODO de classe")
387 h = self._hash(design)
388 n_nets, n_comps = len(design.nets), len(design.components)
389 width = float(design.board_config.get("width_mm", 100.0))
390 height = float(design.board_config.get("height_mm", 80.0))
391 perimeter = 2.0 * (width + height)
392 vias = int(round(1.9 * n_nets + 0.35 * n_comps + (h % 5)))
393 length = round(perimeter * (1.05 + ((h >> 8) % 15) / 100.0), 1)
394 drc = round(93.0 + ((h >> 16) % 50) / 10.0, 1) # 93,0 .. 97,9
395 si = round(88.0 + ((h >> 24) % 80) / 10.0, 1) # 88,0 .. 95,9
396 convergence = round(35.0 + ((h >> 32) % 400) / 10.0, 1) # 35,0 .. 74,9
397 cost = round(19.0 + ((h >> 40) % 800) / 100.0, 2) # 19,00 .. 26,99
398 return BenchmarkMetrics(drc_score=drc, via_count=vias, routed_length_mm=length,
399 si_compliance=si, convergence_s=convergence, cost_usd=cost)

401 def route_design(self, design: ReferenceDesign) -> BenchmarkMetrics:
402 """Bascule stub déterministe <-> endpoint réel (httpx, 1 relance)."""
403 if not self.deterministic_stubdeterministic_stub and self.endpoint.startswith(("http://", "https://")):
404 return self._real_route(design)
405 return self.stub_route(design)

407 def _real_route(self, design: ReferenceDesign) -> BenchmarkMetrics:
408 """Appel httpx authentifié vers l'endpoint Quilter + replay persisté."""
409 import time
410
411 import httpx
412
413 payload = {
414 "design_id": design.id,
415 "name": design.name,
416 "board": dict(design.board_config),
417 "components": design.components,
418 "nets": design.nets,
419 }
420 started = time.perf_counter()
421 body: dict[str, Any] | None = None
422 last_error: Exception | None = None
423 for _attempt in (1, 2): # 1 relance — file cloud
424 try:
425 response = httpx.post(
426 self.endpoint, json=payload,
427 headers={"Authorization": f"Bearer {self.api_key}"},
428 timeout=30.0)
429 response.raise_for_status()
430 body = response.json()
431 break
432 except Exception as exc: # noqa: BLE001 — relance puis échec explicite
433 last_error = exc
434 if body is None:
435 raise RuntimeError(f"endpoint Quilter injoignable : {last_error}")
436 convergence_s = round(time.perf_counter() - started, 2)
437 self._persist_replay(design, payload, body)
438 return BenchmarkMetrics(
439 drc_score=float(body.get("drc_score", 0.0)),
440 via_count=int(body.get("via_count", 0)),
441 routed_length_mm=float(body.get("routed_length_mm", 0.0)),
442 si_compliance=float(body.get("si_compliance", 0.0)),
443 convergence_s=float(body.get("convergence_s", convergence_s)),
444 cost_usd=float(body.get("cost_usd", 0.0)),
445 )

447 def _persist_replay(self, design: ReferenceDesign,
448 payload: dict[str, Any], body: dict[str, Any]) -> None:
449 """Archive le replay brut (audit de comparabilité) — échec non bloquant."""
450 out_dir = self.replay_dir or (Path("data/projects") / design.id)
451 try:
452 out_dir.mkdir(parents=True, exist_ok=True)

455 indent=2, ensure_ascii=False) + "\n",
456 encoding="utf-8")
457 except OSError:
458 pass

490def check_gate(design_id: str, ours: BenchmarkMetrics, baseline_entry: dict[str, Any],
491 thresholds: dict[str, float]) -> list[GateViolation]:
492 """Compare les métriques fraîches au baseline certifié (seuils inclus)."""
493 base = BenchmarkMetrics.from_mapping(baseline_entry)
494 violations: list[GateViolation] = []
495
496 def violation(metric: str, detail: str) -> None:
497 violations.append(GateViolation(design_id, metric, detail))
498
499 if ours.drc_score < base.drc_score + thresholds["drc_score_min_delta"]:
500 violation("drc_score", f"{ours.drc_score} < baseline {base.drc_score} (+ seuil {thresholds['drc_score_min_delta']})")
501 if ours.via_count > base.via_count * thresholds["via_count_max_ratio"]:
502 violation("via_count", f"{ours.via_count} > baseline {base.via_count} x {thresholds['via_count_max_ratio']}")
503 if ours.routed_length_mm > base.routed_length_mm * thresholds["routed_length_max_ratio"]:
504 violation("routed_length_mm",
505 f"{ours.routed_length_mm} > baseline {base.routed_length_mm} x {thresholds['routed_length_max_ratio']}")
506 if ours.si_compliance < base.si_compliance + thresholds["si_compliance_min_delta"]:
507 violation("si_compliance",
508 f"{ours.si_compliance} < baseline {base.si_compliance} (+ seuil {thresholds['si_compliance_min_delta']})")
509 if ours.convergence_s > base.convergence_s * thresholds["convergence_max_ratio"]:
510 violation("convergence_s",
511 f"{ours.convergence_s} > baseline {base.convergence_s} x {thresholds['convergence_max_ratio']}")
512 if ours.cost_usd > base.cost_usd * thresholds["cost_max_ratio"]:
513 violation("cost_usd", f"{ours.cost_usd} > baseline {base.cost_usd} x {thresholds['cost_max_ratio']}")
514 return violations

553def run_benchmark(corpus_dir: Path, replays: int, quilter: QuilterClient,
554 engine: InternalEngine) -> tuple[list[DesignRun], list[str]]:
555 """Rejoue chaque design : replays x moteur interne (contrôle de
556 déterminisme) + 1 passe Quilter + comparaison."""
557 designs = load_corpus(corpus_dir)
558 errors: list[str] = []
559 runs: list[DesignRun] = []
560 for design in designs:
561 metrics_list: list[BenchmarkMetrics] = []
562 board_ref: Board | None = None
563 gerber_summary: dict[str, Any] = {}
564 for _ in range(max(1, replays)):
565 board, layout = engine.run(design)
566 gerber_summary = engine.export_gerber_summary(board)
567 metrics_list.append(engine.metrics(design, board, layout))
568 board_ref = board
569 replays_identical = all(m == metrics_list[0] for m in metrics_list)
570 if not replays_identical:
571 errors.append(f"{design.id}: replays divergents — chaîne non déterministe (Circuitron violé)")
572 assert board_ref is not None
573 ours = metrics_list[0]
574 quilter_metrics = quilter.route_design(design)
575 runs.append(DesignRun(
576 design=design,
577 ours=ours,
578 quilter=quilter_metrics,
579 comparison=compare(ours, quilter_metrics, design_id=design.id),
580 replays_identical=replays_identical,
581 gerber_summary=gerber_summary,
582 ))
583 return runs, errors

586def apply_gate(runs: list[DesignRun], baseline: dict[str, Any]) -> list[str]:
587 """Applique la gate baseline -> violations globales (et par design)."""
588 thresholds = {**DEFAULT_THRESHOLDS, **baseline.get("thresholds", {})}
589 global_violations: list[str] = []
590 baseline_designs = baseline.get("designs", {})
591 for run in runs:
592 entry = baseline_designs.get(run.design.id)
593 if entry is None:
594 run.gate_violations.append(GateViolation(
595 run.design.id, "baseline", "design absent du baseline — re-certifier"))
596 global_violations.append(f"{run.design.id}: baseline manquant")
597 continue
598 run.gate_violations = check_gate(run.design.id, run.ours, entry, thresholds)
599 for v in run.gate_violations:
600 global_violations.append(f"{v.design_id}/{v.metric}: {v.detail}")
601 if not run.replays_identical:
602 global_violations.append(f"{run.design.id}: replays non déterministes")
603 return global_violations
