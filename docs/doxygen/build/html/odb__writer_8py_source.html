26 def write_to(self, base_dir: Path) -> List[Path]:
27 base_dir.mkdir(parents=True, exist_ok=True)
28 written: List[Path] = []
29 for rel_path, content in self.files.items():
30 target = base_dir / rel_path

32 target.write_text(content, encoding="utf-8")

37def export_odb(board: Board, project_id: str = "project") -> OdbStructure:
38 """Construit la structure ODB++ minimale pour le design courant."""
39 files: Dict[str, str] = {}
40 copper_layers = [l.index for l in board.layers if l.is_copper]
41
42 files["job.xml"] = _job_xml(project_id, len(copper_layers))
43 files["steps/pcb/stephdr.xml"] = _step_header(board)
44 files["matrix/matrix.xml"] = _matrix(copper_layers)
45 for layer_index in copper_layers:
46 name = layer_filename(layer_index, len(copper_layers)).replace(".gbr", "")
47 files[f"steps/pcb/layers/{name}/layers"] = render_copper_layer(board, layer_index)
48 return OdbStructure(files=files)
