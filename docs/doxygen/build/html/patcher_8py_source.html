22 # ---- API ------------------------------------------------------------------
23 def patch(self, script: str, issues: list[dict[str, Any]]) -> tuple[str, list[str]]:
24 """Retourne (script_corrigé, descriptions des patches appliqués)."""
25 descriptions: list[str] = []
26 for issue in issues:
27 rule = str(issue.get("rule") or self._classify(issue))
28 handler = getattr(self, f"_patch_{rule}", None)
29 if handler is None:
30 continue
31 new_script, description = handler(script, issue)
32 if description and new_script != script:
33 script = new_script
34 descriptions.append(description)
35 elif description:
36 descriptions.append(f"{description} (idempotent — déjà présent)")
37 return script, descriptions

39 # ---- classification ---------------------------------------------------------
40 def _classify(self, issue: dict[str, Any]) -> str:
41 """Classification par règle à partir du code ERC / du message."""
42 code = str(issue.get("code", "")).upper()
43 message = str(issue.get("message", "")).lower()
44 if "pull" in message or "i2c" in message:
45 return "missing_pullup"
46 if "découplage" in message or "decoupl" in message or "decoupl" in code.lower():
47 return "missing_decoupling"
48 if "empreinte" in message or "footprint" in message or code in ("FP-MISSING", "FP-LIB"):
49 return "missing_footprint"
50 if "connexions" in message or "float" in message or code == "NET-FLOAT":
51 return "floating_net"
52 if code == "NET-REF" or "absente" in message:
53 return "unknown_ref"
54 return "generic"

119 @staticmethod
120 def _append_block(script: str, block: list[str]) -> tuple[str, bool]:
121 """Append un bloc sous le marqueur — idempotent via garde textuelle."""
122 first_line = block[0]
123 if first_line in script:
124 return script, False
125 if CORRECTION_MARKER not in script:
126 script += ("" if script.endswith("\n") else "\n") + \
127 f"\n{CORRECTION_MARKER}\n"
128 script += "\n".join(block) + "\n"
129 return script, True
