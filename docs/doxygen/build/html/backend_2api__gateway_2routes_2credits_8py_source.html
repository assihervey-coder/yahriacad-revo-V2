61@router.post("/topup")
62def topup(body: TopupRequest) -> dict[str, Any]:
63 """Recharge le solde — transaction appendée puis persistée (best-effort)."""
64 if body.amount_usd <= 0:
65 raise_http_for(ValueError("amount_usd doit être positif"))
66 ledger = get_runtime().ledger
67 ledger.balance_usd = round(ledger.balance_usd + body.amount_usd, 4)
68 tx = CreditTransaction(
69 tx_id=uuid.uuid4().hex[:12], project_id=body.project_id, item=0.0,
70 amount_usd=body.amount_usd, balance_after_usd=ledger.balance_usd,
71 ts=time.time(), user="topup")
72 ledger.transactions.append(tx)
73 ledger._persist() # noqa: SLF001 — persistance best-effort du grand livre
74 return {"tx_id": tx.tx_id, "amount_usd": tx.amount_usd,
75 "balance_usd": ledger.balance_usd}
