79def check_factory(board: Board, profile: FactoryProfile) -> List[Violation]:
80 """Vérifie le Board contre les capacités réelles de l'usine cible."""
81 out: List[Violation] = []
82
83 # largeurs de pistes / clearances usine (plus fins que les règles génériques)
84 for net in board.nets.values():
85 for seg in net.routed_segments:
86 if not seg.is_via and seg.width_mm < profile.min_trace_mm:
87 out.append(Violation(
88 rule_id=f"MF-min-trace-{profile.name.lower()}", severity="error",
89 net=net.name,
90 message=f"piste {seg.width_mm} mm < {profile.min_trace_mm} mm ({profile.name})",
91 ))
92 if seg.is_via and seg.width_mm < profile.min_via_dia_mm:
93 out.append(Violation(
94 rule_id=f"MF-min-via-{profile.name.lower()}", severity="error",
95 net=net.name,
96 message=f"via {seg.width_mm} mm < {profile.min_via_dia_mm} mm ({profile.name})",
97 ))
98
99 # nombre de couches
100 layer_count = len([l for l in board.layers if l.is_copper])
101 if layer_count > profile.max_layers:
102 out.append(Violation(
103 rule_id=f"MF-max-layers-{profile.name.lower()}", severity="error",
104 message=f"{layer_count} couches cuivre > {profile.max_layers} autorisées ({profile.name})",
105 ))
106
107 # pastilles minimales
108 for comp in board.components.values():
109 for pad in comp.pads:
110 if pad.diameter_mm < profile.min_pad_mm:
111 out.append(Violation(
112 rule_id=f"MF-min-pad-{profile.name.lower()}", severity="error",
113 component_ref=comp.ref,
114 message=f"pastille {pad.diameter_mm} mm < {profile.min_pad_mm} mm ({profile.name})",
115 ))
116 break # une violation par composant suffit
117 return out
