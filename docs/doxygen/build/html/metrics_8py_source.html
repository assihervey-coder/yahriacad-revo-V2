1"""Métriques comparées du benchmark vs Quilter + logique de gain.
2
3Six métriques par design et par moteur :
4 drc_score (0-100, supérieur = mieux) — complétude + pénalités
5 via_count (entier, inférieur = mieux) — coût usine et bruit SI
6 routed_length_mm (mm, inférieur = mieux) — longueur totale routée
7 si_compliance (0-100, supérieur = mieux) — respect impédance / length match
8 convergence_s (s, inférieur = mieux) — temps jusqu'au résultat routé
9 cost_usd (USD, inférieur = mieux) — crédits (nous) / abonnement amorti (Quilter)
10
11compare(ours, quilter) renvoie un tableau de lignes + le gain moyen en % :
12positif = notre moteur gagne sur la métrique.
13"""

97def _gain_pct(ours: float, quilter: float, metric: str) -> tuple[float, str]:
98 """Gain en % (base : valeur Quilter) + désignation du meilleur.
99
100 Convention : pour une métrique « plus petit = mieux », gain =
101 (quilter - ours) / quilter x 100 — positif si nous consommons moins.
102 Pour « plus grand = mieux », gain = (ours - quilter) / quilter x 100.
103 """
104 base = abs(quilter) or 1.0
105 if metric in LOWER_IS_BETTER:
106 gain = (quilter - ours) / base * 100.0
107 else:
108 gain = (ours - quilter) / base * 100.0
109 if abs(gain) < 0.05: # sous 0,05 % : ex æquo
110 return round(gain, 2), "tie"
111 return round(gain, 2), ("ours" if gain > 0 else "quilter")
112
113
114def compare(ours: BenchmarkMetrics, quilter: BenchmarkMetrics,
115 design_id: str = "") -> ComparisonResult:
116 """Compare les deux moteurs métrique par métrique -> tableau + gain moyen."""
117 rows: list[MetricRow] = []
118 ours_d, quilter_d = ours.to_dict(), quilter.to_dict()
119 for key in METRIC_KEYS:
120 gain, better = _gain_pct(float(ours_d[key]), float(quilter_d[key]), key)
121 rows.append(MetricRow(
122 metric=key,
123 label=METRIC_LABELS_FR.get(key, key),
124 ours=float(ours_d[key]),
125 quilter=float(quilter_d[key]),
126 gain_pct=gain,
127 better=better,
128 ))
129 mean_gain = round(sum(r.gain_pct for r in rows) / len(rows), 2)
130 return ComparisonResult(design_id=design_id, rows=rows, mean_gain_pct=mean_gain)
