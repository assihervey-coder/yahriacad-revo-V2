1"""Collecte du dataset d'entraînement — transitions réelles de l'environnement.
2
3Le dataset sert au pré-entraînement supervisé du world model torch :
4chaque ligne relie les 12 features d'une décision de placement aux trois
5conséquences RÉELLEMENT calculées par l'environnement (heuristiques
6déterministes du simulateur interne). C'est le principe « apprendre le
7monde avant d'agir dedans » de l'approche Dreamer : le réseau de runtime
8n'est jamais alimenté par des données fictives.
9
10Les archives JSONL du keeper (``data/trained_models/keeper_archive.jsonl``)
11ne contiennent pas les features brutes (seulement propositions + scores) :
12``summarize_archive`` les résume pour le rapport d'entraînement, sans jamais
13les présenter comme un dataset de régression — honnêteté des métriques.
14"""

27def collect_dataset(env: PlacementEnv, episodes: int = 30,
28 seed: int = 42) -> Tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray, Dict[str, Any]]:
29 """Rollouts aléatoires → (features X, cibles Y, actions A, rewards R, stats).
30
31 * ``X`` — (n, 12) features du composant courant à son ancre ;
32 * ``Y`` — (n, 3) conséquences réelles [congestion, thermique, si] ;
33 * ``A`` — (n, 4) vecteur d'action appliqué (dx, dy, rot_idx, layer) ;
34 * ``R`` — (n,) récompense de chaque transition.
35 """
36 rows_X: List[np.ndarray] = []
37 rows_Y: List[np.ndarray] = []
38 rows_A: List[np.ndarray] = []
39 rows_R: List[float] = []
40 n_legal, n_steps = 0, 0
41 for _episode in range(max(1, episodes)):
42 state = env.reset()
43 done = False
44 while not done:
45 action = env.random_action()
46 next_state, reward, done, info = env.step(action)
47 rows_X.append(np.asarray(state, dtype=np.float64))
48 cons = info["consequences"]
49 rows_Y.append(np.array([cons["congestion_score"],
50 cons["thermal_rise_estimate"],
51 cons["si_risk"]], dtype=np.float64))
52 rows_A.append(np.asarray(action, dtype=np.float64))
53 rows_R.append(float(reward))
54 n_legal += 1 if info.get("legal") else 0
55 n_steps += 1
56 state = next_state
57 stats = {"episodes": episodes, "transitions": n_steps, "legal_ratio":
58 round(n_legal / n_steps, 3) if n_steps else 0.0,
59 "mean_reward": round(float(np.mean(rows_R)), 4) if rows_R else 0.0}
60 return (np.vstack(rows_X) if rows_X else np.zeros((0, 12)),
61 np.vstack(rows_Y) if rows_Y else np.zeros((0, 3)),
62 np.vstack(rows_A) if rows_A else np.zeros((0, 4)),
63 np.array(rows_R, dtype=np.float64), stats)

66def summarize_archive(path: Optional[Path]) -> Dict[str, Any]:
67 """Résumé d'une archive keeper JSONL — contexte du rapport, pas un dataset."""
68 if path is None or not Path(path).exists():
69 return {"available": False}
70 iterations, kept, scores = 0, 0, []
71 try:
72 with Path(path).open("r", encoding="utf-8") as handle:
73 for line in handle:
74 line = line.strip()
75 if not line:
76 continue
77 try:
78 record = json.loads(line)
79 except json.JSONDecodeError:
80 continue
81 iterations += 1
82 kept += 1 if record.get("kept") else 0
83 if isinstance(record.get("eval_score"), (int, float)):
84 scores.append(float(record["eval_score"]))
85 except OSError:
86 return {"available": False}
87 return {"available": True, "iterations": iterations, "kept": kept,
88 "keep_ratio": round(kept / iterations, 3) if iterations else 0.0,
89 "mean_eval_score": round(sum(scores) / len(scores), 4) if scores else None}
