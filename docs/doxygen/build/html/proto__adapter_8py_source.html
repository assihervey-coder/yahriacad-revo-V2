25def board_from_proto(pb_board) -> Board:
26 """common_pb2.Board → modèle interne Board (conversions tolérantes)."""
27 from common.design_model import Component, Net, Pad, Placement, Segment
28
29 board = Board(width_mm=pb_board.width_mm or 100.0, height_mm=pb_board.height_mm or 80.0)
30 for comp in pb_board.components:
31 pads = [
32 Pad(name=p.name, x_mm=p.x_mm, y_mm=p.y_mm)
33 for p in getattr(comp, "pads", [])
34 ]
35 board.components[comp.ref] = Component(
36 ref=comp.ref, mpn=comp.mpn, value=comp.value, footprint=comp.footprint,
37 pins=comp.pins, width_mm=comp.width_mm or 5.0, height_mm=comp.height_mm or 5.0,
38 power_w=comp.power_w, price_usd=comp.price_usd, stock=comp.stock,
39 functional_block=comp.functional_block, pads=pads,
40 )
41 for pl in pb_board.placements:
42 board.placements[pl.ref] = Placement(
43 ref=pl.ref, x_mm=pl.x_mm, y_mm=pl.y_mm,
44 rotation_deg=pl.rotation_deg, layer=pl.layer, locked=pl.locked,
45 )
46 for net in pb_board.nets:
47 segments = [
48 Segment(
49 net=net.name, x1_mm=s.start.x_mm, y1_mm=s.start.y_mm,
50 x2_mm=s.end.x_mm, y2_mm=s.end.y_mm, layer=s.layer,
51 width_mm=s.width_mm or 0.2, is_via=s.is_via,
52 )
53 for s in net.routed_segments
54 ]
55 board.nets[net.name] = Net(
56 name=net.name,
57 connections=[tuple(c.split("/", 1)) for c in net.connections if "/" in c],
58 net_class=net.net_class or "default",
59 impedance_target_ohm=(net.impedance_target_ohm or None),
60 length_match_group=(net.length_match_group or None),
61 routed_segments=segments,
62 )
63 return board

82 def Simulate(self, request, context): # noqa: N802 — nom du contrat proto
83 simulator_pb2, _, common_pb2 = _load_stubs()
84 board = board_from_proto(request.board) if request.board and request.board.components else None
85 if board is None:
86 context.abort(3, "board absente du SimulateRequest") # INVALID_ARGUMENT
87
88 def stream() -> Iterator:
89 for partial in self._simulator.run(
90 board, physics=request.physics or "multi_physics",
91 streaming=request.streaming, project_id=request.design.project_id,
92 design_version=request.design.version or 1,
93 ):
94 event = simulator_pb2.SimulateEvent()
95 if partial.phase == "thermal" and partial.thermal is not None:
96 thermal = event.thermal
97 thermal.max_temp_c = partial.thermal.max_temp_c
98 for hotspot in partial.thermal.hotspots:
99 area = thermal.hotspots.add().area
100 area.x_min_mm, area.y_min_mm = hotspot.x_min_mm, hotspot.y_min_mm
101 area.x_max_mm, area.y_max_mm = hotspot.x_max_mm, hotspot.y_max_mm
102 thermal.hotspots[-1].temp_c = hotspot.temp_c
103 thermal.hotspots[-1].component_ref = hotspot.component_ref
104 elif partial.phase == "em" and partial.em is not None:
105 em = event.em
106 em.crosstalk_max_db = partial.em.crosstalk_max_db
107 em.ground_bounce_mv = partial.em.ground_bounce_mv
108 for zone in partial.em.emi_zones:
109 bb = em.emi_zones.add()
110 bb.x_min_mm, bb.y_min_mm = zone.x_min_mm, zone.y_min_mm
111 bb.x_max_mm, bb.y_max_mm = zone.x_max_mm, zone.y_max_mm
112 elif partial.phase == "signal_integrity" and partial.si is not None:
113 si = event.si
114 si.eye_opening_pct = partial.si.eye_opening_min_pct
115 si.overshoot_mv = partial.si.overshoot_max_mv
116 si.ir_drop_mv = partial.ir.ir_drop_mv if partial.ir else 0.0
117 si.failing_nets.extend(partial.si.failing_nets)
118 else:
119 _fill_metrics(event.progress, board)
120 event.progress.convergence_s = partial.metrics.get("elapsed_ms", 0.0) / 1000.0
121 yield event
122
123 return stream()
