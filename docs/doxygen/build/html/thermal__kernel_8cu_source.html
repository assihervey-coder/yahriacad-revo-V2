2// thermal_kernel.cu — noyau CUDA de diffusion thermique 2D (schéma explicite).
3//
4// Miroir exact du schéma numpy de thermal_sim/diffusion.py : les coefficients
5// par cellule (conduction c_int, source c_src, convection c_border) sont
6// précalculés côté hôte (numpy) et passés au kernel — le kernel n'applique
7// qu'un pas FTCS sur un stencil 5 points. Le binding Python (ctypes) appelle
8// `thermal_step_cuda` en boucle jusqu'à convergence.
9//
10// Choix d'implémentation :
11// - tuiles partagées 32×32 + halos d'une cellule (34×34 en shared memory) ;
12// - halos CLAMPÉS au bord de la carte (condition adiabatique — identique au
13// np.pad(mode="edge") du fallback numpy) ;
14// - la perte vers l'ambiant passe par c_conv (convection de Robin appliquée
15// aux deux faces de la carte, uniforme), cohérent avec la version numpy ;
16// - float32 partout : précision largement suffisante pour une carte < 200 °C.
17//
18// API C exportée (ctypes — voir README.md pour les signatures Python) :
19// int thermal_available(void);
20// int thermal_step_cuda(const float* temp, const float* c_int, const float* c_src,
21// float* out, int nx, int ny, float t_amb, float c_conv);
22// int thermal_run_cuda(const float* temp, const float* c_int, const float* c_src,
23// float* out, int nx, int ny, float t_amb, float c_conv,
24// int n_steps);
25// Codes de retour : 0 = succès, négatif = erreur CUDA (allocation/copie/lancement).

33// ---------------------------------------------------------------------------
34// Kernel : un pas FTCS sur la grille (ny × nx), row-major (index = y*nx + x).
35// ---------------------------------------------------------------------------
36extern "C" __global__ void thermal_step_kernel(
37 const float* __restrict__ temp, // température courante (°C)
38 const float* __restrict__ c_int, // coef. de conduction par cellule (sans unité)
39 const float* __restrict__ c_src, // apport des sources par cellule (K par pas)
40 float* __restrict__ out, // température suivante (°C)
41 int nx, int ny,
42 float t_amb, // température ambiante (°C)
43 float c_conv) // coef. de convection surfacique (sans unité)
44{
45 // Tuile partagée avec halos — réduit les accès globaux d'un facteur ~4.
46 __shared__ float tile[TILE + 2][TILE + 2];
47
48 const int x = blockIdx.x * TILE + threadIdx.x;
49 const int y = blockIdx.y * TILE + threadIdx.y;
50 const int tx = threadIdx.x + 1;
51 const int ty = threadIdx.y + 1;
52 const bool inside = (x < nx && y < ny);
53
54 // Chargement du centre — coordonnées clampées aux bords (adiabatique).
55 const int xc = max(0, min(nx - 1, x));
56 const int yc = max(0, min(ny - 1, y));
57 tile[ty][tx] = temp[yc * nx + xc];
58
59 // Halos : chaque thread de bord charge une cellule voisine (clampée).
60 if (threadIdx.x == 0)
61 tile[ty][tx - 1] = temp[yc * nx + max(0, xc - 1)];
62 if (threadIdx.x == TILE - 1)
63 tile[ty][tx + 1] = temp[yc * nx + min(nx - 1, xc + 1)];
64 if (threadIdx.y == 0)
65 tile[ty - 1][tx] = temp[max(0, yc - 1) * nx + xc];
66 if (threadIdx.y == TILE - 1)
67 tile[ty + 1][tx] = temp[min(ny - 1, yc + 1) * nx + xc];
68
69 __syncthreads();
70
71 if (!inside)
72 return;
73
74 const float t = tile[ty][tx];
75 const float lap = tile[ty][tx - 1] + tile[ty][tx + 1]
76 + tile[ty - 1][tx] + tile[ty + 1][tx] - 4.0f * t;
77
78 float next = t + c_int[y * nx + x] * lap + c_src[y * nx + x];
79 // convection de Robin : refroidissement des deux faces vers l'ambiant
80 if (c_conv > 0.0f)
81 next += c_conv * (t_amb - t);
82
83 out[y * nx + x] = next;
84}

98// ---------------------------------------------------------------------------
99// Wrapper hôte : un pas (les tableaux hôte sont copiés aller-retour).
100// ---------------------------------------------------------------------------
101extern "C" int thermal_step_cuda(
102 const float* h_temp, const float* h_c_int, const float* h_c_src,
103 float* h_out, int nx, int ny, float t_amb, float c_conv)
104{
105 const size_t bytes = (size_t)nx * (size_t)ny * sizeof(float);
106 float *d_temp = nullptr, *d_c_int = nullptr, *d_c_src = nullptr, *d_out = nullptr;
107
108 if (cuda_check(cudaMalloc(&d_temp, bytes), "cudaMalloc d_temp")) return -1;
109 if (cuda_check(cudaMalloc(&d_c_int, bytes), "cudaMalloc d_c_int")) return -2;
110 if (cuda_check(cudaMalloc(&d_c_src, bytes), "cudaMalloc d_c_src")) return -3;
111 if (cuda_check(cudaMalloc(&d_out, bytes), "cudaMalloc d_out")) return -4;
112
113 int rc = 0;
114 if (cuda_check(cudaMemcpy(d_temp, h_temp, bytes, cudaMemcpyHostToDevice), "H2D temp")) rc = -5;
115 if (!rc && cuda_check(cudaMemcpy(d_c_int, h_c_int, bytes, cudaMemcpyHostToDevice), "H2D c_int")) rc = -6;
116 if (!rc && cuda_check(cudaMemcpy(d_c_src, h_c_src, bytes, cudaMemcpyHostToDevice), "H2D c_src")) rc = -7;
117
118 if (!rc) {
119 dim3 block(TILE, TILE);
120 dim3 grid((nx + TILE - 1) / TILE, (ny + TILE - 1) / TILE);
121 thermal_step_kernel<<<grid, block>>>(d_temp, d_c_int, d_c_src, d_out,
122 nx, ny, t_amb, c_conv);
123 if (cuda_check(cudaGetLastError(), "lancement kernel")) rc = -8;
124 else if (cuda_check(cudaDeviceSynchronize(), "synchronisation")) rc = -9;
125 }
126 if (!rc && cuda_check(cudaMemcpy(h_out, d_out, bytes, cudaMemcpyDeviceToHost), "D2H out"))
127 rc = -10;
128
129 cudaFree(d_temp); cudaFree(d_c_int); cudaFree(d_c_src); cudaFree(d_out);
130 return rc;
131}

133// ---------------------------------------------------------------------------
134// Wrapper hôte : n pas en double-buffering GPU (pas de va-et-vient PCIe).
135// ---------------------------------------------------------------------------
136extern "C" int thermal_run_cuda(
137 const float* h_temp, const float* h_c_int, const float* h_c_src,
138 float* h_out, int nx, int ny, float t_amb, float c_conv, int n_steps)
139{
140 const size_t bytes = (size_t)nx * (size_t)ny * sizeof(float);
141 float *d_a = nullptr, *d_b = nullptr, *d_ci = nullptr, *d_cs = nullptr;
142
143 if (cuda_check(cudaMalloc(&d_a, bytes), "cudaMalloc d_a")) return -1;
144 if (cuda_check(cudaMalloc(&d_b, bytes), "cudaMalloc d_b")) return -2;
145 if (cuda_check(cudaMalloc(&d_ci, bytes), "cudaMalloc d_ci")) return -3;
146 if (cuda_check(cudaMalloc(&d_cs, bytes), "cudaMalloc d_cs")) return -4;
147
148 int rc = 0;
149 if (cuda_check(cudaMemcpy(d_a, h_temp, bytes, cudaMemcpyHostToDevice), "H2D temp")) rc = -5;
150 if (!rc && cuda_check(cudaMemcpy(d_ci, h_c_int, bytes, cudaMemcpyHostToDevice), "H2D c_int")) rc = -6;
151 if (!rc && cuda_check(cudaMemcpy(d_cs, h_c_src, bytes, cudaMemcpyHostToDevice), "H2D c_src")) rc = -7;
152
153 if (!rc) {
154 dim3 block(TILE, TILE);
155 dim3 grid((nx + TILE - 1) / TILE, (ny + TILE - 1) / TILE);
156 for (int step = 0; step < n_steps && !rc; ++step) {
157 thermal_step_kernel<<<grid, block>>>(d_a, d_ci, d_cs, d_b,
158 nx, ny, t_amb, c_conv);
159 if (cuda_check(cudaGetLastError(), "lancement kernel")) { rc = -8; break; }
160 // double buffer : a ← b (échange de pointeurs, zéro copie)
161 float* tmp = d_a; d_a = d_b; d_b = tmp;
162 }
163 if (!rc && cuda_check(cudaDeviceSynchronize(), "synchronisation")) rc = -9;
164 if (!rc && cuda_check(cudaMemcpy(h_out, d_a, bytes, cudaMemcpyDeviceToHost), "D2H out"))
165 rc = -10;
166 }
167
168 cudaFree(d_a); cudaFree(d_b); cudaFree(d_ci); cudaFree(d_cs);
169 return rc;
170}
