64 "rotation_deg": p.rotation_deg, "layer": p.layer, "locked": p.locked}

68 "routed_length_mm": round(n.routed_length_mm, 1), "via_count": n.via_count,
69 "connections": [list(c) for c in n.connections]}

71 "zones": [
72 {"name": z.name, "kind": z.kind, "bbox": [z.x_min_mm, z.y_min_mm,

181 def apply_surgical_edit(self, project_id: str, targets: list[str],
182 transformation: dict[str, Any],
183 author: str = "human:surgical",
184 dry_run: bool = False) -> dict[str, Any]:
185 """Bounding box d'impact → verrou humain → edit → revue accept/reject."""
186 state = self.state_manager.get_or_create(project_id)
187 bbox = self._impact_bbox(state.board, targets)
188 lock = self.state_manager.acquire_zone(project_id, bbox, owner=author, ttl_s=120.0)
189 if lock is None:
190 raise ZoneConflictError("zone verrouillée par rl_agent — l'agent attendra")
191 try:
192 import copy
193 before = copy.deepcopy(state.board)
194 applied = self.super_agent.adapter_call(
195 "scoped_edit", state.board, targets, transformation)
196 result = {"changed_refs": applied.value.get("changed_refs", []),
197 "source": applied.source, "impact_bbox": list(bbox.__dict__.values())}
198 if not dry_run:
199 entry = self.state_manager.commit_delta(
200 project_id, before, author,
201 f"Modification chirurgicale de {result['changed_refs']}",
202 result=result)
203 result["entry"] = entry
204 finally:
205 self.state_manager.release_zone(project_id, lock.ticket_id)
206 result["review"] = {"status": "accepted" if not dry_run else "pending_accept",
207 "targets": targets, "transformation": transformation}
208 return result

210 @staticmethod
211 def _impact_bbox(board: Board, targets: list[str], margin_mm: float = 2.0) -> BBox:
212 """Bounding box d'impact = union des empreintes des cibles + marge."""
213 xs: list[float] = []
214 ys: list[float] = []
215 for ref in targets:
216 comp = board.components.get(ref)
217 place = board.placements.get(ref)
218 if comp is None or place is None:
219 continue
220 x0, y0, x1, y1 = comp.bounding_box(place)
221 xs += [x0, x1]
222 ys += [y0, y1]
223 if not xs:
224 return BBox(0.0, 0.0, board.width_mm, board.height_mm)
225 return BBox(max(0.0, min(xs) - margin_mm), max(0.0, min(ys) - margin_mm),
226 min(board.width_mm, max(xs) + margin_mm),
227 min(board.height_mm, max(ys) + margin_mm))

257 """Upload « multipart simulé » → adaptateur parser → résumé stocké."""
258 try:
