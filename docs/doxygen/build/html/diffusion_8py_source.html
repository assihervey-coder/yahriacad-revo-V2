1"""Diffusion thermique 2D — schéma explicite (FTCS) vectorisé numpy, pas de temps stable CFL.
2
3Modèle :
4 - grille cartésienne (cell_size_mm), sources = composants.power_w réparties
5 uniformément sur les cellules de leur empreinte ;
6 - conductivité locale accrue autour des vias thermiques (facteur VIA_BOOST) ;
7 - convection surfacique (condition de Robin) appliquée aux deux faces de la
8 carte — les bords de carte eux-mêmes restent adiabatiques (halos répliqués) ;
9 - dt borné par la condition CFL : dt ≤ CFL·dx²/(4·α_max).
10
11Le même stencil est implémenté côté CUDA (kernels/thermal_kernel.cu, api C
12`thermal_step_cuda` appelable par ctypes). Si la librairie est compilée et
13ENABLE_CUDA actif, la boucle temporelle est déléguée au GPU ; sinon (et en cas
14de moindre erreur), le schéma numpy prend le relais — le résultat est identique
15car les coefficients par cellule (c_int, c_src, c_conv) sont calculés une seule
16fois en numpy et partagés entre les deux backends.
17"""

140 self, temp: np.ndarray, c_int: np.ndarray, c_src: np.ndarray,
141 out: np.ndarray, t_amb: float, c_conv: float,
142 ) -> None:
143 """Un pas FTCS sur GPU (tableaux float32 C-contigus)."""
144 nx, ny = temp.shape[1], temp.shape[0]

199def _cell_region(
200 x0: float, y0: float, x1: float, y1: float, cell: float, nx: int, ny: int,
201) -> Tuple[int, int, int, int]:
202 """Convertit un rectangle mm → indices de cellules inclusifs (col0, col1, row0, row1)."""
203 c0 = int(np.clip(math.floor(x0 / cell), 0, nx - 1))
204 c1 = int(np.clip(math.ceil(x1 / cell) - 1, 0, nx - 1))
205 r0 = int(np.clip(math.floor(y0 / cell), 0, ny - 1))
206 r1 = int(np.clip(math.ceil(y1 / cell) - 1, 0, ny - 1))
207 return c0, c1, r0, r1

211class _Mesh:
212 """Maillage préparé : diffusivité, sources, rectangles de composants."""
213
214 nx: int
215 ny: int
216 cell_mm: float
217 alpha: np.ndarray # diffusivité locale (m²/s), boostée par les vias
218 src_power: np.ndarray # W par cellule
219 comp_cells: Dict[str, Tuple[int, int, int, int]]
220 via_count: int

223def _build_mesh(board, cell_size_mm: float) -> _Mesh:
224 """Maillage : sources = power_w réparties sur les cellules de l'empreinte ; vias = boost α."""
225 nx = max(4, int(round(board.width_mm / cell_size_mm)))
226 ny = max(4, int(round(board.height_mm / cell_size_mm)))
227 alpha = np.full((ny, nx), ALPHA_DEFAULT, dtype=np.float64)
228 src_power = np.zeros((ny, nx), dtype=np.float64)
229 comp_cells: Dict[str, Tuple[int, int, int, int]] = {}
230
231 for ref, comp in board.components.items():
232 placement = board.placements.get(ref)
233 if placement is None:
234 continue
235 x0, y0, x1, y1 = comp.bounding_box(placement)
236 x0 = max(0.0, x0); y0 = max(0.0, y0)
237 x1 = min(board.width_mm, x1); y1 = min(board.height_mm, y1)
238 if x1 <= x0 or y1 <= y0:
239 continue
240 c0, c1, r0, r1 = _cell_region(x0, y0, x1, y1, cell_size_mm, nx, ny)
241 comp_cells[ref] = (c0, c1, r0, r1)
242 if comp.power_w > 0:
243 n_cells = (c1 - c0 + 1) * (r1 - r0 + 1)
244 src_power[r0:r1 + 1, c0:c1 + 1] += comp.power_w / n_cells
245
246 # vias thermiques : un via concentre la chaleur entre couches → conductivité locale accrue
247 via_count = 0
248 for net in board.nets.values():
249 for seg in net.routed_segments:
250 if not seg.is_via:
251 continue
252 via_count += 1
253 c0, c1, r0, r1 = _cell_region(
254 seg.x1_mm - cell_size_mm, seg.y1_mm - cell_size_mm,
255 seg.x1_mm + cell_size_mm, seg.y1_mm + cell_size_mm,
256 cell_size_mm, nx, ny,
257 )
258 alpha[r0:r1 + 1, c0:c1 + 1] *= VIA_BOOST
259
260 return _Mesh(nx=nx, ny=ny, cell_mm=cell_size_mm, alpha=alpha,
261 src_power=src_power, comp_cells=comp_cells, via_count=via_count)

264def _timestep(mesh: _Mesh, dx_m: float) -> float:
265 """Pas de temps CFL-stable : dt ≤ CFL·dx²/(4·α_max)."""
266 return CFL * dx_m * dx_m / (4.0 * float(mesh.alpha.max()))

269def _coefficients(mesh: _Mesh, dt: float, dx_m: float, h_conv: float, thickness_m: float):
270 """Coefficients par cellule partagés numpy/CUDA :
271 c_int (conduction, sans unité), c_src (K par pas), c_conv (convection des deux
272 faces, sans unité) — condition de Robin uniforme."""
273 c_int = (mesh.alpha * dt) / (dx_m * dx_m)
274 cell_volume = thickness_m * dx_m * dx_m
275 c_src = (mesh.src_power * dt) / (RHO_CP_DEFAULT * cell_volume)
276 c_conv = (2.0 * h_conv * dt) / (RHO_CP_DEFAULT * thickness_m) # deux faces
277 return c_int, c_src, c_conv

280def _step_numpy(
281 T: np.ndarray, c_int: np.ndarray, c_src: np.ndarray, c_conv: float, t_amb: float,
282) -> np.ndarray:
283 """Un pas FTCS vectorisé — halos adiabatiques (mode edge) + convection surfacique."""
284 padded = np.pad(T, 1, mode="edge")
285 lap = (padded[2:, 1:-1] + padded[:-2, 1:-1]
286 + padded[1:-1, 2:] + padded[1:-1, :-2] - 4.0 * T)
287 # condition de Robin : refroidissement des deux faces vers l'ambiant
288 return T + c_int * lap + c_src + c_conv * (t_amb - T)

292 """Choix du backend thermique : 'auto' → ctypes CUDA si présent, sinon numpy.

310def _detect_hotspots(
311 T: np.ndarray, mesh: _Mesh, board, t_amb: float,
312 threshold_delta_c: float = HOTSPOT_DELTA_C,
313) -> List[Hotspot]:
314 """Hotspots par composant dissipateur + violations des zones thermiques déclarées."""
315 cell = mesh.cell_mm
316 hotspots: List[Hotspot] = []
317
318 for ref, (c0, c1, r0, r1) in mesh.comp_cells.items():
319 comp = board.components[ref]
320 if comp.power_w <= 0:
321 continue
322 tmax = float(T[r0:r1 + 1, c0:c1 + 1].max())
323 if tmax >= t_amb + threshold_delta_c:
324 hotspots.append(Hotspot(
325 x_min_mm=round(c0 * cell, 2), y_min_mm=round(r0 * cell, 2),
326 x_max_mm=round((c1 + 1) * cell, 2), y_max_mm=round((r1 + 1) * cell, 2),
327 temp_c=round(tmax, 2), component_ref=ref, kind="component",
328 ))
329
330 # zones thermiques déclarées sur la carte (consigne max_temp_c)
331 for zone in getattr(board, "zones", []):
332 if zone.kind != "thermal" or zone.max_temp_c is None:
333 continue
334 c0, c1, r0, r1 = _cell_region(
335 zone.x_min_mm, zone.y_min_mm, zone.x_max_mm, zone.y_max_mm, cell, mesh.nx, mesh.ny,
336 )
337 tmax = float(T[r0:r1 + 1, c0:c1 + 1].max())
338 if tmax > zone.max_temp_c:
339 hotspots.append(Hotspot(
340 x_min_mm=zone.x_min_mm, y_min_mm=zone.y_min_mm,
341 x_max_mm=zone.x_max_mm, y_max_mm=zone.y_max_mm,
342 temp_c=round(tmax, 2), component_ref="", kind="zone",
343 limit_c=zone.max_temp_c,
344 ))
345
346 hotspots.sort(key=lambda h: -h.temp_c)
347 return hotspots

353def steady_state(
354 board,
355 cell_size_mm: float = 1.0,
356 t_amb_c: float = T_AMB_DEFAULT,
357 h_conv: float = H_CONV_DEFAULT,
358 thickness_mm: float = BOARD_THICKNESS_MM,
359 max_iter: int = 20000,
360 tol_c: float = 1e-3,
361 backend: str = "auto",
362 enable_cuda: Optional[bool] = None,
363) -> ThermalResult:
364 """Régime permanent : itère jusqu'à stabilisation (ΔT max < tol_c) ou max_iter.
365
366 Retourne la carte de température et la liste des hotspots (composants et
367 zones thermiques violées) — consommée par le multi_physics_loop et le
368 service gRPC simulator.
369 """
370 t0 = time.perf_counter()
371 mesh = _build_mesh(board, cell_size_mm)
372 dx_m = cell_size_mm * 1e-3
373 dt = _timestep(mesh, dx_m)
374 c_int, c_src, c_conv = _coefficients(mesh, dt, dx_m, h_conv, thickness_mm * 1e-3)
375 c_int32 = c_int.astype(np.float32)
376 c_src32 = c_src.astype(np.float32)
377
378 kernel, backend_name = _resolve_backend(backend, enable_cuda)
379 T = np.full((mesh.ny, mesh.nx), t_amb_c, dtype=np.float64)
380 T32 = T.astype(np.float32)
381 out32 = np.empty_like(T32)
382
383 iterations = 0
384 converged = False
385 if isinstance(kernel, CudaThermalKernel):
386 # boucle GPU — double buffer via tableaux ctypes (float32, C-contigus)
387 for iterations in range(1, max_iter + 1):
388 kernel.step(T32, c_int32, c_src32, out32, t_amb_c, c_conv)
389 delta = float(np.abs(out32 - T32).max())
390 T32, out32 = out32, T32
391 if iterations % 25 == 0 and delta < tol_c:
392 converged = True
393 break
394 T = T32.astype(np.float64)
395 else:
396 for iterations in range(1, max_iter + 1):
397 Tn = _step_numpy(T, c_int, c_src, c_conv, t_amb_c)
398 delta = float(np.abs(Tn - T).max())
399 T = Tn
400 if iterations % 25 == 0 and delta < tol_c:
401 converged = True
402 break
403
404 hotspots = _detect_hotspots(T, mesh, board, t_amb_c)
405 return ThermalResult(
406 max_temp_c=round(float(T.max()), 2),
407 ambient_c=t_amb_c,
408 hotspots=hotspots,
409 cell_size_mm=cell_size_mm,
410 grid_nx=mesh.nx, grid_ny=mesh.ny,
411 iterations=iterations,
412 elapsed_s=round(time.perf_counter() - t0, 4),
413 sim_time_s=round(iterations * dt, 3),
414 converged=converged,
415 backend=backend_name,
416 via_count=mesh.via_count,
417 temperature_map=T,
418 )

421def transient(
422 board,
423 t_s: float,
424 frames: int = 8,
425 cell_size_mm: float = 1.0,
426 t_amb_c: float = T_AMB_DEFAULT,
427 h_conv: float = H_CONV_DEFAULT,
428 thickness_mm: float = BOARD_THICKNESS_MM,
429 backend: str = "auto",
430 enable_cuda: Optional[bool] = None,
431) -> Iterator[ThermalResult]:
432 """Transitoire sur t_s secondes — générateur de résultats partiels (streaming gRPC).
433
434 Yield ~`frames` ThermalResult intermédiaires (converged=False) puis l'état
435 final. Utilisé par Simulator.run(physics="thermal", streaming=True).
436 """
437 mesh = _build_mesh(board, cell_size_mm)
438 dx_m = cell_size_mm * 1e-3
439 dt = _timestep(mesh, dx_m)
440 c_int, c_src, c_conv = _coefficients(mesh, dt, dx_m, h_conv, thickness_mm * 1e-3)
441 kernel, backend_name = _resolve_backend(backend, enable_cuda)
442
443 T = np.full((mesh.ny, mesh.nx), t_amb_c, dtype=np.float64)
444 total_steps = max(1, int(round(t_s / dt)))
445 snapshot_every = max(1, total_steps // max(1, frames))
446 t_start = time.perf_counter()
447
448 for step_idx in range(1, total_steps + 1):
449 if isinstance(kernel, CudaThermalKernel):
450 c_int32 = c_int.astype(np.float32)
451 c_src32 = c_src.astype(np.float32)
452 T32 = T.astype(np.float32)
453 out32 = np.empty_like(T32)
454 kernel.step(T32, c_int32, c_src32, out32, t_amb_c, c_conv)
455 T = out32.astype(np.float64)
456 else:
457 T = _step_numpy(T, c_int, c_src, c_conv, t_amb_c)
458 if step_idx % snapshot_every == 0 or step_idx == total_steps:
459 yield ThermalResult(
460 max_temp_c=round(float(T.max()), 2),
461 ambient_c=t_amb_c,
462 hotspots=_detect_hotspots(T, mesh, board, t_amb_c),
463 cell_size_mm=cell_size_mm,
464 grid_nx=mesh.nx, grid_ny=mesh.ny,
465 iterations=step_idx,
466 elapsed_s=round(time.perf_counter() - t_start, 4),
467 sim_time_s=round(step_idx * dt, 3),
468 converged=(step_idx == total_steps),
469 backend=backend_name,
470 via_count=mesh.via_count,
471 temperature_map=T,
472 )
