74 def acquire(self, project_id: str, bbox: BBox, owner: str,
75 ttl_s: float = 60.0) -> Optional[ZoneLock]:
76 """Acquiert un verrou ; retourne None si le demandeur doit ATTENDRE.
77
78 Sémantique :
79 - zone libre → verrou accordé ;
80 - même owner → réentrant (TTL prolongé) ;
81 - rl_agent vs human:surgical → l'HUMAIN gagne : si le demandeur est
82 humain, le verrou RL est révoqué (préemption) ; si le demandeur est
83 rl_agent → None (il attend, sans lever d'erreur) ;
84 - toute autre collision → ZoneConflictError.
85 """
86 with self._mutex_mutex:
87 self._sweep(project_id)
88 active = self._locks.setdefault(project_id, [])
89 for lock in list(active):
90 if not bbox.intersects(lock.bbox) or lock.owner == owner:
91 continue
92 human_involved = _is_human(lock.owner) or _is_human(owner)
93 rl_involved = RL_AGENT_OWNER in {lock.owner, owner}
94 if human_involved and rl_involved:
95 if _is_human(owner):
96 active.remove(lock) # préemption : l'humain gagne
97 logger.info("verrou RL préempté par un humain",
98 extra={"project_id": project_id, "owner": lock.owner})
99 continue
100 logger.info("rl_agent attend un verrou humain",
101 extra={"project_id": project_id})
102 return None # rl_agent attend
103 raise ZoneConflictError(
104 f"zone {bbox} déjà verrouillée par {lock.owner!r} (demandeur {owner!r})")
105 lock = ZoneLock(ticket_id=uuid.uuid4().hex[:12], project_id=project_id,
106 bbox=bbox, owner=owner, ttl_s=ttl_s)
107 active.append(lock)
108 return lock
